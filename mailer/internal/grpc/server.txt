package grpc

import (
	"context"
	"middleman/internal/errorsotel"
	"middleman/mailer/internal/application"
	"middleman/mailer/internal/application/commands"
	"middleman/mailer/internal/domain"
	"middleman/mailer/mailerpb"

	"github.com/google/uuid"
	"go.opentelemetry.io/otel/attribute"
	"go.opentelemetry.io/otel/codes"
	"go.opentelemetry.io/otel/trace"
	"google.golang.org/grpc"
)

type server struct {
	app application.App
	mailerpb.UnimplementedMailerServiceServer
}

var _ mailerpb.MailerServiceServer = (*server)(nil)

func RegisterServer(_ context.Context, app application.App, registrar grpc.ServiceRegistrar) error {
	mailerpb.RegisterMailerServiceServer(registrar, server{app: app})
	return nil
}

func (s server) CreateMailer(ctx context.Context, request *mailerpb.CreateMailerRequest) (*mailerpb.CreateMailerResponse, error) {
	span := trace.SpanFromContext(ctx)

	mailerID := uuid.New().String()

	span.SetAttributes(
		attribute.String("MailerID", mailerID),
	)

	err := s.app.CreateMailer(ctx, commands.CreateMailer{
		ID:       mailerID,
		ItemID:   request.GetItemId(),
		ItemType: domain.ItemType(request.GetItemType()),
		UserID:   request.GetUserId(),
	})
	if err != nil {
		span.RecordError(err, trace.WithAttributes(errorsotel.ErrAttrs(err)...))
		span.SetStatus(codes.Error, err.Error())
		return nil, err
	}

	return &mailerpb.CreateMailerResponse{
		Id: mailerID,
	}, nil
}
