"use client";

import React, { useMemo, useState } from 'react';
import PropTypes from 'prop-types';
import { useTranslations } from 'next-intl';
import { Tag, DollarSign, Calendar, Gauge } from "lucide-react";
import styles from '../../post/CreatePost.module.css';
import RichTextEditor from "@/components/Shared/TextEditor/index.jsx";
import { FormActions } from "@/components/Shared/FormActions.jsx";

export function BasicInfoStep({
    initialData,
    onComplete,
    onBack,
    isLoading,
    errors,
    categories = [],
    isCategoriesLoading,
}) {
    const t = useTranslations('CreateVehicleModal');

    // Local state for form fields
    const [vehicleTitle, setVehicleTitle] = useState(initialData?.title || "");
    const [vehicleDescription, setVehicleDescription] = useState(initialData?.description || "");
    const [price, setPrice] = useState(initialData?.price || "");
    const [make, setMake] = useState(initialData?.make || "");
    const [model, setModel] = useState(initialData?.model || "");
    const [year, setYear] = useState(initialData?.year || "");
    const [mileage, setMileage] = useState(initialData?.mileage || "");
    const [condition, setCondition] = useState(initialData?.condition || "used");
    const [fuelType, setFuelType] = useState(initialData?.fuelType || "gasoline");
    const [transmission, setTransmission] = useState(initialData?.transmission || "automatic");
    const [category, setCategory] = useState(initialData?.category || "");
    const [tags, setTags] = useState(initialData?.tags || "");

    // Memoized values
    const isPrimaryDisabled = useMemo(() => isLoading, [isLoading]);
    const primaryLabel = useMemo(() => {
        if (isLoading) return t("basicInfoSavingButton");
        return t("basicInfoContinueButton");
    }, [isLoading, t]);

    // Current year for validation
    const currentYear = new Date().getFullYear();

    // Handle form submission
    const handleSubmit = (e) => {
        e.preventDefault();
        onComplete({
            title: vehicleTitle,
            description: vehicleDescription,
            price,
            make,
            model,
            year,
            mileage,
            condition,
            fuelType,
            transmission,
            category,
            tags
        });
    };

    // Helper to translate error keys
    const translateError = (errorKey) => {
        if (!errorKey) return null;
        const values = {};
        if (errorKey === 'errorTitleMinLength') values.minLength = 3;
        if (errorKey === 'errorContentMinLength') values.minLength = 20;
        return t(errorKey, values);
    };

    return (
        <div className={styles.formContainer}>
            <h2 className={styles.formTitle}>{t('basicInfoTitle')}</h2>
            <p className={styles.formDescription}>{t('basicInfoDescription')}</p>

            {/* Display Submit Error Message */}
            {errors?.submit && (
                <div className={`${styles.fieldError} ${styles.submitError}`} role="alert">
                    {translateError(errors.submit)}
                </div>
            )}

            <form className={styles.form} onSubmit={handleSubmit} noValidate>
                {/* Vehicle Title field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-title">
                        {t('basicInfoTitleLabel')} <span className={styles.requiredMark}>*</span>
                    </label>
                    <input
                        id="vehicle-title"
                        className={`${styles.formInput} ${errors?.title ? styles.inputError : ""}`}
                        value={vehicleTitle}
                        onChange={(e) => setVehicleTitle(e.target.value)}
                        placeholder={t('basicInfoTitlePlaceholder')}
                        maxLength={100}
                        aria-invalid={!!errors?.title}
                        aria-describedby={errors?.title ? "title-error" : undefined}
                    />
                    {errors?.title && (
                        <div className={styles.fieldError} id="title-error" role="alert">
                            {translateError(errors.title)}
                        </div>
                    )}
                    <div className={styles.charCount} aria-live="polite">
                        {t('basicInfoCharCount', { count: vehicleTitle.length })}
                    </div>
                </div>

                {/* Price field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-price">
                        {t('basicInfoPriceLabel')} <span className={styles.requiredMark}>*</span>
                    </label>
                    <div className={styles.tagsInputWrapper}>
                        <DollarSign size={16} className={styles.tagIcon} aria-hidden="true" />
                        <input
                            id="vehicle-price"
                            type="number"
                            step="0.01"
                            min="0"
                            className={`${styles.formInput} ${errors?.price ? styles.inputError : ""}`}
                            value={price}
                            onChange={(e) => setPrice(e.target.value)}
                            placeholder={t('basicInfoPricePlaceholder')}
                            aria-invalid={!!errors?.price}
                            aria-describedby={errors?.price ? "price-error" : undefined}
                        />
                    </div>
                    {errors?.price && (
                        <div className={styles.fieldError} id="price-error" role="alert">
                            {translateError(errors.price)}
                        </div>
                    )}
                </div>

                {/* Make and Model fields */}
                <div className={styles.formRowGroup}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-make">
                            {t('basicInfoMakeLabel')} <span className={styles.requiredMark}>*</span>
                        </label>
                        <input
                            id="vehicle-make"
                            className={`${styles.formInput} ${errors?.make ? styles.inputError : ""}`}
                            value={make}
                            onChange={(e) => setMake(e.target.value)}
                            placeholder={t('basicInfoMakePlaceholder')}
                            maxLength={50}
                            aria-invalid={!!errors?.make}
                            aria-describedby={errors?.make ? "make-error" : undefined}
                        />
                        {errors?.make && (
                            <div className={styles.fieldError} id="make-error" role="alert">
                                {translateError(errors.make)}
                            </div>
                        )}
                    </div>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-model">
                            {t('basicInfoModelLabel')} <span className={styles.requiredMark}>*</span>
                        </label>
                        <input
                            id="vehicle-model"
                            className={`${styles.formInput} ${errors?.model ? styles.inputError : ""}`}
                            value={model}
                            onChange={(e) => setModel(e.target.value)}
                            placeholder={t('basicInfoModelPlaceholder')}
                            maxLength={50}
                            aria-invalid={!!errors?.model}
                            aria-describedby={errors?.model ? "model-error" : undefined}
                        />
                        {errors?.model && (
                            <div className={styles.fieldError} id="model-error" role="alert">
                                {translateError(errors.model)}
                            </div>
                        )}
                    </div>
                </div>

                {/* Year and Mileage fields */}
                <div className={styles.formRowGroup}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-year">
                            {t('basicInfoYearLabel')} <span className={styles.requiredMark}>*</span>
                        </label>
                        <div className={styles.tagsInputWrapper}>
                            <Calendar size={16} className={styles.tagIcon} aria-hidden="true" />
                            <input
                                id="vehicle-year"
                                type="number"
                                min="1900"
                                max={currentYear + 1}
                                className={`${styles.formInput} ${errors?.year ? styles.inputError : ""}`}
                                value={year}
                                onChange={(e) => setYear(e.target.value)}
                                placeholder={t('basicInfoYearPlaceholder')}
                                aria-invalid={!!errors?.year}
                                aria-describedby={errors?.year ? "year-error" : undefined}
                            />
                        </div>
                        {errors?.year && (
                            <div className={styles.fieldError} id="year-error" role="alert">
                                {translateError(errors.year)}
                            </div>
                        )}
                    </div>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-mileage">
                            {t('basicInfoMileageLabel')}
                        </label>
                        <div className={styles.tagsInputWrapper}>
                            <Gauge size={16} className={styles.tagIcon} aria-hidden="true" />
                            <input
                                id="vehicle-mileage"
                                type="number"
                                min="0"
                                className={styles.formInput}
                                value={mileage}
                                onChange={(e) => setMileage(e.target.value)}
                                placeholder={t('basicInfoMileagePlaceholder')}
                            />
                        </div>
                        <div className={styles.inputHint}>{t('basicInfoMileageHint')}</div>
                    </div>
                </div>

                {/* Condition field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-condition">
                        {t('basicInfoConditionLabel')}
                    </label>
                    <select
                        id="vehicle-condition"
                        className={styles.formInput}
                        value={condition}
                        onChange={(e) => setCondition(e.target.value)}
                    >
                        <option value="new">{t('conditionNew')}</option>
                        <option value="like-new">{t('conditionLikeNew')}</option>
                        <option value="excellent">{t('conditionExcellent')}</option>
                        <option value="good">{t('conditionGood')}</option>
                        <option value="fair">{t('conditionFair')}</option>
                        <option value="poor">{t('conditionPoor')}</option>
                        <option value="salvage">{t('conditionSalvage')}</option>
                    </select>
                </div>

                {/* Fuel Type and Transmission fields */}
                <div className={styles.formRowGroup}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-fuel-type">
                            {t('basicInfoFuelTypeLabel')}
                        </label>
                        <select
                            id="vehicle-fuel-type"
                            className={styles.formInput}
                            value={fuelType}
                            onChange={(e) => setFuelType(e.target.value)}
                        >
                            <option value="gasoline">{t('fuelTypeGasoline')}</option>
                            <option value="diesel">{t('fuelTypeDiesel')}</option>
                            <option value="hybrid">{t('fuelTypeHybrid')}</option>
                            <option value="electric">{t('fuelTypeElectric')}</option>
                            <option value="other">{t('fuelTypeOther')}</option>
                        </select>
                    </div>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="vehicle-transmission">
                            {t('basicInfoTransmissionLabel')}
                        </label>
                        <select
                            id="vehicle-transmission"
                            className={styles.formInput}
                            value={transmission}
                            onChange={(e) => setTransmission(e.target.value)}
                        >
                            <option value="automatic">{t('transmissionAutomatic')}</option>
                            <option value="manual">{t('transmissionManual')}</option>
                            <option value="cvt">{t('transmissionCVT')}</option>
                        </select>
                    </div>
                </div>

                {/* Category field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-category">
                        {t('basicInfoCategoryLabel')}
                    </label>
                    <select
                        id="vehicle-category"
                        className={styles.formInput}
                        value={category}
                        onChange={(e) => setCategory(e.target.value)}
                        disabled={isCategoriesLoading}
                    >
                        <option value="">{t('basicInfoCategoryPlaceholder')}</option>
                        {categories.map((cat) => (
                            <option key={cat.id} value={cat.id}>
                                {cat.name}
                            </option>
                        ))}
                    </select>
                </div>

                {/* Description field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-description">
                        {t('basicInfoContentLabel')} <span className={styles.requiredMark}>*</span>
                    </label>
                    <div className={`${styles.editorContainer} ${errors?.description ? styles.editorError : ""}`}>
                        <RichTextEditor
                            value={vehicleDescription}
                            onChange={setVehicleDescription}
                            placeholder={t('basicInfoContentPlaceholder')}
                            mediaId={initialData?.mediaId}
                            id="vehicle-description"
                        />
                    </div>
                    {errors?.description && (
                        <div className={styles.fieldError} id="description-error" role="alert">
                            {translateError(errors.description)}
                        </div>
                    )}
                </div>

                {/* Tags field */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="vehicle-tags">
                        {t('basicInfoTagsLabel')}
                    </label>
                    <div className={styles.tagsInputWrapper}>
                        <Tag size={16} className={styles.tagIcon} aria-hidden="true" />
                        <input
                            id="vehicle-tags"
                            className={styles.formInput}
                            value={tags}
                            onChange={(e) => setTags(e.target.value)}
                            placeholder={t('basicInfoTagsPlaceholder')}
                            maxLength={200}
                        />
                    </div>
                    <div className={styles.inputHint}>{t('basicInfoTagsHint')}</div>
                </div>

                <FormActions
                    primaryLabel={primaryLabel}
                    primaryIcon="arrow-right"
                    onCancel={onBack}
                    isPrimaryDisabled={isPrimaryDisabled}
                />
            </form>
        </div>
    );
}

BasicInfoStep.propTypes = {
    initialData: PropTypes.shape({
        title: PropTypes.string,
        description: PropTypes.string,
        price: PropTypes.oneOfType([PropTypes.string, PropTypes.number]),
        make: PropTypes.string,
        model: PropTypes.string,
        year: PropTypes.oneOfType([PropTypes.string, PropTypes.number]),
        mileage: PropTypes.oneOfType([PropTypes.string, PropTypes.number]),
        condition: PropTypes.string,
        fuelType: PropTypes.string,
        transmission: PropTypes.string,
        category: PropTypes.string,
        tags: PropTypes.string,
        mediaId: PropTypes.string
    }),
    onComplete: PropTypes.func.isRequired,
    onBack: PropTypes.func.isRequired,
    isLoading: PropTypes.bool,
    errors: PropTypes.object,
    categories: PropTypes.array,
    isCategoriesLoading: PropTypes.bool,
};

BasicInfoStep.defaultProps = {
    initialData: { 
        title: '', 
        description: '', 
        price: '', 
        make: '', 
        model: '', 
        year: '', 
        mileage: '', 
        condition: 'used', 
        fuelType: 'gasoline', 
        transmission: 'automatic', 
        category: '', 
        tags: '', 
        mediaId: null 
    },
    isLoading: false,
    errors: {},
    categories: [],
    isCategoriesLoading: false,
};
