import React from 'react';
import JobsPageClient from './JobsPage.client';
import {getAllJobs} from '../../../api/jobsApi';

export const dynamic = 'force-dynamic';

// Helper function to safely resolve props in Next.js 15+
async function resolveProps(props) {
    // In Next.js 15+, we need to await both params and searchParams
    const params = props.params ? await props.params : {};
    const searchParams = props.searchParams ? await props.searchParams : {};

    return {params, searchParams};
}

export async function generateMetadata(props) {
    // Safely resolve props first
    const {params} = await resolveProps(props);

    return {
        title: 'Job Listings | Vaveme',
        description: 'Find the latest job openings.',
    };
}

export default async function JobsIndexPage(props) {
    // 1. First await BOTH params and searchParams correctly
    const {params, searchParams} = await resolveProps(props);

    // 2. Now it's safe to destructure params
    const {locale} = await params;

    // 3. Extract searchParams values safely
    const displayMode = searchParams?.displayMode || 'grid';
    const page = parseInt(searchParams?.page || '1', 10);
    const limit = 20;
    const sortBy = searchParams?.sortBy || '';

    // 4. Construct the filters object using extracted values
    const initialClientFilters = {
        displayMode,
        page,
        limit,
        sortBy,
    };

    // 5. Fetch INITIAL jobs for the server render
    let featuredData = {jobs: [], totalCount: 0, totalPages: 0};
    let fetchError = null;

    try {
        const serverApiFilters = {
            page: 1,
            limit,
            // Additional filters if needed
        };

        featuredData = await getAllJobs(serverApiFilters);
    } catch (err) {
        console.error('Error fetching initial jobs on server:', err);
        fetchError = "Could not load jobs at this time. Please try again later.";
    }

    // 6. Process the fetched data
    const serverJobs = featuredData?.jobs || [];
    const totalPages = parseInt(featuredData?.totalPages || '0', 10);
    const totalCount = parseInt(featuredData?.totalCount || '0', 10);
    const currentPage = parseInt(featuredData?.currentPage || '1', 10);

    // 7. Construct JSON-LD for SEO
    const jobsJsonLd = {
        '@context': 'https://schema.org',
        '@type': 'ItemList',
        name: 'Job Listings',
        itemListElement: serverJobs.map((job, index) => {
            return {
                '@type': 'ListItem',
                position: index + 1,
                item: {
                    '@type': 'JobPosting',
                    'title': job.title,
                    'description': job.description?.substring(0, 160),
                    'identifier': {
                        '@type': 'PropertyValue',
                        'name': 'jobId',
                        'value': job.id
                    },
                    'datePosted': job.postedDate || job.createdAt,
                    'employmentType': job.jobType ? job.jobType.toUpperCase() : 'FULL_TIME',
                    'hiringOrganization': {
                        '@type': 'Organization',
                        'name': job.company || 'Unknown Employer',
                    },
                    'jobLocation': {
                        '@type': 'Place',
                        'address': {
                            '@type': 'PostalAddress',
                            'addressLocality': job.locationCity || job.location?.split(',')[0]?.trim() || '',
                            'addressRegion': job.locationState || job.location?.split(',')[1]?.trim() || '',
                            'addressCountry': job.locationCountry || 'DE',
                        }
                    },
                },
                url: `https://www.sfx-markt.de/jobs/${job.category || 'all'}/${job.id}`,
            };
        }),
    };

    // 8. Return the UI components
    return (
        <>
            <script
                type="application/ld+json"
                dangerouslySetInnerHTML={{
                    __html: JSON.stringify(jobsJsonLd),
                }}
                key="job-listings-jsonld"
            />

            <JobsPageClient
                serverJobs={serverJobs}
                serverFilters={initialClientFilters}
                fetchError={fetchError}
                totalPages={totalPages}
                currentPage={currentPage}
                totalCount={totalCount}
            />
        </>
    );
}
