"use client"; // Essential for hooks

import React, {useCallback, useMemo, useState} from 'react';
import PropTypes from 'prop-types';
import {useTranslations} from 'next-intl';
import {
    BadgeCheck, BarChart, Bookmark, Briefcase, BriefcaseBusiness, Building, Calendar,
    CalendarRange, Check, Clock, DollarSign, Eye
    , Globe, GraduationCap, Heart, Laptop, MapPin
    , MessagesSquare, Share2,
    Terminal, ThumbsUp, Users
} from 'lucide-react';
import dayjs from 'dayjs';
import relativeTime from 'dayjs/plugin/relativeTime';

import styles from './JobCard.module.css';
import CommentsSetup from "../../../../../features/Comments/CommentsSetup";
import {useAuth} from "../../../../../context/AuthContext"; // Adjust path & name

dayjs.extend(relativeTime);

// Define default structure matching expected props
const defaultJob = {
    id: null, title: '', company: '', companyVerified: false, location: '', remoteOptions: '',
    jobType: '', contractType: '', salary: '', experience: '', education: '', postedDate: null,
    applicationDeadline: null, category: '', subcategory: '', description: '', responsibilities: [],
    requirements: [], benefits: [], skills: [], companyInfo: {}, views: 0, savedCount: 0,
    likes: 0, dislikes: 0, comments: 0, messages: 0, applicants: 0, heatScore: 0,
    isFeatured: false, isUrgent: false
};

/**
 * Simple console notification replacement for removed toast functionality
 */
const useToastNotification = () => {
    const t = useTranslations('JobCard');
    return useCallback((type, messageKey, values = {}) => {
        const message = t(messageKey, values);
        // Since react-toastify was removed, use console logging
        console.warn(`[${type.toUpperCase()}] ${message}`);
        
        // Show alert for errors only
        if (type === 'error') {
            alert(`Error: ${message}`);
        }
    }, [t]);
};

/**
 * JobCard - Displays detailed job information.
 * Accepts job data via props. Uses next-intl for translations.
 */
// Renamed component to JobCardDisplay for clarity, or keep as JobCard if preferred
const JobCardDisplay = ({job: jobProp}) => { // Accept job prop
    const t = useTranslations('JobCard');
    const showToast = useToastNotification();
    const {user} = useAuth();
    const userId = user?.userId;
    // const { handleLike, handleDislike } = useActivityApi();
    // const { handleToggleWishlist, isInWishlist } = useWishlist(jobProp?.id);

    // Merge passed job data with defaults
    const jobData = useMemo(() => ({...defaultJob, ...(jobProp || {})}), [jobProp]);

    // State variables
    const [expanded, setExpanded] = useState(false);
    const [saved, setSaved] = useState(false); // TODO: Connect to wishlist state
    const [favorite, setFavorite] = useState(false);
    const [liked, setLiked] = useState(false); // TODO: Connect to user reaction state
    const [disliked, setDisliked] = useState(false); // TODO: Connect to user reaction state
    const [showContact, setShowContact] = useState(false);
    const [selectedSection, setSelectedSection] = useState("overview");

    // --- Derived Data & Formatting ---
    const timeAgo = useMemo(() => (jobData.postedDate ? dayjs(jobData.postedDate).fromNow() : ''), [jobData.postedDate]);
    const deadlineText = useMemo(() => (jobData.applicationDeadline ? dayjs(jobData.applicationDeadline).fromNow(true) : ''), [jobData.applicationDeadline]);

    const heatInfo = useMemo(() => {
        const score = jobData.heatScore;
        if (score >= 80) return {className: styles.heatHigh, labelKey: 'heatHot'};
        if (score >= 60) return {className: styles.heatMedium, labelKey: 'heatPopular'};
        return {className: styles.heatLow, labelKey: 'heatJustListed'};
    }, [jobData.heatScore]);

    // --- Event Handlers ---
    const handleSaveToggle = useCallback(() => {
        setSaved(prev => !prev);
        showToast('info', saved ? 'toastJobUnsaved' : 'toastJobSaved');
    }, [saved, showToast]);

    const handleContactSubmit = useCallback((e) => {
        e.preventDefault();
        showToast('success', 'toastContactSent');
        setShowContact(false);
    }, [jobData.id, showToast]);

    // --- Render Logic ---
    if (!jobData.id) {
        return <div className={styles.cardContainer} role="alert">{t('errorLoadingJob')}</div>;
    }

    // Helper function to get translated job/contract type safely
    const getTranslatedType = (typeKeyPrefix, typeValue) => {
        return typeValue ? t(`${typeKeyPrefix}_${typeValue.toLowerCase()}`, {}, {defaultValue: typeValue}) : '';
    };

    return (
        <div className={styles.cardContainer}>
            <div className={`${styles.heatIndicator} ${heatInfo.className}`}></div>

            {/* Header */}
            <div className={styles.cardHeader}>
                <div className={styles.headerFlex}>
                    <div className={styles.logoContainer}>
                        {jobData.companyInfo?.logo ? (
                            <img src={jobData.companyInfo.logo}
                                 alt={t('companyLogoAlt', {companyName: jobData.company})} className={styles.logo}/>
                        ) : (
                            <BriefcaseBusiness size={32} className={styles.logoPlaceholder}/>
                        )}
                    </div>
                    <div className={styles.headerContent}>
                        <div className={styles.titleRow}>
                            <h2 className={styles.jobTitle}>{jobData.title}</h2>
                            <div className={styles.badgeContainer}>
                                {jobData.isFeatured && <div className={styles.badgeFeatured}>{t('badgeFeatured')}</div>}
                                {jobData.isUrgent && <div className={styles.badgeUrgent}>{t('badgeUrgent')}</div>}
                            </div>
                        </div>
                        <div className={styles.companyRow}>
                            <span className={styles.companyName}>{jobData.company}</span>
                            {jobData.companyVerified && (
                                <span className={styles.verifiedBadge}>
                                    <BadgeCheck size={12} className={styles.verifiedIcon} aria-hidden="true"/>
                                    {t('badgeVerified')}
                                </span>
                            )}
                        </div>
                        <div className={styles.quickStatsContainer}>
                            {jobData.location && <QuickStat icon={<MapPin size={12}/>} text={jobData.location}/>}
                            {jobData.jobType && <QuickStat icon={<Briefcase size={12}/>}
                                                           text={getTranslatedType('jobType', jobData.jobType)}/>}
                            {jobData.salary && <QuickStat icon={<DollarSign size={12}/>} text={jobData.salary}/>}
                            {timeAgo && <QuickStat icon={<Clock size={12}/>} text={timeAgo}/>}
                        </div>
                    </div>
                </div>
            </div>

            {/* Main Content */}
            <div className={styles.contentContainer}>
                {/* Left Column */}
                <div className={styles.mainContent}>
                    <div className={styles.tabNav}>
                        <button type="button"
                                className={`${styles.tabButton} ${selectedSection === "overview" ? styles.tabButtonActive : ''}`}
                                onClick={() => setSelectedSection("overview")}>{t('tabOverview')}</button>
                        <button type="button"
                                className={`${styles.tabButton} ${selectedSection === "requirements" ? styles.tabButtonActive : ''}`}
                                onClick={() => setSelectedSection("requirements")}>{t('tabRequirements')}</button>
                        <button type="button"
                                className={`${styles.tabButton} ${selectedSection === "company" ? styles.tabButtonActive : ''}`}
                                onClick={() => setSelectedSection("company")}>{t('tabCompany')}</button>
                    </div>

                    <div className={styles.tabContent}>
                        {/* Overview Tab */}
                        {selectedSection === "overview" && (
                            <div>
                                <div className={styles.contentSection}>
                                    <h3 className={styles.sectionTitle}>{t('sectionJobDescription')}</h3>
                                    <p className={styles.sectionText}>{jobData.description}</p>
                                </div>
                                {jobData.responsibilities?.length > 0 && (
                                    <div className={styles.contentSection}>
                                        <h3 className={styles.sectionTitle}>{t('sectionKeyResponsibilities')}</h3>
                                        <ul className={styles.listDisc}>
                                            {jobData.responsibilities.map((item, index) => <li key={index}>{item}</li>)}
                                        </ul>
                                    </div>
                                )}
                                {jobData.benefits?.length > 0 && (
                                    <div className={styles.contentSection}>
                                        <h3 className={styles.sectionTitle}>{t('sectionBenefitsPerks')}</h3>
                                        <ul className={styles.benefitsList}>
                                            {jobData.benefits.map((benefit, index) => <li key={index}
                                                                                          className={styles.benefitItem}>
                                                <Check size={12} className={styles.checkIcon}/><span>{benefit}</span>
                                            </li>)}
                                        </ul>
                                    </div>
                                )}
                            </div>
                        )}
                        {/* Requirements Tab */}
                        {selectedSection === "requirements" && (
                            <div>
                                {jobData.requirements?.length > 0 && (
                                    <div className={styles.contentSection}>
                                        <h3 className={styles.sectionTitle}>{t('sectionRequirements')}</h3>
                                        <ul className={styles.listDisc}>
                                            {jobData.requirements.map((item, index) => <li key={index}>{item}</li>)}
                                        </ul>
                                    </div>
                                )}
                                {jobData.skills?.length > 0 && (
                                    <div className={styles.contentSection}>
                                        <h3 className={styles.sectionTitle}>{t('sectionRequiredSkills')}</h3>
                                        <div className={styles.skillsContainer}>
                                            {jobData.skills.map((skill, index) => <span key={index}
                                                                                        className={styles.skillBadge}>{skill}</span>)}
                                        </div>
                                    </div>
                                )}
                                <div className={styles.infoGridContainer}>
                                    {jobData.experience && (
                                        <div className={styles.infoCard}>
                                            <div className={styles.infoCardTitle}>
                                                <Briefcase size={12} className={styles.infoCardIcon}/><span
                                                className={styles.infoCardLabel}>{t('infoCardExperienceLabel')}</span>
                                            </div>
                                            <div className={styles.infoCardValue}>{jobData.experience}</div>
                                        </div>
                                    )}
                                    {jobData.education && (
                                        <div className={styles.infoCard}>
                                            <div className={styles.infoCardTitle}>
                                                <GraduationCap size={12} className={styles.infoCardIcon}/><span
                                                className={styles.infoCardLabel}>{t('infoCardEducationLabel')}</span>
                                            </div>
                                            <div className={styles.infoCardValue}>{jobData.education}</div>
                                        </div>
                                    )}
                                </div>
                            </div>
                        )}
                        {/* Company Tab */}
                        {selectedSection === "company" && jobData.companyInfo && (
                            <div>
                                <div className={styles.contentSection}>
                                    <h3 className={styles.sectionTitle}>{t('sectionAboutCompany', {companyName: jobData.company})}</h3>
                                    <p className={styles.sectionText}>{jobData.companyInfo.description}</p>
                                </div>
                                <div className={styles.infoGridContainer}>
                                    {jobData.companyInfo.industry && <div className={styles.infoCard}>
                                        <div className={styles.infoCardTitle}><Building size={12}
                                                                                        className={styles.infoCardIcon}/><span
                                            className={styles.infoCardLabel}>{t('infoCardIndustryLabel')}</span></div>
                                        <div className={styles.infoCardValue}>{jobData.companyInfo.industry}</div>
                                    </div>}
                                    {jobData.companyInfo.size && <div className={styles.infoCard}>
                                        <div className={styles.infoCardTitle}><Users size={12}
                                                                                     className={styles.infoCardIcon}/><span
                                            className={styles.infoCardLabel}>{t('infoCardCompanySizeLabel')}</span>
                                        </div>
                                        <div className={styles.infoCardValue}>{jobData.companyInfo.size}</div>
                                    </div>}
                                    {jobData.companyInfo.founded && <div className={styles.infoCard}>
                                        <div className={styles.infoCardTitle}><Calendar size={12}
                                                                                        className={styles.infoCardIcon}/><span
                                            className={styles.infoCardLabel}>{t('infoCardFoundedLabel')}</span></div>
                                        <div className={styles.infoCardValue}>{jobData.companyInfo.founded}</div>
                                    </div>}
                                    {jobData.companyInfo.website && <div className={styles.infoCard}>
                                        <div className={styles.infoCardTitle}><Globe size={12}
                                                                                     className={styles.infoCardIcon}/><span
                                            className={styles.infoCardLabel}>{t('infoCardWebsiteLabel')}</span></div>
                                        <div className={styles.infoCardValue}><a
                                            href={`https://${jobData.companyInfo.website}`} target="_blank"
                                            rel="noopener noreferrer">{jobData.companyInfo.website}</a></div>
                                    </div>}
                                </div>
                            </div>
                        )}
                    </div>
                </div>

                {/* Right Sidebar */}
                <div className={styles.sidebar}>
                    <div className={styles.contentSection}>
                        <h3 className={styles.sectionTitle}>{t('sidebarTitle')}</h3>
                        <div className={styles.jobGlanceList}>
                            {jobData.applicationDeadline && <QuickStat icon={<CalendarRange size={14}/>}
                                                                       text={t('glanceDeadline', {deadlineRelative: deadlineText})}/>}
                            {jobData.remoteOptions && <QuickStat icon={<Laptop size={14}/>}
                                                                 text={t('glanceRemoteWork', {options: jobData.remoteOptions})}/>}
                            {jobData.contractType && <QuickStat icon={<BriefcaseBusiness size={14}/>}
                                                                text={t('glanceContractType', {type: getTranslatedType('contractType', jobData.contractType)})}/>}
                            {jobData.category && <QuickStat icon={<Terminal size={14}/>}
                                                            text={t('glanceCategory', {category: jobData.category})}/>}
                            {jobData.applicants >= 0 && <QuickStat icon={<BarChart size={14}/>}
                                                                   text={t('glanceApplicantsText', {count: jobData.applicants})}/>}
                        </div>
                    </div>
                    <div className={styles.interactionBar}>
                        <div className={styles.interactionButtons}>
                            <button type="button" className={styles.interactionButton}
                                    onClick={() => setFavorite(!favorite)}
                                    aria-label={t('saveButtonAriaLabel', {count: jobData.savedCount})}><Heart size={20}
                                                                                                              className={favorite ? styles.heartActive : styles.heartInactive}/>
                                <span className={styles.interactionCount}>{jobData.savedCount}</span></button>
                            <button type="button" className={styles.interactionButton} onClick={() => {
                                setLiked(!liked);
                                if (disliked) setDisliked(false);
                            }} aria-label={t('likeButtonAriaLabel', {count: jobData.likes})}><ThumbsUp size={20}
                                                                                                       className={liked ? styles.thumbsUpActive : styles.thumbsUpInactive}/>
                                <span
                                    className={styles.interactionCount}>{liked ? jobData.likes + 1 : jobData.likes}</span>
                            </button>
                            <button type="button" className={styles.interactionButton}
                                    aria-label={t('commentButtonAriaLabel', {count: jobData.comments})}><MessagesSquare
                                size={20} className={styles.messagesIcon}/> <span
                                className={styles.interactionCount}>{jobData.comments}</span></button>
                        </div>
                        <div>
                            <span className={styles.viewsCounter}
                                  aria-label={t('viewsAriaLabel', {count: jobData.views})}> <Eye size={12}
                                                                                                 className={styles.viewsIcon}/>{jobData.views} </span>
                        </div>
                    </div>
                    <div className={styles.actionButtonsContainer}>
                        <button type="button" className={styles.applyButton}>{t('applyButton')}</button>
                        <div className={styles.secondaryActionsContainer}>
                            <button type="button" className={styles.secondaryButton} onClick={handleSaveToggle}
                                    aria-pressed={saved}><Bookmark size={16}
                                                                   className={saved ? styles.bookmarkActive : styles.bookmarkInactive}/>
                                <span>{t(saved ? 'savedButton' : 'saveButton')}</span></button>
                            <button type="button" className={styles.secondaryButton}><Share2 size={16}
                                                                                             className={styles.shareIcon}/>
                                <span>{t('shareButton')}</span></button>
                        </div>
                    </div>
                    <div className={styles.similarJobsContainer}>
                        <h3 className={styles.sectionTitle}>{t('similarJobsTitle')}</h3>
                        <div className={styles.similarJobsList}>
                            <p>({t('similarJobsPlaceholder')})</p> {/* Placeholder */}
                        </div>
                        <button type="button"
                                className={styles.viewMoreButton}>{t('viewMoreSimilarJobsButton')}</button>
                    </div>
                </div>
            </div>
            {/* Contact Form Modal */}
            {showContact && (
                <form className={styles.contactForm} onSubmit={handleContactSubmit}>
                    <div className={styles.formTitle}>{t('contactFormTitle', {jobTitle: jobData.title})}</div>
                    <div className={styles.formFields}>
                        <input type="text" placeholder={t('contactFormNamePlaceholder')} className={styles.formInput}
                               required aria-label={t('contactFormNamePlaceholder')}/>
                        <input type="email" placeholder={t('contactFormEmailPlaceholder')} className={styles.formInput}
                               required aria-label={t('contactFormEmailPlaceholder')}/>
                        <input type="tel" placeholder={t('contactFormPhonePlaceholder')} className={styles.formInput}
                               aria-label={t('contactFormPhonePlaceholder')}/>
                        <textarea
                            placeholder={t('contactFormMessagePlaceholder')}
                            rows={3} // Increased rows
                            className={styles.formInput}
                            defaultValue={t('contactFormDefaultMessage', {
                                jobTitle: jobData.title,
                                companyName: jobData.company
                            })}
                            aria-label={t('contactFormMessagePlaceholder')}
                            required // Make message required
                        ></textarea>
                        <div className={styles.formActions}>
                            <button type="button" onClick={() => setShowContact(false)}
                                    className={styles.cancelButton}>{t('cancelButton')}</button>
                            {/* Add Cancel Button */}
                            <button type="submit" className={styles.sendButton}>{t('contactFormSendButton')}</button>
                        </div>
                    </div>
                </form>
            )}
            {/* Comments Section - Consider loading/rendering this on demand */}
            {interactionState?.showComments && (
                <div className={styles.commentsWrapper}>
                    <CommentsSetup
                        userId={userId}
                        itemId={jobData.id}
                        itemType="job" // Specify item type
                        toggleCommentsList={() => {
                        }}
                        // Pass other necessary props like categoryId if needed by CommentsSetup
                        categoryId={"some"}/>
                </div>
            )}
        </div>
    );
};

// Add PropTypes
JobCardDisplay.propTypes = { // Renamed from JobCard to JobCardDisplay
    job: PropTypes.object.isRequired,
};

// ⬅️ FIX: Export the component as default
export default JobCardDisplay;

    showComments: false, // Default value
};
