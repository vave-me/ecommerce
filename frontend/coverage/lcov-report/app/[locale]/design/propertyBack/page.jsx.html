                                 images = ["/images/house.webp", "/api/placeholder/500/300", "/api/placeholder/500/300"],
                                 title = "Modern 2-Bedroom Apartment with Balcony",
                                 price = "$1,250 /month",
                                 priceSuffix = "/month",
                                 priceType = "rent", // "rent" or "sale"
                                 deposit = "$2,500",
                                 furnished = true,
                                 yearBuilt = "2018",
                                 size = "85m²",
                                 address = "Friedrichshain, Berlin 10243",
                                 fullAddress = "Warschauer Str. 58, 10243 Berlin, Germany",
                                 propertyType = "Apartment",
                                 bedrooms = 2,
                                 bathrooms = 1,
                                 availableFrom = "2025-04-15",
                                 pets = true,
                                 parking = 1,
                                 amenities = ["Balcony", "Washer/Dryer", "Hardwood Floors", "Elevator", "Heating", "Dishwasher"],
                                 description = "Beautiful, bright apartment in a prime location. Recently renovated with modern appliances and fixtures. Large windows provide plenty of natural light. Close to public transportation, shops, and restaurants. The living room opens to a spacious balcony perfect for morning coffee or evening relaxation.",

                                     name: "Emma Schmidt",
                                     company: "Berlin Prime Properties",
                                     verified: true,
                                     rating: 4.9,
                                     listings: 32,
                                     experience: "7 years"
                                 },
                                 views = 387,
                                 savedCount = 42,
                                 inquiries = 15,
                                 isFavorite = false,
                                 isPromoted = true,
                                 isPremium = true,
                                 tags = ["U-Bahn Nearby", "Quiet Area", "Pet Friendly", "Recent Renovation"],
                                 likes = 53,
                                 dislikes = 3,
                                 comments = 9,
                                 messages = 21,
                                 heatScore = 88, // 0-100 heat score for popularity
                                 energyRating = "B", // Energy efficiency rating
                                 virtualTour = true,
                                 floorplanAvailable = true,
                                 mortgage = {
                                     available: true,
                                     monthlyPayment: "$1,450",
                                     interestRate: "3.25%",
                                     term: "30 years"
                                 }
                             }) => {
    const [expanded, setExpanded] = useState(false);
    const [saved, setSaved] = useState(false);
    const [favorite, setFavorite] = useState(isFavorite);
    const [liked, setLiked] = useState(false);
    const [disliked, setDisliked] = useState(false);
    const [currentImageIndex, setCurrentImageIndex] = useState(0);
    const [showContact, setShowContact] = useState(false);

    // Next image function
    const nextImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === images.length - 1 ? 0 : prevIndex + 1
        );
    };

    // Previous image function
    const prevImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === 0 ? images.length - 1 : prevIndex - 1
        );
    };

    // Determine heat level class
    const getHeatLevelClass = (score) => {
        if (score >= 80) return styles.heatIndicatorHot;
        if (score >= 60) return styles.heatIndicatorPopular;
        if (score >= 40) return styles.heatIndicatorGood;
        if (score >= 20) return styles.heatIndicatorNew;
        return styles.heatIndicatorListed;
    };

    const getHeatLabel = (score) => {
        if (score >= 80) return "HOT!";
        if (score >= 60) return "Popular";
        if (score >= 40) return "Good";
        if (score >= 20) return "New";
        return "Listed";
    };

    // Format date to more readable format
    const formatDate = (dateString) => {
        const date = new Date(dateString);
        return date.toLocaleDateString('en-US', {year: 'numeric', month: 'short', day: 'numeric'});
    };

    // Get energy efficiency class
    const getEnergyRatingClass = (rating) => {
        switch (rating) {
            case 'A':
                return styles.energyRatingA;
            case 'B':
                return styles.energyRatingB;
            case 'C':
                return styles.energyRatingC;
            case 'D':
                return styles.energyRatingD;
            case 'E':
                return styles.energyRatingE;
            case 'F':
                return styles.energyRatingF;
            case 'G':
                return styles.energyRatingG;
            default:
                return "";
        }
    };

    return (
        <div className={styles.propertyCard}>
            {/* Heat indicator at top */}
            <div className={`${styles.heatIndicator} ${getHeatLevelClass(heatScore)}`}></div>

            {/* Image and interaction section - 50% width on desktop */}
            <div className={styles.imageSection}>
                {/* Promoted/Premium badge */}
                {(isPromoted || isPremium) && (
                    <div className={isPremium ? styles.premiumBadge : styles.promotedBadge}>
                        {isPremium ? 'PREMIUM LISTING' : 'PROMOTED LISTING'}
                    </div>
                )}

                <div className={styles.imageContainer}>
                    <img
                        src={images[currentImageIndex]}
                        alt={`${title} - Image ${currentImageIndex + 1}`}
                        className={styles.propertyImage}
                    />

                    {/* Image counter badge */}
                    <div className={`${styles.badge} ${styles.imageCounterBadge}`}>
                        <Camera size={12} className={styles.iconCamera} strokeWidth={2.5}/>
                        <span>{currentImageIndex + 1}/{images.length}</span>
                    </div>

                    {/* Available from badge */}
                    <div className={`${styles.badge} ${styles.availableBadge}`}>
                        <Key size={12} className={styles.iconKey} strokeWidth={2.5}/>
                        <span>Available {formatDate(availableFrom)}</span>
                    </div>

                    {/* Energy rating badge */}
                    <div className={`${styles.energyBadge} ${getEnergyRatingClass(energyRating)}`}>
                        <Zap size={12} className={styles.iconZap} strokeWidth={2.5}/>
                        <span>Energy: {energyRating}</span>
                    </div>

                    {/* Heat indicator badge */}
                    <div className={`${styles.badge} ${styles.heatBadge}`}>
                        <Flame size={12} className={styles.iconFlame} strokeWidth={2.5}/>
                        <span>{getHeatLabel(heatScore)}</span>
                    </div>

                    {/* Virtual tour badge - if available */}
                    {virtualTour && (
                        <div className={`${styles.badge} ${styles.virtualTourBadge}`}>
                            <span>Virtual Tour</span>
                        </div>
                    )}

                    {/* Image navigation */}
                    {images.length > 1 && (
                        <>
                            <button
                                onClick={prevImage}
                                className={`${styles.imageNavButton} ${styles.imageNavButtonPrev}`}
                                aria-label="Previous image"
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor">
                                    <path fillRule="evenodd"
                                          d="M12.707 5.293a1 1 0 010 1.414L9.414 10l3.293 3.293a1 1 0 01-1.414 1.414l-4-4a1 1 0 010-1.414l4-4a1 1 0 011.414 0z"
                                          clipRule="evenodd"/>
                                </svg>
                            </button>
                            <button
                                onClick={nextImage}
                                className={`${styles.imageNavButton} ${styles.imageNavButtonNext}`}
                                aria-label="Next image"
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor">
                                    <path fillRule="evenodd"
                                          d="M7.293 14.707a1 1 0 010-1.414L10.586 10 7.293 6.707a1 1 0 011.414-1.414l4 4a1 1 0 010 1.414l-4 4a1 1 0 01-1.414 0z"
                                          clipRule="evenodd"/>
                                </svg>
                            </button>
                        </>
                    )}

                    {/* Main Interaction Icons in one column on the right */}
                    <div className={styles.interactionColumn}>
                        {/* Heart/Favorite Icon */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => setFavorite(!favorite)}
                            aria-label={favorite ? "Remove from favorites" : "Add to favorites"}
                        >
                            <Heart
                                size={16}
                                className={favorite ? `${styles.iconHeart} ${styles.active}` : styles.iconHeart}
                                strokeWidth={2}
                            />
                            <span>{savedCount}</span>
                        </button>

                        {/* Like Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => {
                                setLiked(!liked);
                                if (disliked) setDisliked(false);
                            }}
                            aria-label={liked ? "Unlike" : "Like"}
                        >
                            <ThumbsUp
                                size={16}
                                className={liked ? `${styles.iconThumbsUp} ${styles.active}` : styles.iconThumbsUp}
                                strokeWidth={2}
                            />
                            <span>{liked ? likes + 1 : likes}</span>
                        </button>

                        {/* Dislike Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => {
                                setDisliked(!disliked);
                                if (liked) setLiked(false);
                            }}
                            aria-label={disliked ? "Remove dislike" : "Dislike"}
                        >
                            <ThumbsDown
                                size={16}
                                className={disliked ? `${styles.iconThumbsDown} ${styles.active}` : styles.iconThumbsDown}
                                strokeWidth={2}
                            />
                            <span>{disliked ? dislikes + 1 : dislikes}</span>
                        </button>

                        {/* Comment Button */}
                        <button
                            className={styles.interactionButton}
                            aria-label="View comments"
                        >
                            <MessagesSquare
                                size={16}
                                className={styles.iconComment}
                                strokeWidth={2}
                            />
                            <span>{comments}</span>
                        </button>

                        {/* Message Button */}
                        <button
                            className={styles.interactionButton}
                            aria-label="Send message"
                        >
                            <MessageCircle
                                size={16}
                                className={styles.iconMessage}
                                strokeWidth={2}
                            />
                            <span>{messages}</span>
                        </button>
                    </div>
                </div>
            </div>

            {/* Content section - 50% width on desktop */}
            <div className={styles.contentSection}>

                <div className={styles.titleSection}>

                        <h3 className={styles.propertyTitle}>{title}</h3>
                        <div className={styles.viewsBadge}>
                            <Eye size={12} className={styles.iconEye}/>

                            )}

                                    <PercentCircle size={12} className={styles.iconPercentCircle}/>
                                    <span>Mortgage available</span>
                                </div>

                <div className={styles.propertySpecs}>

                    <div className={styles.specItem}>

                    </div>
                    <div className={styles.specItem}>
                        <Ruler size={14} className={styles.iconRuler} strokeWidth={2}/>
                        <span>{size}</span>
                    </div>
                    {parking > 0 && (
                        <div className={styles.specItem}>
                            <CarFront size={14} className={styles.iconCarFront} strokeWidth={2}/>
                            <span>{parking} {parking === 1 ? 'Parking' : 'Parkings'}</span>
                        </div>
                    )}
                    {furnished && (
                        <div className={styles.specItem}>
                            <DoorOpen size={14} className={styles.iconDoorOpen} strokeWidth={2}/>
                            <span>Furnished</span>
                        </div>
                    )}
                </div>

                {/* Location and Property Type */}
                <div className={styles.metaInfo}>
                    <div className={styles.metaBadge}>
                        <MapPin size={12} className={styles.iconMapPin} strokeWidth={2}/>
                        <span>{address}</span>
                    </div>
                    <div className={styles.metaBadge}>
                        <Building size={12} className={styles.iconBuilding} strokeWidth={2}/>
                        <span>{propertyType} • Built {yearBuilt}</span>
                    </div>
                </div>

                {/* Tags */}
                <div className={styles.tagsSection}>
                    {tags.map((tag, index) => (
                        <span key={index} className={styles.tag}>
              {tag}
            </span>
                    ))}
                </div>

                {/* Description - more compact */}
                <div className={styles.descriptionSection}>
                    <div className={styles.descriptionHeader}>
                        <div className={styles.descriptionTitle}>
                            <span>Description</span>
                        </div>
                        <button
                            onClick={() => setExpanded(!expanded)}
                            className={styles.expandButton}
                            aria-label={expanded ? "Show less" : "Show more"}

                            {expanded ? <ChevronUp size={14} strokeWidth={2.5}/> :
                                <ChevronDown size={14} strokeWidth={2.5}/>}

                                        <div className={styles.mortgageRow}>

                                            <span className={styles.mortgageValue}>{mortgage.monthlyPayment}</span>
                                        </div>

                                            <span className={styles.mortgageLabel}>Interest Rate:</span>
                                            <span className={styles.mortgageValue}>{mortgage.interestRate}</span>
                                        </div>
                                        <div className={styles.mortgageRow}>
                                            <span className={styles.mortgageLabel}>Term:</span>
                                            <span className={styles.mortgageValue}>{mortgage.term}</span>
                                        </div>
                                        <div className={styles.mortgageDetails}>
                                            <span>See full mortgage details</span>
                                        </div>
                                    </div>
                                </div>
                            )}
                        </div>
                    )}
                </div>

                {/* Agent info with Contact button */}
                <div className={styles.agentSection}>
                    <div className={styles.agentInfo}>
                        <div className={styles.agentAvatar}>
                            <div className={styles.avatarCircle}>
                                {agent.name.charAt(0).toUpperCase()}
                            </div>
                            {agent.verified && (
                                <div className={styles.verifiedBadge}>
                                    <Check size={8} className={styles.iconCheck}/>
                                </div>
                            )}
                        </div>
                        <div className={styles.agentDetails}>
                            <div className={styles.agentName}>
                                {agent.name}
                                {agent.verified && (
                                    <span className={styles.agentBadge}>
                    <ShieldCheck size={8} className={styles.iconShield}/>
                    Agent
                  </span>
                                )}
                            </div>
                            <div className={styles.agentCompanyInfo}>
                                <span className={styles.companyName}>{agent.company}</span>
                                <span className={styles.separator}>•</span>
                                <span>{agent.experience}</span>
                            </div>
                        </div>
                    </div>

                    <button
                        className={styles.contactButton}
                        onClick={() => setShowContact(!showContact)}
                    >
                        <MessageCircle size={12} className={styles.iconMessageCircle} strokeWidth={2}/>
                        Contact
                    </button>
                </div>

                {/* Contact form - conditionally shown */}
                {showContact && (
                    <div className={styles.contactForm}>
                        <div className={styles.contactFormTitle}>Contact {agent.name}</div>
                        <div className={styles.contactFormFields}>
                            <input
                                type="text"
                                placeholder="Your name"
                                className={styles.formInput}
                            />
                            <input
                                type="email"
                                placeholder="Your email"
                                className={styles.formInput}
                            />
                            <textarea
                                placeholder="Your message"
                                rows={2}
                                className={styles.formInput}
                                defaultValue={`Hi ${agent.name}, I'm interested in the ${bedrooms}-bedroom ${propertyType.toLowerCase()} at ${address}. Is it still available?`}
                            ></textarea>
                            <div className={styles.formButtonContainer}>
                                <button className={styles.sendButton}>
                                    Send Message
                                </button>
                            </div>
                        </div>
                    </div>
                )}

                {/* Share, Save, Report buttons at the bottom */}
                <div className={styles.actionButtons}>
                    {/* Save Button */}
                    <button
                        className={styles.actionButton}
                        onClick={() => setSaved(!saved)}
                    >
                        <Bookmark
                            size={12}
                            className={saved ? `${styles.bookmarkIcon} ${styles.active}` : styles.bookmarkIcon}
                            strokeWidth={2}
                        />
                        <span>{saved ? "Saved" : "Save"}</span>
                    </button>

                    {/* Share Button */}
                    <button className={styles.actionButton}>
                        <Share2
                            size={12}
                            className={styles.shareIcon}
                            strokeWidth={2}
                        />
                        <span>Share</span>
                    </button>

                    {/* Special button - Virtual Tour or Floorplan */}
                    <button className={styles.specialActionButton}>
                        <Grid
                            size={12}
                            className={styles.iconGrid}
                            strokeWidth={2}
                        />
                        <span>{floorplanAvailable ? "Floorplan" : "Schedule Visit"}</span>
                    </button>
                </div>
            </div>
        </div>
    );
};
