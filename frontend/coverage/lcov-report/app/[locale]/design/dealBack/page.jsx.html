"use client";

import React, { useState } from 'react';
import styles from './page.module.css'; // Keep your existing CSS
import {
    Heart,
    ThumbsUp,
    ThumbsDown,
    MessageCircle,
    Bookmark,
    Share2,
    MapPin,
    Clock,
    Check,
    ChevronDown,
    ChevronUp,
    ExternalLink,
    ShoppingBag,
    ChartLine,
    Camera,
    Flame,
    Tag,
    Eye,
    Flag,
    MessagesSquare,
    ShieldCheck
} from 'lucide-react';

const PremiumDealCard = ({
                             images = ["/images/lenovo.png", "/api/placeholder/500/300", "/api/placeholder/500/300"],
                             title = "Wireless Noise Cancelling Headphones - Premium Sound Quality",
                             currentPrice = "$89.99",
                             originalPrice = "$299.99",
                             discount = "70% OFF",
                             merchantName = "Amazon",
                             location = "Berlin, Friedrichshain 10243",
                             username = "techdeals",
                             userRating = 4.8,
                             userVerified = true,
                             userJoined = "Jan 2023",
                             postedTime = "2h ago",
                             category = "Electronics",
                             subcategory = "Audio",
                             description = "Premium wireless headphones with active noise cancellation. Enjoy crystal clear sound with deep bass and 30 hours of battery life. Includes travel case and fast charging.",
                             views = 243,
                             savedCount = 18,
                             inquiries = 7,
                             likes = 87,
                             dislikes = 4,
                             comments = 32,
                             messages = 12,
                             heatScore = 92,
                             dealExpiry = "Expires in 3 days",
                             dealId = "#A7X912",
                             tags = ["Electronics", "Audio", "Premium", "Wireless"]
                         }) => {
    const [expanded, setExpanded] = useState(false);
    const [saved, setSaved] = useState(false);
    const [favorite, setFavorite] = useState(false);
    const [liked, setLiked] = useState(false);
    const [disliked, setDisliked] = useState(false);
    const [currentImageIndex, setCurrentImageIndex] = useState(0);

    // Next image function
    const nextImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === images.length - 1 ? 0 : prevIndex + 1
        );
    };

    // Previous image function
    const prevImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === 0 ? images.length - 1 : prevIndex - 1
        );
    };

    // Handle likes/dislikes
    const toggleLike = () => {
        setLiked(!liked);
        if (disliked) setDisliked(false);
    };

    const toggleDislike = () => {
        setDisliked(!disliked);
        if (liked) setLiked(false);
    };

    // Determine heat level color and label
    const getHeatLevelClass = (score) => {
        if (score >= 80) return styles.heatIndicatorHot;
        if (score >= 60) return styles.heatIndicatorWarm;
        if (score >= 40) return styles.heatIndicatorGood;
        if (score >= 20) return styles.heatIndicatorCool;
        return styles.heatIndicatorCold;
    };

    const getHeatLabel = (score) => {
        if (score >= 80) return "HOT!";
        if (score >= 60) return "Warm";
        if (score >= 40) return "Good";
        if (score >= 20) return "Cool";
        return "Cold";
    };

    return (
        <div className={styles.classifiedCard}>
            {/* Heat indicator at top */}
            <div className={`${styles.heatIndicator} ${getHeatLevelClass(heatScore)}`}></div>

            {/* Image and interaction section */}
            <div className={styles.imageSection}>
                <div className={styles.imageContainer}>
                    <img
                        src={images[currentImageIndex]}
                        alt={`${title} - Image ${currentImageIndex + 1}`}
                        className={styles.classifiedImage}
                    />

                    {/* Discount badge */}
                    <div className={`${styles.badge} ${styles.discountBadge}`}>
                        {discount}
                    </div>

                    {/* Image counter badge */}
                    <div className={`${styles.badge} ${styles.imageCounterBadge}`}>
                        <Camera size={14} className={styles.iconCamera} strokeWidth={2.5}/>
                        <span>{currentImageIndex + 1}/{images.length}</span>
                    </div>

                    {/* Deal expiry badge */}
                    <div className={`${styles.badge} ${styles.timeBadge}`}>
                        <Clock size={14} className={styles.iconClock} strokeWidth={2.5}/>
                        <span>{dealExpiry}</span>
                    </div>

                    {/* Heat indicator badge */}
                    <div className={`${styles.badge} ${styles.heatBadge}`}>
                        <Flame size={14} className={styles.iconFlame} strokeWidth={2.5}/>
                        <span>{getHeatLabel(heatScore)}</span>
                    </div>

                    {/* Image navigation */}
                    {images.length > 1 && (
                        <>
                            <button
                                onClick={prevImage}
                                className={`${styles.imageNavButton} ${styles.imageNavButtonPrev}`}
                                aria-label="Previous image"
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor">
                                    <path fillRule="evenodd"
                                          d="M12.707 5.293a1 1 0 010 1.414L9.414 10l3.293 3.293a1 1 0 01-1.414 1.414l-4-4a1 1 0 010-1.414l4-4a1 1 0 011.414 0z"
                                          clipRule="evenodd"/>
                                </svg>
                            </button>
                            <button
                                onClick={nextImage}
                                className={`${styles.imageNavButton} ${styles.imageNavButtonNext}`}
                                aria-label="Next image"
                            >
                                <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor">
                                    <path fillRule="evenodd"
                                          d="M7.293 14.707a1 1 0 010-1.414L10.586 10 7.293 6.707a1 1 0 011.414-1.414l4 4a1 1 0 010 1.414l-4 4a1 1 0 01-1.414 0z"
                                          clipRule="evenodd"/>
                                </svg>
                            </button>
                        </>
                    )}

                    {/* Main Interaction Icons in one column on the right */}
                    <div className={styles.interactionColumn}>
                        {/* Heart/Favorite Icon */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => setFavorite(!favorite)}
                            aria-label={favorite ? "Remove from favorites" : "Add to favorites"}
                        >
                            <Heart
                                size={18}
                                className={`${styles.iconHeart} ${favorite ? styles.active : ''}`}
                                strokeWidth={2}
                            />
                            <span>{savedCount}</span>
                        </button>

                        {/* Like Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={toggleLike}
                            aria-label={liked ? "Unlike" : "Like"}
                        >
                            <ThumbsUp
                                size={18}
                                className={`${styles.iconThumbsUp} ${liked ? styles.active : ''}`}
                                strokeWidth={2}
                            />
                            <span>{liked ? likes + 1 : likes}</span>
                        </button>

                        {/* Dislike Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={toggleDislike}
                            aria-label={disliked ? "Remove dislike" : "Dislike"}
                        >
                            <ThumbsDown
                                size={18}
                                className={`${styles.iconThumbsDown} ${disliked ? styles.active : ''}`}
                                strokeWidth={2}
                            />
                            <span>{disliked ? dislikes + 1 : dislikes}</span>
                        </button>

                        {/* Comment Button */}
                        <button
                            className={styles.interactionButton}
                            aria-label="View comments"
                        >
                            <MessagesSquare
                                size={18}
                                className={styles.iconComment}
                                strokeWidth={2}
                            />
                            <span>{comments}</span>
                        </button>

                        {/* Message Button */}
                        <button
                            className={styles.interactionButton}
                            aria-label="Send message"
                        >
                            <MessageCircle
                                size={18}
                                className={styles.iconMessage}
                                strokeWidth={2}
                            />
                            <span>{messages}</span>
                        </button>
                    </div>
                </div>
            </div>

            {/* Content section */}
            <div className={styles.contentSection}>
                {/* Title, price and view details button */}
                <div className={styles.titleSection}>
                    <div className={styles.titleRow}>
                        <h3 className={styles.adTitle}>{title}</h3>
                        <div className={styles.viewsBadge}>
                            <Eye size={12} className={styles.iconEye}/>
                            {views}
                        </div>
                    </div>
                    <div className={styles.priceRow}>
                        <div className={styles.price}>
                            <div>{currentPrice}</div>
                            <div className={styles.originalPrice}>{originalPrice}</div>
                            <div className={styles.discountBadgeInline}>
                                {discount}
                            </div>
                            <div className={styles.merchantInfo}>
                                {merchantName}
                            </div>
                        </div>
                        <button className={styles.viewDetailsButton}>
                            <ExternalLink size={14} className={styles.iconExternalLink} strokeWidth={2}/>
                            View Details
                        </button>
                    </div>
                </div>

                {/* Location and category */}
                <div className={styles.metaInfo}>
                    <div className={styles.metaBadge}>
                        <MapPin size={12} className={styles.iconMapPin} strokeWidth={2}/>
                        <span>{location}</span>
                    </div>
                    <div className={styles.metaBadge}>
                        <Tag size={12} className={styles.iconTag} strokeWidth={2}/>
                        <span>{category} • {subcategory}</span>
                    </div>
                    <div className={styles.metaBadge}>
                        <Clock size={12} className={styles.iconClock} strokeWidth={2}/>
                        <span>ID: {dealId}</span>
                    </div>
                </div>

                {/* Tags */}
                <div className={styles.tagsSection}>
                    {tags.map((tag, index) => (
                        <span key={index} className={styles.tag}>
                            {tag}
                        </span>
                    ))}
                </div>

                {/* Description - more compact */}
                <div className={styles.descriptionSection}>
                    <div className={styles.descriptionHeader}>
                        <div className={styles.descriptionTitle}>
                            <span>Description</span>
                            <span className={styles.conditionLabel}>
                                Premium
                            </span>
                        </div>
                        <button
                            onClick={() => setExpanded(!expanded)}
                            className={styles.expandButton}
                            aria-label={expanded ? "Show less" : "Show more"}
                        >
                            {expanded ? <ChevronUp size={16} strokeWidth={2.5}/> :
                                <ChevronDown size={16} strokeWidth={2.5}/>}
                        </button>
                    </div>

                    <div className={`${styles.descriptionText} ${expanded ? '' : styles.collapsed}`}>
                        {description}
                    </div>

                    {expanded && (
                        <div className={styles.descriptionDetails}>
                            <div className={styles.detailRow}>
                                <span className={styles.detailLabel}>Brand:</span>
                                <span className={styles.detailValue}>Sony</span>
                            </div>
                            <div className={styles.detailRow}>
                                <span className={styles.detailLabel}>Model:</span>
                                <span className={styles.detailValue}>WH-1000XM4</span>
                            </div>
                            <div className={styles.detailRow}>
                                <span className={styles.detailLabel}>Warranty:</span>
                                <span className={styles.detailValue}>1 Year</span>
                            </div>
                            <div className={styles.detailRow}>
                                <span className={styles.detailLabel}>Battery Life:</span>
                                <span className={styles.detailValue}>30 hours</span>
                            </div>
                        </div>
                    )}
                </div>

                {/* Seller info with Contact Seller button */}
                <div className={styles.sellerSection}>
                    <div className={styles.sellerInfo}>
                        <div className={styles.sellerAvatar}>
                            <div className={styles.avatarCircle}>
                                {username.charAt(0).toUpperCase()}
                            </div>
                            {userVerified && (
                                <div className={styles.verifiedBadge}>
                                    <Check size={12} className={styles.iconCheck}/>
                                </div>
                            )}
                        </div>
                        <div className={styles.sellerDetails}>
                            <div className={styles.sellerName}>
                                {username}
                                {userVerified && (
                                    <span className={styles.verifiedLabel}>
                                        <ShieldCheck size={12} className={styles.iconShield}/>
                                        Verified
                                    </span>
                                )}
                            </div>
                            <div className={styles.sellerRating}>
                                <div className={styles.ratingStars}>
                                    {[...Array(5)].map((_, i) => (
                                        <svg key={i}
                                             className={`${styles.star} ${i < Math.floor(userRating) ? styles.starFilled : styles.starEmpty}`}
                                             fill="currentColor" viewBox="0 0 20 20">
                                            <path
                                                d="M9.049 2.927c.3-.921 1.603-.921 1.902 0l1.07 3.292a1 1 0 00.95.69h3.462c.969 0 1.371 1.24.588 1.81l-2.8 2.034a1 1 0 00-.364 1.118l1.07 3.292c.3.921-.755 1.688-1.54 1.118l-2.8-2.034a1 1 0 00-1.175 0l-2.8 2.034c-.784.57-1.838-.197-1.539-1.118l1.07-3.292a1 1 0 00-.364-1.118L2.98 8.72c-.783-.57-.38-1.81.588-1.81h3.461a1 1 0 00.951-.69l1.07-3.292z"></path>
                                        </svg>
                                    ))}
                                </div>
                                <span className={styles.ratingText}>{userRating} ({inquiries} deals)</span>
                            </div>
                        </div>
                    </div>

                    <button className={styles.contactButton}>
                        <MessageCircle size={14} className={styles.iconMessageCircle} strokeWidth={2}/>
                        Contact Seller
                    </button>
                </div>

                {/* Action buttons (Get This Deal) */}
                <div className={styles.dealActionButtons}>
                    <button className={styles.primaryButton}>
                        <ShoppingBag size={16} className={styles.icon} strokeWidth={2}/>
                        Get This Deal
                    </button>
                </div>

                {/* Share, Save, Report buttons at the bottom */}
                <div className={styles.actionButtons}>
                    {/* Save Button */}
                    <button
                        className={styles.actionButton}
                        onClick={() => setSaved(!saved)}
                    >
                        <Bookmark
                            size={16}
                            className={`${styles.bookmarkIcon} ${saved ? styles.active : ''}`}
                            strokeWidth={2}
                        />
                        <span>{saved ? "Saved" : "Save"}</span>
                    </button>

                    {/* Share Button */}
                    <button className={styles.actionButton}>
                        <Share2 size={16} strokeWidth={2}/>
                        <span>Share</span>
                    </button>

                    {/* Report Button */}
                    <button className={styles.actionButton}>
                        <Flag size={16} strokeWidth={2}/>
                        <span>Report</span>
                    </button>
                </div>
            </div>
        </div>
    );
};

export default PremiumDealCard;
