    const [showBooking, setShowBooking] = useState(false);

        setCurrentImageIndex((prevIndex) =>

    const getHeatClass = (score) => {
        if (score >= 80) return styles.heatTopRated;
        if (score >= 60) return styles.heatPopular;
        if (score >= 40) return styles.heatRecommended;
        if (score >= 20) return styles.heatRising;
        return styles.heatNewBusiness;

    // Determine heat label
    const getHeatLabel = (score) => {
        if (score >= 80) return "TOP RATED";
        if (score >= 60) return "Popular";
        if (score >= 40) return "Recommended";
        if (score >= 20) return "Rising";
        return "New Business";

    const renderStars = (rating) => {
        const stars = [];
        const fullStars = Math.floor(rating);
        const hasHalfStar = rating - fullStars >= 0.5;

        for (let i = 0; i < 5; i++) {
            if (i < fullStars) {
                stars.push(<Star key={i} size={12} className={styles.starFull}/>);
            } else if (i === fullStars && hasHalfStar) {
                stars.push(
                    <div key={i} className={styles.starHalf}>
                        <Star size={12} className={styles.starHalfEmpty}/>
                        <div className={styles.starHalfOverlay}>
                            <Star size={12} className={styles.starHalfFull}/>
                        </div>
                    </div>
                );
            } else {
                stars.push(<Star key={i} size={12} className={styles.starEmpty}/>);
            }
        }

        return stars;

                return <Truck size={18} className={styles.serviceIcon}/>;

                return <Utensils size={18} className={styles.serviceIcon}/>;

                return <Briefcase size={18} className={styles.serviceIcon}/>;

                        </svg>

                                    Send Message
                                </button>
                            </div>
                        </div>
                    </div>
                )}

                {/* Booking form - conditionally shown */}
                {showBooking && (
                    <div className={styles.formContainer}>
                        <div className={styles.formTitle}>Book an Appointment</div>
                        <div className={styles.formFields}>
                            <input
                                type="text"
                                placeholder="Your name"
                                className={styles.formInput}
                            />
                            <input
                                type="tel"
                                placeholder="Your phone"
                                className={styles.formInput}
                            />
                            <select className={styles.formSelect}>
                                <option value="">Select service</option>
                                {businessData.services.map((service, index) => (
                                    <option key={index} value={service}>{service}</option>
                                ))}
                            </select>
                            <input
                                type="date"
                                className={styles.formInput}
                            />
                            <select className={styles.formSelect}>
                                <option value="">Select time</option>
                                <option value="morning">Morning (8-12)</option>
                                <option value="afternoon">Afternoon (12-17)</option>
                                <option value="evening">Evening (17-20)</option>
                            </select>
                            <div className={styles.formActions}>
                                <button className={styles.formSubmitButton}>
                                    Request Booking
                                </button>
                            </div>
                        </div>
                    </div>
                )}

                {/* Action Buttons at the bottom */}
                <div className={styles.actionButtonsContainer}>
                    {/* Book Now Button */}
                    <button
                        className={styles.bookButton}
                        onClick={() => setShowBooking(!showBooking)}
                    >
                        <Calendar size={14} className={`${styles.buttonIcon} ${styles.bookButtonIcon}`}/>
                        <span>Book Now</span>
                    </button>

                    {/* Call Button */}
                    <button className={styles.callButton}>
                        <Phone size={14} className={`${styles.buttonIcon} ${styles.callButtonIcon}`}/>
                        <span>Call Business</span>
                    </button>
                </div>
            </div>
        </div>
    );
};

export default ServiceBusinessCard;
