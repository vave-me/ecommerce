import React from 'react';
import {notFound} from 'next/navigation';
import {getTranslations} from 'next-intl/server'; // Import
import {getPropertiesBySlug} from '../../../../../api/propertiesApi'; // Adjust path
import PropertyCard from './PropertyCard';
import styles from '../../../../shared.module.css';

// Helper function (assume defined)
async function resolveProps(props) {
    const params = props.params ? await props.params : {};
    const searchParams = props.searchParams ? await props.searchParams : {};
    return {params, searchParams};
}

export async function generateMetadata(props) {
    let t;
    let slug;
    let property = null; // Changed variable name

    try {
        const {params} = await resolveProps(props);
        const {locale, category} = await params; // Get category
        slug = params.slug;

        t = await getTranslations({locale, namespace: 'PropertyDetailPage'}); // Use correct namespace

        property = await getPropertiesBySlug(slug);

        if (!property) {
            // Corrected log key and text
            return {title: t('metaTitleNotFound')}; // Translated Not Found
        }

        const propertyName = property.name || t('metaTitleDefault');
        // Try to get translated category name from PropertiesPage namespace
        const categoryName = category ? t(`PropertiesPage.category_${category}_label`, {}, {defaultValue: category}) : t('fallbackCategoryName');
        // Corrected description key and text
        const description = property.description?.substring(0, 160) || t('metaDescriptionDefault', {
            propertyName,
            categoryName
        });

        return {
            title: propertyName,
            description: description,
            openGraph: { // Add relevant OG data for properties
                title: propertyName,
                description: description,
                type: 'profile', // OG type for real estate can be 'profile' or specific types
                // 'place:location:latitude': property.geo?.latitude, // Example if geo data exists
                // 'place:location:longitude': property.geo?.longitude,
                images: property.images?.map(img => ({url: img})) || (property.thumbnail ? [{url: property.thumbnail}] : []),
                // Add price if applicable (especially for sale)
                ...(property.priceType === 'sale' && property.basePrice && {
                    'product:price:amount': String(property.basePrice),
                    'product:price:currency': t('common.currencyCode')
                })
            },
            alternates: {
                canonical: `/${locale}/properties/${category}/${slug}`, // Correct canonical URL
            },
        };
    } catch (err) {
        // Corrected log key and text
        console.error(t ? t('logMetaFetchError', {slug}) : `Metadata fetch error for property slug ${slug}:`, err);
        const errorTitle = t ? t('metaTitleError') : "Error Loading Property";
        return {title: errorTitle};
    }
}


export default async function PropertyDetailPage(props) { // Renamed component
    let t; // For potential use in catch block
    let slug; // For potential use in catch block
    let category; // For potential use in catch block
    let locale; // For potential use in catch block

    try {
        const {params} = await resolveProps(props);
        locale = params.locale; // Assign locale
        slug = params.slug; // Assign slug
        category = params.category; // Assign category

        t = await getTranslations({locale, namespace: 'PropertyDetailPage'}); // Use correct namespace

        let propertyData = null;
        try {
            // Corrected log key and text

            propertyData = await getPropertiesBySlug(slug);
            if (!propertyData) {
                // Corrected log key and text
                notFound();
            }
        } catch (err) {
            // Corrected log key and text
            console.error(t('logFetchError', {slug}), err);
            notFound();
        }

        const containerClass = styles?.detailPageContainer || styles.container;

        // --- Corrected JSON-LD for specific Property ---
        const propertyTypeSchema = propertyData.propertyType === 'Apartment' ? 'Apartment' :
            propertyData.propertyType === 'House' ? 'House' :
                'Residence';
        const propertyUrl = `https://www.sfx-markt.de/${locale}/properties/${category}/${slug}`;

        const propertyJsonLd = {
            '@context': 'https://schema.org',
            '@type': propertyTypeSchema,
            name: propertyData.name || t('fallbackPropertyName'),
            description: propertyData.description || '',
            image: propertyData.images?.[0] || propertyData.thumbnail || '',
            url: propertyUrl,
            address: {
                '@type': 'PostalAddress',
                streetAddress: propertyData.address?.street || propertyData.address,
                addressLocality: propertyData.address?.city || undefined,
                addressRegion: propertyData.address?.region || undefined,
                postalCode: propertyData.address?.zip || undefined,
                addressCountry: propertyData.address?.countryCode || 'DE'
            },
            ...(propertyData.size && {
                floorSize: {
                    '@type': 'QuantitativeValue',
                    value: parseFloat(propertyData.size),
                    unitCode: 'MTK'
                }
            }),
            ...(propertyData.bedrooms && {numberOfBedrooms: propertyData.bedrooms}),
            ...(propertyData.bathrooms && {numberOfBathroomsTotal: propertyData.bathrooms}),
            ...(propertyData.yearBuilt && {yearBuilt: propertyData.yearBuilt}),
            ...(propertyData.petsAllowed !== undefined && {petsAllowed: propertyData.petsAllowed}),
            ...(propertyData.amenities && {
                amenityFeature: propertyData.amenities.map(a => ({
                    '@type': 'LocationFeatureSpecification',
                    name: a,
                    value: true
                }))
            }),
            offers: {
                '@type': propertyData.priceType === 'rent' ? 'OfferForLease' : 'OfferForSale',
                price: String(propertyData.basePrice ?? '0.00'),
                priceCurrency: t('common.currencyCode'),
                url: propertyUrl,
                ...(propertyData.availableFrom && {availabilityStarts: propertyData.availableFrom}),
                // Include seller/agent info if available
                // 'seller': {
                //     '@type': 'RealEstateAgent', // or Organization
                //     'name': propertyData.agent?.name || undefined
                // }
            }
        };

        return (
            <div className={containerClass}>
                <script
                    type="application/ld+json"
                    dangerouslySetInnerHTML={{__html: JSON.stringify(propertyJsonLd)}}
                />
                {/* Pass property data and locale to the client card component */}
                {/* Ensure PropertyCard component is adapted to receive 'property' prop */}
                <PropertyCard property={propertyData} locale={locale}/>
            </div>
        );
    } catch (e) {
        console.error(`[PropertyDetailPage] Render failed for slug ${slug}:`, e);
        // Attempt to get translations for error page
        let errorTitle = "Error Loading Property";
        let errorMessage = "Could not load the property details. Please try again later.";
        try {
            if (!t) t = await getTranslations({locale: locale || 'en', namespace: "PropertyDetailPage"});
            errorTitle = t('globalErrorTitle');
            errorMessage = t('globalErrorMessage');
        } catch (transErr) {
            console.error("Failed to load translations for error page:", transErr);
        }
        return (
            <div className={styles.containerError}>
                <h1 className={styles.containerErrorTitle}>{errorTitle}</h1>
                <p>{errorMessage}</p>
            </div>
        );
    }
}
