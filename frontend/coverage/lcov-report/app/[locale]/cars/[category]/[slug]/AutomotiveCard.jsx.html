    Users

        images: ["/images/car.jpeg", "/api/placeholder/500/300", "/api/placeholder/500/300"],
        title: "2020 Audi A4 2.0T Premium Plus Quattro",

        accidentFree: true,

        views: 287,
        savedCount: 37,

        comments: 7,
        messages: 19,
        heatScore: 85,
        isCertified: true,
        seating: 5,
        sellerInfo: {

            yearsInBusiness: 15,

            salesCount: 342
        },
        tags: ["Premium Package", "All-Wheel Drive", "Low Miles", "One Owner", "No Accidents"]
    };

    const [expanded, setExpanded] = useState(false);
    const [saved, setSaved] = useState(false);
    const [favorite, setFavorite] = useState(false);
    const [liked, setLiked] = useState(false);
    const [disliked, setDisliked] = useState(false);
    const [currentImageIndex, setCurrentImageIndex] = useState(0);
    const [showContact, setShowContact] = useState(false);

    // Next image function
    const nextImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === vehicleData.images.length - 1 ? 0 : prevIndex + 1
        );
    };

    // Previous image function
    const prevImage = () => {
        setCurrentImageIndex((prevIndex) =>
            prevIndex === 0 ? vehicleData.images.length - 1 : prevIndex - 1
        );
    };

    // Determine heat level class
    const getHeatClass = (score) => {
        if (score >= 80) return styles.heatHot;
        if (score >= 60) return styles.heatPopular;
        if (score >= 40) return styles.heatGoodDeal;
        if (score >= 20) return styles.heatFairPrice;
        return styles.heatJustListed;
    };

    // Determine heat label
    const getHeatLabel = (score) => {
        if (score >= 80) return "HOT!";
        if (score >= 60) return "Popular";
        if (score >= 40) return "Good Deal";
        if (score >= 20) return "Fair Price";
        return "Just Listed";
    };

    return (
        <div className={styles.container}>
            {/* Heat indicator at top */}
            <div className={`${styles.heatIndicator} ${getHeatClass(vehicleData.heatScore)}`}></div>

            {/* Image and interaction section - 50% width on desktop */}

                {vehicleData.isCertified && (

                    <div className={`${styles.badge} ${styles.darkBadge} ${styles.imageCounterBadge}`}>
                        <Camera size={14} className="text-gray-300 mr-1"/>
                        <span>{currentImageIndex + 1}/{vehicleData.images.length}</span>
                    </div>

                    {/* Mileage badge */}
                    <div className={`${styles.badge} ${styles.darkBadge} ${styles.mileageBadge}`}>

                    </div>

                    {/* Year and condition badge */}
                    <div className={`${styles.badge} ${styles.yearConditionBadge}`}>
                        <Calendar size={14} className="mr-1"/>
                        <span>{vehicleData.year} • {vehicleData.condition}</span>
                    </div>

                    {/* Heat indicator badge */}
                    <div className={`${styles.badge} ${styles.darkBadge} ${styles.heatBadge}`}>
                        <Flame size={14} className="text-orange-500 mr-1"/>
                        <span>{getHeatLabel(vehicleData.heatScore)}</span>

                    {vehicleData.warranty && (
                        <div className={`${styles.badge} ${styles.warrantyBadge}`}>

                        </div>

                    {/* Image navigation */}
                    <button
                        onClick={prevImage}
                        className={`${styles.navButton} ${styles.prevButton}`}
                        aria-label="Previous image"
                    >
                        <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20"
                             fill="currentColor">
                            <path fillRule="evenodd"
                                  d="M12.707 5.293a1 1 0 010 1.414L9.414 10l3.293 3.293a1 1 0 01-1.414 1.414l-4-4a1 1 0 010-1.414l4-4a1 1 0 011.414 0z"
                                  clipRule="evenodd"/>
                        </svg>
                    </button>
                    <button
                        onClick={nextImage}
                        className={`${styles.navButton} ${styles.nextButton}`}
                        aria-label="Next image"
                    >
                        <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20"
                             fill="currentColor">
                            <path fillRule="evenodd"
                                  d="M7.293 14.707a1 1 0 010-1.414L10.586 10 7.293 6.707a1 1 0 011.414-1.414l4 4a1 1 0 010 1.414l-4 4a1 1 0 01-1.414 0z"
                                  clipRule="evenodd"/>
                        </svg>
                    </button>

                    {/* Main Interaction Icons in one column on the right */}
                    <div className={styles.interactionColumn}>
                        {/* Heart/Favorite Icon */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => setFavorite(!favorite)}
                        >
                            <Heart
                                size={20}
                                className={favorite ? styles.favoriteIconActive : styles.favoriteIcon}
                            />
                            <span className={styles.iconText}>{vehicleData.savedCount}</span>
                        </button>

                        {/* Like Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => {
                                setLiked(!liked);
                                if (disliked) setDisliked(false);
                            }}
                        >
                            <ThumbsUp
                                size={20}
                                className={liked ? styles.likeIconActive : styles.likeIcon}
                            />
                            <span className={styles.iconText}>{liked ? vehicleData.likes + 1 : vehicleData.likes}</span>
                        </button>

                        {/* Dislike Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => {
                                setDisliked(!disliked);
                                if (liked) setLiked(false);
                            }}
                        >
                            <ThumbsDown
                                size={20}
                                className={disliked ? styles.dislikeIconActive : styles.dislikeIcon}
                            />
                            <span
                                className={styles.iconText}>{disliked ? vehicleData.dislikes + 1 : vehicleData.dislikes}</span>
                        </button>

                        {/* Comment Button */}
                        <button className={styles.interactionButton}>
                            <MessagesSquare
                                size={20}
                                className={styles.commentIcon}
                            />
                            <span className={styles.iconText}>{vehicleData.comments}</span>
                        </button>

                        {/* Message Button */}
                        <button
                            className={styles.interactionButton}
                            onClick={() => setShowContact(!showContact)}
                        >
                            <MessageCircle
                                size={20}
                                className={styles.messageIcon}
                            />
                            <span className={styles.iconText}>{vehicleData.messages}</span>
                        </button>
                    </div>
                </div>
            </div>

            {/* Content section - 50% width on desktop */}
            <div className={styles.contentSection}>
                {/* Title, price and view details button */}
                <div className={styles.titleContainer}>
                    <div className={styles.titleRow}>
                        <h3 className={styles.title}>{vehicleData.title}</h3>
                        <div className={styles.viewsBadge}>
                            <Eye size={12} className="mr-0.5"/>
                            {vehicleData.views}
                        </div>
                    </div>
                    <div className={styles.priceRow}>
                        <div className={styles.priceContainer}>
                            <div className={styles.price}>{vehicleData.price}</div>
                            <div className={styles.financingBadge}>
                                <PiggyBank size={12} className="mr-0.5"/>
                                <span>Financing Available</span>
                            </div>
                        </div>
                        <button className={styles.detailsButton}>
                            <ExternalLink size={12} className="mr-1"/>
                            View Details
                        </button>
                    </div>
                </div>

                {/* Vehicle specs summary */}
                <div className={styles.specsGrid}>
                    <div className={styles.specItem}>
                        <Settings size={14} className={styles.specIcon}/>
                        <span className={styles.specText}>{vehicleData.engine}</span>
                    </div>
                    <div className={styles.specItem}>
                        <Settings size={14} className={styles.specIcon}/>
                        <span className={styles.specText}>{vehicleData.transmission}</span>

                    <div className={styles.specItem}>

                            <KeyRound size={12} className={styles.positiveDetailIcon}/>
                            <span>One Owner</span>

                            <ShieldCheck size={12} className={styles.positiveDetailIcon}/>
                            <span>No Accidents</span>
                        </div>

                </div>

                {/* Colors and VIN */}
                <div className={styles.infoContainer}>
                    <div className={styles.infoItem}>
                        <div className={`${styles.colorSwatch} ${styles.exteriorSwatch}`}></div>
                        <span>Ext: {vehicleData.exteriorColor}</span>
                    </div>
                    <div className={styles.infoItem}>
                        <div className={`${styles.colorSwatch} ${styles.interiorSwatch}`}></div>
                        <span>Int: {vehicleData.interiorColor}</span>
                    </div>
                    <div className={styles.infoItem}>
                        <span>VIN: {vehicleData.vin.slice(-6)}</span>
                    </div>
                    <div className={styles.infoItem}>
                        <MapPin size={12} className={styles.detailItemIcon}/>
                        <span>{vehicleData.address}</span>
                    </div>
                </div>

                {/* Tags */}
                <div className={styles.tagsContainer}>
                    {vehicleData.tags.map((tag, index) => (
                        <span key={index} className={styles.tag}>
              {tag}
            </span>
                    ))}
                </div>

                {/* Description - more compact */}
                <div className={styles.descriptionContainer}>
                    <div className={styles.descriptionHeader}>
                        <div className={styles.descriptionTitle}>
                            <span>Description</span>
                        </div>
                        <button
                            onClick={() => setExpanded(!expanded)}
                            className={styles.expandButton}
                        >
                            {expanded ? <ChevronUp size={14}/> : <ChevronDown size={14}/>}
                        </button>
                    </div>

                    <div className={`${styles.descriptionText} ${!expanded ? styles.lineClamp : ''}`}>
                        {vehicleData.description}
                    </div>

                    {expanded && (
                        <div className={styles.featuresSection}>
                            <div className={styles.featuresTitle}>Features & Equipment:</div>
                            <div className={styles.featuresGrid}>
                                {["Navigation System", "Heated Seats", "Sunroof", "Apple CarPlay", "LED Headlights", "Backup Camera"].map((feature, index) => (
                                    <div key={index} className={styles.featureItem}>
                                        <Check size={12} className={styles.featureIcon}/>
                                        <span>{feature}</span>
                                    </div>
                                ))}
                            </div>
                        </div>
                    )}
                </div>

                {/* Seller info with Contact button */}
                <div className={styles.sellerContainer}>
                    <div className={styles.sellerInfo}>
                        <div className={styles.sellerAvatar}>
                            <div className={styles.avatarCircle}>
                                {vehicleData.sellerInfo.name.charAt(0).toUpperCase()}
                            </div>
                            {vehicleData.sellerInfo.verified && (
                                <div className={styles.verifiedBadge}>
                                    <Check size={8} className={styles.verifiedIcon}/>
                                </div>
                            )}
                        </div>
                        <div>
                            <div className={styles.sellerName}>
                                {vehicleData.sellerInfo.name}

                                    <span className={styles.userTypeBadge}>
                    <ShieldCheck size={8} className="mr-0.5"/>
                                        {vehicleData.sellerInfo.type}

                                )}

                                <span className={styles.statSeparator}>•</span>

                    >
                        <MessageCircle size={12} className="mr-1"/>

                </div>

                {/* Contact form - conditionally shown */}
                {showContact && (

                        <div className={styles.formFields}>

                                type="text"
                                placeholder="Your name"
                                className={styles.formInput}
                            />
                            <input
                                type="email"
                                placeholder="Your email"
                                className={styles.formInput}
                            />
                            <input
                                type="tel"
                                placeholder="Your phone (optional)"
                                className={styles.formInput}
                            />
                            <textarea
                                placeholder="Your message"
                                rows={2}
                                className={styles.formInput}
                                defaultValue={`I'm interested in this ${vehicleData.year} ${vehicleData.make} ${vehicleData.model}. Please contact me with more information.`}
                            ></textarea>
                            <div className={styles.formActions}>
                                <button className={styles.sendButton}>
                                    Send Message
                                </button>
                            </div>
                        </div>
                    </div>
                )}

                {/* Action Buttons at the bottom */}
                <div className={styles.actionButtonsContainer}>
                    {/* Save Button */}
                    <button
                        className={`${styles.actionButton} ${styles.saveButton}`}
                        onClick={() => setSaved(!saved)}
                    >
                        <Bookmark
                            size={12}
                            className={saved ? styles.saveIconActive : styles.saveIcon}
                        />
                        <span>{saved ? "Saved" : "Save"}</span>
                    </button>

                    {/* Share Button */}
                    <button
                        className={`${styles.actionButton} ${styles.shareButton}`}
                    >
                        <Share2
                            size={12}
                            className={styles.shareIcon}
                        />
                        <span>Share</span>
                    </button>

                    {/* Schedule Test Drive Button */}
                    <button
                        className={`${styles.actionButton} ${styles.testDriveButton}`}
                    >
                        <Car
                            size={12}
                            className={styles.testDriveIcon}
                        />
                        <span>Test Drive</span>
                    </button>
                </div>
            </div>
        </div>
    );
};
