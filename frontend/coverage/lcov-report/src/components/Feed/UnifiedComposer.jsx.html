"use client";
import React, { useState, useRef, useCallback, useEffect } from 'react';
import { useTranslations } from 'next-intl';
import { Image as ImageIcon, Mic, Send, X, Check } from '@/icons';
import { useAuth } from '../../context/AuthContext';
import styles from './UnifiedComposer.module.css';
import assistantsApi from '../../api/client/assistantsApi';
import { fetchProductsByFilters } from '../../api/productsApi';

/**
 * UnifiedComposer - AI Co-pilot Chat Interface
 * Designed to match the Advanced Command Cockpit UI
 */
const UnifiedComposer = ({ onFilterUpdate }) => {
  const { user } = useAuth();
  const t = useTranslations('AI');
  
  // State
  const [inputText, setInputText] = useState('');
  const [messages, setMessages] = useState([
    { role: 'ai', text: t('greeting', 'Hello! How can I help you filter our products today?') }
  ]);
  const [isLoading, setIsLoading] = useState(false);
  const [mediaFiles, setMediaFiles] = useState([]);
  const [conversationId, setConversationId] = useState(null);
  const [activeAssistantId, setActiveAssistantId] = useState(null);
  
  // Refs
  const chatBoxRef = useRef(null);
  const fileInputRef = useRef(null);
  const inputRef = useRef(null);
  
  // Load active assistant on mount
  useEffect(() => {
    const loadActiveAssistant = async () => {
      try {
        console.log('Loading active assistants...');
        const res = await assistantsApi.getActiveAssistants({ limit: 1 });
        console.log('Active assistants response:', res);
        
        if (res.success && res.data?.assistants?.length > 0) {
          const assistantId = res.data.assistants[0].id;
          console.log('Setting active assistant ID:', assistantId);
          setActiveAssistantId(assistantId);
        } else {
          console.error('No active assistants found in response:', res);
        }
      } catch (error) {
        console.error('Failed to load assistant:', error);
      }
    };
    loadActiveAssistant();
  }, []);
  
  // Auto-scroll to bottom when new messages arrive
  useEffect(() => {
    if (chatBoxRef.current) {
      chatBoxRef.current.scrollTop = chatBoxRef.current.scrollHeight;
    }
  }, [messages]);
  
  // Handle file selection
  const handleFileSelect = useCallback((e) => {
    const files = Array.from(e.target.files);
    setMediaFiles(prev => [...prev, ...files]);
  }, []);
  
  // Remove media file
  const handleRemoveMedia = useCallback((index) => {
    setMediaFiles(prev => prev.filter((_, i) => i !== index));
  }, []);
  
  // Send message to AI
  const sendMessage = async (message, context = {}) => {
    // Ensure we have an assistant ID first
    if (!activeAssistantId) {
      console.error('No active assistant ID available');
      throw new Error('No assistant available. Please refresh the page.');
    }
    
    let convoId = conversationId;
    
    // Create conversation if needed
    if (!convoId) {
      try {
        console.log('Creating conversation with assistant:', activeAssistantId);
        const createRes = await assistantsApi.createConversation({
          assistantId: activeAssistantId,
          initialContext: context
        });
        
        console.log('Create conversation response:', createRes);
        
        if (!createRes.success) {
          throw new Error(createRes.error || 'Failed to create conversation');
        }
        
        convoId = createRes.data?.conversation_id || 
                 createRes.data?.conversationId;
        
        if (!convoId) {
          console.error('No conversation ID in response:', createRes);
          throw new Error('Invalid conversation response');
        }
        
        console.log('Setting conversation ID:', convoId);
        setConversationId(convoId);
      } catch (error) {
        console.error('Failed to create conversation:', error);
        throw new Error('Failed to start conversation');
      }
    }
    
    // Send chat message
    const chatRes = await assistantsApi.chatWithConversation({
      conversationId: convoId,
      chatBody: { message, context }
    });
    
    if (!chatRes.success) {
      throw new Error(chatRes.error || 'Chat failed');
    }
    
    return chatRes.data?.response || chatRes.data?.message || 'Sorry, I could not process that.';
  };
  
  // Handle form submission
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    if (!inputText.trim() || isLoading) return;
    
    const userMessage = inputText.trim();
    setInputText('');
    setIsLoading(true);
    
    // Add user message to chat
    setMessages(prev => [...prev, { role: 'user', text: userMessage }]);
    
    try {
      // Send to AI and get response
      const aiResponse = await sendMessage(userMessage, {
        timestamp: new Date().toISOString(),
        hasMedia: String(mediaFiles.length > 0) // Convert boolean to string for protobuf
      });
      
      // Add AI response to chat
      setMessages(prev => [...prev, { role: 'ai', text: aiResponse }]);
      
      // Parse response for filter updates
      parseFiltersFromResponse(userMessage, aiResponse);
      
      // Clear media after sending
      setMediaFiles([]);
    } catch (error) {
      console.error('AI chat error:', error);
      const errorMessage = error.message || t('error', 'I apologize, but I encountered an error. Please try again.');
      setMessages(prev => [...prev, { 
        role: 'ai', 
        text: errorMessage
      }]);
    } finally {
      setIsLoading(false);
    }
  };
  
  // Parse AI response and user query for filter updates
  const parseFiltersFromResponse = async (query, response) => {
    const lowerQuery = query.toLowerCase();
    const lowerResponse = response.toLowerCase();
    const filters = {};
    
    // Check for category mentions
    const categories = ['wallboxes', 'led panels', 'smart home', 'solar', 'batteries', 'electrical', 'lighting', 'automation'];
    categories.forEach(cat => {
      if (lowerQuery.includes(cat) || lowerResponse.includes(cat)) {
        filters.categories = [cat];
      }
    });
    
    // Get available brands dynamically
    try {
      const result = await fetchProductsByFilters({ pageSize: 100 });
      const availableBrands = new Set();
      if (result?.products) {
        result.products.forEach(product => {
          if (product.brand && product.brand.trim()) {
            availableBrands.add(product.brand.toLowerCase());
          }
        });
      }
      
      // Check for brand mentions from available brands
      availableBrands.forEach(brand => {
        if (lowerQuery.includes(brand) || lowerResponse.includes(brand)) {
          if (!filters.brands) filters.brands = [];
          // Find the original case-sensitive brand name
          const originalBrand = result.products.find(p => p.brand && p.brand.toLowerCase() === brand)?.brand;
          if (originalBrand) {
            filters.brands.push(originalBrand);
          }
        }
      });
    } catch (error) {
      console.error('Failed to fetch brands for AI parsing:', error);
      // Fallback to common brands
      const commonBrands = ['siemens', 'schneider', 'abb', 'zaptec', 'philips', 'osram', 'bosch', 'samsung'];
      commonBrands.forEach(brand => {
        if (lowerQuery.includes(brand) || lowerResponse.includes(brand)) {
          if (!filters.brands) filters.brands = [];
          filters.brands.push(brand.charAt(0).toUpperCase() + brand.slice(1));
        }
      });
    }
    
    // Check for price ranges
    const pricePatterns = [
      /under (\d+)/,
      /below (\d+)/,
      /less than (\d+)/,
      /max (\d+)/,
      /up to (\d+)/,
      /between (\d+) and (\d+)/,
      /(\d+) to (\d+)/,
      /above (\d+)/,
      /over (\d+)/,
      /more than (\d+)/,
      /min (\d+)/
    ];
    
    for (const pattern of pricePatterns) {
      const match = lowerQuery.match(pattern);
      if (match) {
        if (pattern.toString().includes('between') || pattern.toString().includes('to')) {
          filters.price = [parseInt(match[1]), parseInt(match[2])];
        } else if (pattern.toString().includes('above') || pattern.toString().includes('over') || pattern.toString().includes('more than') || pattern.toString().includes('min')) {
          filters.price = [parseInt(match[1]), 5000];
        } else {
          filters.price = [0, parseInt(match[1])];
        }
        break;
      }
    }
    
    // Apply filters if any were found
    if (Object.keys(filters).length > 0 && onFilterUpdate) {
      onFilterUpdate(filters);
      
      // Add confirmation message
      setMessages(prev => [...prev, { 
        role: 'ai', 
        text: t('filtersApplied', "I've updated the filters based on your request.") 
      }]);
    }
  };
  
  // Handle voice recording (placeholder)
  const handleVoiceRecord = () => {
    console.log('Voice recording not yet implemented');
  };
  
  return (
    <div className={styles.aiCopilotContent}>
      {/* Chat Box */}
      <div className={styles.aiChatBox} ref={chatBoxRef}>
        {messages.map((msg, index) => (
          <div key={index} className={`${styles.aiMessage} ${styles[msg.role]}`}>
            <div className={`${styles.aiMessageBubble} ${styles[msg.role]}`}>
              {msg.text}
            </div>
          </div>
        ))}
        
        {isLoading && (
          <div className={`${styles.aiMessage} ${styles.loading}`}>
            <div className={styles.loadingBubble}>
              <div className={styles.loadingDot} />
              <div className={styles.loadingDot} />
              <div className={styles.loadingDot} />
            </div>
          </div>
        )}
      </div>
      
      {/* Media Preview */}
      {mediaFiles.length > 0 && (
        <div className={styles.mediaPreview}>
          {mediaFiles.map((file, index) => (
            <div key={index} className={styles.previewItem}>
              {file.type.startsWith('image/') ? (
                <img src={URL.createObjectURL(file)} alt={file.name} />
              ) : (
                <video src={URL.createObjectURL(file)} />
              )}
              <button
                className={styles.removeMediaButton}
                onClick={() => handleRemoveMedia(index)}
                aria-label={t('removeMedia')}
              >
                ×
              </button>
            </div>
          ))}
        </div>
      )}
      
      {/* Input Form */}
      <form className={styles.aiForm} onSubmit={handleSubmit}>
        <div className={styles.aiInputWrapper}>
          <textarea
            ref={inputRef}
            className={styles.aiInput}
            placeholder={t('placeholder', 'Type your request...')}
            value={inputText}
            onChange={(e) => setInputText(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                handleSubmit(e);
              }
            }}
            rows={3}
          />
          
          <div className={styles.aiInputActions}>
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*,video/*"
              multiple
              onChange={handleFileSelect}
              className={styles.hiddenInput}
            />
            <button
              type="button"
              className={styles.aiActionButton}
              onClick={() => fileInputRef.current?.click()}
              title={t('addImage')}
            >
              <ImageIcon />
            </button>
            <button
              type="button"
              className={styles.aiActionButton}
              onClick={handleVoiceRecord}
              title={t('recordVoice')}
            >
              <Mic />
            </button>
          </div>
          
          <button
            type="submit"
            className={styles.aiSendButton}
            disabled={!inputText.trim() || isLoading}
            title={t('sendMessage')}
          >
            {isLoading ? <div className={styles.spinner} /> : <Check />}
          </button>
        </div>
      </form>
    </div>
  );
};

export default UnifiedComposer;
