/**
 * Simple toast notification hook
 */
const useToastNotification = () => {
    return useCallback((type, message) => {
        const options = {theme: "colored"};
        switch (type) {
            case "success":
                toast.success(message, options);
                break;
            case "info":
                toast.info(message, options);
                break;
            case "error":
                toast.error(message, options);
                break;
            case "warn":
                toast.warn(message, options);
                break;
            default:
                toast(message, options);
        }
    }, []);
};

/**
 * DealCard - Unified deal card component
 * Modern design with proper server/client separation
 */
const DealCard = memo(({deal, preloadedMedia = null, className = ""}) => {
    // Debug log removed for production

    const isMountedRef = useRef(true);
    const {user} = useAuth();
    const userId = user?.userId;
    const dispatch = useDispatch();
    const showToast = useToastNotification();
    const {handleLike, handleDislike} = useActivityApi();
    const [mediaItems, setMediaItems] = useState([]);
    const [isLoading, setIsLoading] = useState(false);

    // Add state for client-side time calculations to avoid hydration errors
    const [isNew, setIsNew] = useState(false);
    const [isExpiringWithin24h, setIsExpiringWithin24h] = useState(false);

    // Basic validation without debug logs
    useEffect(() => {
        if (!deal) {
            console.error('[DealCard] No deal data provided!');
            return;
        }

        if (typeof deal !== 'object') {
            console.error('[DealCard] Invalid deal data type:', typeof deal);
            return;
        }
    }, [deal]);

    // Initialize media from preloaded data or fetch if needed
    useEffect(() => {
        // If we have preloaded media, use it immediately
        if (preloadedMedia && preloadedMedia.length > 0) {
            setMediaItems(preloadedMedia);
            setIsLoading(false);
            return;
        }

        // Otherwise, fetch media using the same pattern as working routes
        const loadMedia = async () => {
            if (!deal?.id) {
                return;
            }

            setIsLoading(true);
            try {
                const mediaResponse = await getMediaByItem(deal.id);

                if (mediaResponse?.media?.mediaOrder?.length > 0) {
                    const formattedMedia = mediaResponse.media.mediaOrder.map(item => ({
                        id: item.mediaItemId || item.id,
                        url: item.url,
                        type: item.type || 'image',
                        alt: item.altText || deal.name || 'Deal image'
                    }));
                    setMediaItems(formattedMedia);
                } else {
                    // Check deal object for any image data
                    const dealImages = [];
                    if (deal.thumbnail) dealImages.push({url: deal.thumbnail, type: 'image'});
                    if (deal.images?.length > 0) {
                        deal.images.forEach(img => dealImages.push({url: img, type: 'image'}));
                    }
                    if (deal.media?.length > 0) {
                        deal.media.forEach(media => dealImages.push(media));
                    }

                    if (dealImages.length > 0) {
                        setMediaItems(dealImages);
                    }
                }
            } catch (error) {
                console.error('[DealCard] Error loading media:', error);
            } finally {
                setIsLoading(false);
            }
        };

        loadMedia();
    }, [deal?.id, deal?.name, preloadedMedia]);

    // Combine API media with deal media, prioritizing API media
    const resolvedMedia = useMemo(() => {
        // If we have API media, use it
        if (mediaItems && mediaItems.length > 0) {
            return mediaItems;
        }

        // Otherwise, try to use deal media
        if (deal.media && deal.media.length > 0) {
            return deal.media;
        }

        // If deal has images array, use it
        if (deal.images && deal.images.length > 0) {
            const mappedImages = deal.images.map(img => ({url: img}));
            return mappedImages;
        }

        // If deal has thumbnail, use it
        if (deal.thumbnail) {
            const thumbnailMedia = [{url: deal.thumbnail}];
            return thumbnailMedia;
        }

        // No media available - CardImageContainer will handle by returning null
        return [];
    }, [mediaItems, deal?.media, deal?.images, deal?.thumbnail, deal?.id]);

    // Full deal data merger with defaults - using real API data
    const dealWithDefaults = useMemo(() => {
        // Extract actual deal data from API response structure
        const actualDeal = deal?.deal || deal;
        const dealMetrics = actualDeal?.metrics || {};

        // Handle completely missing deal data
        if (!actualDeal) {
            console.error('[DealCard] No deal data provided');
            return {
                id: '',
                name: 'Deal data missing',
                description: 'This deal could not be loaded.',
                basePrice: '0',
                dealPrice: '0',
                added: new Date().toISOString(),
                dealDuration: 86400,
                condition: 'new',
                thumbnail: '/images/default-deal.png',
                dealUrl: '',
                media: [{url: '/images/default-deal.png', type: 'image'}],
                metrics: {likes: 0, dislikes: 0, comments: 0, shares: 0, views: 0}
            };
        }

        // Try to extract pricing from various possible API response fields
        let basePrice = '0';
        let dealPrice = '0';

        // Priority order for base price
        if (actualDeal?.basePrice) basePrice = String(actualDeal.basePrice);
        else if (actualDeal?.originalPrice) basePrice = String(actualDeal.originalPrice);
        else if (actualDeal?.price) basePrice = String(actualDeal.price);

        // Priority order for deal price
        if (actualDeal?.dealPrice) dealPrice = String(actualDeal.dealPrice);
        else if (actualDeal?.salePrice) dealPrice = String(actualDeal.salePrice);
        else if (actualDeal?.discountedPrice) dealPrice = String(actualDeal.discountedPrice);
        else if (actualDeal?.price) dealPrice = String(actualDeal.price);

        // Extract merchant name ONLY from API - no defaults or URL parsing
        const merchantName = actualDeal?.merchantName || actualDeal?.merchant || actualDeal?.storeName || actualDeal?.seller;

        return {
            // Real API data from the deal object
            ...actualDeal,
            id: actualDeal?.id || '',
            name: actualDeal?.name || actualDeal?.title || 'Unnamed Deal',
            description: actualDeal?.description || 'No description available.',
            basePrice: basePrice,
            dealPrice: dealPrice,
            added: deal?.createdAt || deal?.updatedAt || new Date().toISOString(),
            dealDuration: parseInt(actualDeal?.dealDuration || '0', 10) || 86400,
            condition: actualDeal?.condition || 'new',
            thumbnail: actualDeal?.thumbnail || '/images/default-deal.png',
            dealUrl: actualDeal?.dealUrl || actualDeal?.userId || '',
            brand: actualDeal?.brand || '',
            categorySlug: actualDeal?.categorySlug || 'deals',
            categoryId: actualDeal?.categoryId || '',
            // Convert string metrics to numbers
            likeCount: parseInt(dealMetrics?.likesCount || '0', 10),
            dislikeCount: parseInt(dealMetrics?.dislikesCount || '0', 10),
            tags: actualDeal?.tags || [],
            shippingCost: actualDeal?.shippingCost || '0',
            lat: actualDeal?.lat || null,
            lng: actualDeal?.lng || null,
            // Calculate popularity based on views
            popularity: parseInt(dealMetrics?.visitedCount || '0', 10) > 50 ? 2 :
                parseInt(dealMetrics?.visitedCount || '0', 10) > 20 ? 1 : 0,
            wishlistCount: parseInt(dealMetrics?.addedToWishlistCount || '0', 10),
            comments: [],
            messages: 0,
            hasVariants: actualDeal?.hasVariants || false,
            variants: actualDeal?.variants || [],
            shipping: actualDeal?.shipping || [],
            userId: actualDeal?.userId || '',

            // Deal-specific data from API - only use actual merchant name from API
            merchantName: merchantName,
            location: (actualDeal?.lat && actualDeal?.lng && (actualDeal.lat !== 0 || actualDeal.lng !== 0))
                ? `${actualDeal.lat}, ${actualDeal.lng}`
                : null,
            // Use fetched media items or fallback
            media: resolvedMedia,
            // Real metrics from API
            metrics: {
                likes: parseInt(dealMetrics?.likesCount || '0', 10),
                dislikes: parseInt(dealMetrics?.dislikesCount || '0', 10),
                comments: parseInt(dealMetrics?.commentsCount || '0', 10),
                shares: parseInt(dealMetrics?.sharedCount || '0', 10),
                views: parseInt(dealMetrics?.visitedCount || '0', 10),
            },
        };
    }, [deal, resolvedMedia]);

    // Destructure properties for easier access
    const {
        id,
        name,
        description,
        added,
        basePrice,
        dealPrice,
        dealDuration,
        dealUrl,
        condition,
        tags,
        hasVariants,
        variants,
        shipping,
        location,
        merchantName,
        media,
        metrics,
        popularity
    } = dealWithDefaults;

    // State management
    const [interactionState, setInteractionState] = useState({
        favorite: false,
        liked: false,
        disliked: false,
        showComments: false,
        currentMediaIndex: 0,
    });

    const [selectionState, setSelectionState] = useState({
        shipping: '',
        variant: '',
        quantity: 1,
    });

    const [discountPercentage, setDiscountPercentage] = useState(0);

    // Derived data - Format price with localization (no debug logs)
    const formattedDealPrice = useMemo(() => {
        const price = parseInt(dealPrice, 10) || 0;
        return price.toLocaleString('en-US', {style: 'currency', currency: 'EUR'});
    }, [dealPrice]);

    const formattedBasePrice = useMemo(() => {
        const price = parseInt(basePrice, 10) || 0;
        return price.toLocaleString('en-US', {style: 'currency', currency: 'EUR'});
    }, [basePrice]);

    // Time-related calculations
    const timeAgo = useMemo(() => (
        added ? dayjs(added).fromNow() : ''
    ), [added]);

    const expirationTime = useMemo(() => {
        if (!added || !dealDuration) return null;
        const addedDate = dayjs(added);
        return addedDate.add(dealDuration, 'second');
    }, [added, dealDuration]);

    const isExpired = useMemo(() => {
        if (!expirationTime) return false;
        return dayjs().isAfter(expirationTime);
    }, [expirationTime]);

    // Move time-dependent calculations to useEffect to avoid hydration errors
    useEffect(() => {
        // Calculate if deal is new (less than 24 hours old)
        const isNewDeal = deal?.isNew || (added && dayjs().diff(dayjs(added), 'hour') < 24);
        setIsNew(isNewDeal);

        // Calculate if deal is expiring within 24 hours
        const isExpiring = !isExpired && expirationTime && dayjs().add(24, 'hour').isAfter(expirationTime);
        setIsExpiringWithin24h(isExpiring);
    }, [added, expirationTime, isExpired, deal?.isNew]);

    // Calculate discount percentage
    useEffect(() => {
        if (basePrice && dealPrice) {
            const basePriceNum = parseInt(basePrice, 10);
            const dealPriceNum = parseInt(dealPrice, 10);
            if (basePriceNum > 0) {
                const discount = Math.round(((basePriceNum - dealPriceNum) / basePriceNum) * 100);
                setDiscountPercentage(discount);
            }
        }
    }, [basePrice, dealPrice]);

    // Calculate savings amount
    const savingsAmount = useMemo(() => {
        if (parseInt(basePrice) > parseInt(dealPrice)) {
            return parseInt(basePrice) - parseInt(dealPrice);
        }
        return 0;
    }, [basePrice, dealPrice]);

    // Format savings
    const formattedSavings = useMemo(() => {
        if (savingsAmount > 0) {
            return savingsAmount.toLocaleString('en-US', {style: 'currency', currency: 'EUR'});
        }
        return null;
    }, [savingsAmount]);

    // Cleanup on unmount
    useEffect(() => {
        return () => {
            isMountedRef.current = false;
        };
    }, []);

    // Get popularity indicator
    const getPopularityStatus = useCallback(() => {
        if (popularity >= 2) return {icon: <Flame size={14}/>, label: "Hot Deal", className: styles.hotBadge};
        if (popularity >= 0.5) return {icon: <Star size={14}/>, label: "Popular", className: styles.popularBadge};
        if (popularity <= -2) return {icon: <Snowflake size={14}/>, label: "Cold", className: styles.coldBadge};
        return null;
    }, [popularity]);

    const popularityStatus = useMemo(() => getPopularityStatus(), [getPopularityStatus]);

    // Event handlers
    const handleCommentClick = useCallback(() => {
        setInteractionState(prev => ({
            ...prev,
            showComments: !prev.showComments
        }));
    }, []);

    const handleFavorite = useCallback(() => {
        setInteractionState(prev => ({
            ...prev,
            favorite: !prev.favorite
        }));
    }, []);

    const handleLikeClick = useCallback(() => {
        if (!userId) {
            showToast("warn", "Please log in to like deals.");
            return;
        }

        if (!id) return;

        handleLike(id, userId).catch(() => {
            // handle error if needed
        });

        setInteractionState(prev => ({
            ...prev,
            liked: true,
            disliked: false
        }));
    }, [id, userId, handleLike, showToast]);

    const handleDislikeClick = useCallback(() => {
        if (!userId) {
            showToast("warn", "Please log in to dislike deals.");
            return;
        }

        if (!id) return;

        handleDislike(id, userId).catch(() => {
            // handle error if needed
        });

        setInteractionState(prev => ({
            ...prev,
            liked: false,
            disliked: true
        }));
    }, [id, userId, handleDislike, showToast]);

    const handleOpenMessage = useCallback(() => {
        if (!id || !dealWithDefaults.userId) return;
        dispatch(openMessageModal({
            recipientId: dealWithDefaults.userId,
            itemId: id
        }));
    }, [id, dealWithDefaults.userId, dispatch]);

    const handleShippingChange = useCallback((value) => {
        setSelectionState(prev => ({
            ...prev,
            shipping: value
        }));
    }, []);

    const handleVariantChange = useCallback((value) => {
        setSelectionState(prev => ({
            ...prev,
            variant: value
        }));
    }, []);

    const handleQuantityChange = useCallback((value) => {
        setSelectionState(prev => ({
            ...prev,
            quantity: value
        }));
    }, []);

    const handleMediaNavigation = useCallback((direction, e) => {
        e?.stopPropagation(); // Prevent triggering card click
        setInteractionState(prev => ({
            ...prev,
            currentMediaIndex: direction === 'next'
                ? (prev.currentMediaIndex + 1) % media.length
                : (prev.currentMediaIndex - 1 + media.length) % media.length
        }));
    }, [media.length]);

    const handleKeyboardNavigation = useCallback((e) => {
        if (e.key === 'ArrowLeft') {
            handleMediaNavigation('prev', e);
        } else if (e.key === 'ArrowRight') {
            handleMediaNavigation('next', e);
        }
    }, [handleMediaNavigation]);

    const handleGoToDeal = useCallback(() => {
        if (dealUrl) {
            window.open(dealUrl, '_blank', 'noopener,noreferrer');
        }
    }, [dealUrl]);

    return (
        <div className={`${styles.card} ${className}`} tabIndex={0} onKeyDown={handleKeyboardNavigation}>
            {/* User header if we have user info */}
            {dealWithDefaults.userId && (
                <UserHeader
                    userData={{
                        id: dealWithDefaults.userId,
                        authorName: dealWithDefaults.authorName || dealWithDefaults.merchantName || 'Deal Poster',
                        authorUsername: dealWithDefaults.authorUsername || '',
                        avatar: dealWithDefaults.avatar || '/images/default-avatar.jpg',
                        verified: dealWithDefaults.verified || false
                    }}
                    userId={dealWithDefaults.userId}
                    entityId={id}
                    entityType="deal"
                    timeAgo={timeAgo}
                    location={location}
                    onEdit={(dealId) => {/* TODO: Implement edit handler */
                    }}
                    onArchive={(dealId) => {/* TODO: Implement archive handler */
                    }}
                    onDelete={(dealId) => {/* TODO: Implement delete handler */
                    }}
                    onToggleVisibility={(dealId) => {/* TODO: Implement visibility toggle */
                    }}
                />
            )}

            {/* STATUS INDICATORS (top-right overlay) */}
            <div className={styles.statusIndicators}>
                {isNew && (
                    <span className={styles.newBadge} aria-label="New deal">
                        NEW
                    </span>
                )}

                {discountPercentage > 0 && (
                    <div className={styles.discountBadge} aria-label={`${discountPercentage}% discount`}>
                        <span className={styles.discountPercentage}>
                            -{discountPercentage}%
                        </span>
                        {formattedSavings && (
                            <span className={styles.discountSavings}>
                                Save {formattedSavings}
                            </span>
                        )}
                    </div>
                )}

                {isExpiringWithin24h && (
                    <span className={styles.expiringSoonBadge} aria-label="Expires soon">
                        <Clock size={12} aria-hidden="true"/>
                        <span>Expires Soon</span>
                    </span>
                )}
            </div>

            {/* IMAGE */}
            <CardImageContainer
                media={media}
                currentIndex={interactionState.currentMediaIndex}
                isLoading={isLoading}

                alt={media[interactionState.currentMediaIndex]?.alt || name}
                onClick={handleGoToDeal}
                onNavigate={handleMediaNavigation}
                ariaLabel="Deal images"
                badges={
                    dealPrice && parseInt(dealPrice) > 0 && (
                        <div aria-label={`Price: ${formattedDealPrice}`}>
                            <ShoppingBag size={12} aria-hidden="true"/>
                            <span>{formattedDealPrice}</span>
                        </div>
                    )
                }
            />

            {/* Deal info badges - Using shared BadgeRow component */}
            <BadgeRow
                type="deals"
                category={dealWithDefaults.categorySlug || 'General'}
                merchantName={merchantName}
                views={metrics.views}
                createdAt={added}
                isHot={popularityStatus && popularity >= 2}
            />

            {/* DEAL TITLE */}
            <CardTitle
                title={name}
                href={`/deals/${id}`}
            />

            {/* PRICE ROW - Only show if we have actual price data */}
            {(parseInt(dealPrice) > 0 || parseInt(basePrice) > 0) && (
                <PriceLocationRow
                    price={parseInt(dealPrice) > 0 ? formattedDealPrice : ''}
                    priceLabel={`Price: ${formattedDealPrice}`}
                    location={location}
                    lat={dealWithDefaults.lat}
                    lng={dealWithDefaults.lng}
                    merchantName={merchantName}
                    dealUrl={dealUrl}
                    additionalContent={basePrice && parseInt(basePrice) > parseInt(dealPrice) && parseInt(basePrice) > 0 && (
                        <span className={styles.originalPrice} aria-label={`Original price: ${formattedBasePrice}`}>
                            {formattedBasePrice}
                        </span>
                    )}
                />
            )}

            {/* DESCRIPTION */}
            <ExpandableDescription text={description}/>

            {/* TAGS */}
            <Tags
                tags={tags}
                ariaLabel="Deal tags"
            />

            {/* SHIPPING & VARIANTS SECTIONS (if available) */}
            {(hasVariants || shipping.length > 0) && (
                <div className={styles.optionsSection}>
                    {/* Shipping options */}
                    {shipping.length > 0 && (
                        <div className={styles.shippingOptions}>
                            <div className={styles.optionTitle}>Shipping Options</div>

                            <div className={styles.optionButtons} role="radiogroup" aria-label="Shipping options">
                                {shipping.map((option, index) => (
                                    <button
                                        key={index}
                                        className={`${styles.optionButton} ${
                                            selectionState.shipping === option.id
                                                ? styles.optionButtonSelected
                                                : ''
                                        }`}
                                        onClick={() => handleShippingChange(option.id)}
                                        aria-checked={selectionState.shipping === option.id}
                                        role="radio"
                                    >
                                        {option.name || option.label}
                                    </button>
                                ))}
                            </div>
                        </div>
                    )}

                    {/* Variants */}
                    {hasVariants && variants.length > 0 && (
                        <div className={styles.variants}>
                            <div className={styles.optionTitle}>Variants</div>
                            <div className={styles.optionButtons} role="radiogroup" aria-label="Deal variants">
                                {variants.map((variant, index) => (
                                    <button
                                        key={index}
                                        className={`${styles.optionButton} ${
                                            selectionState.variant === variant.id
                                                ? styles.optionButtonSelected
                                                : ''
                                        }`}
                                        onClick={() => handleVariantChange(variant.id)}
                                        aria-checked={selectionState.variant === variant.id}
                                        role="radio"
                                    >
                                        {variant.name || variant.label}
                                    </button>
                                ))}
                            </div>
                        </div>
                    )}

                    {/* Quantity */}
                    <div className={styles.quantity}>
                        <div className={styles.optionTitle}>Quantity</div>
                        <div className={styles.quantityButtons} role="radiogroup" aria-label="Quantity selection">
                            {[1, 2, 3, 4, 5].map((qty) => (
                                <button
                                    key={qty}
                                    className={`${styles.quantityButton} ${
                                        selectionState.quantity === qty
                                            ? styles.quantityButtonSelected
                                            : ''
                                    }`}
                                    onClick={() => handleQuantityChange(qty)}
                                    aria-checked={selectionState.quantity === qty}
                                    role="radio"
                                >
                                    {qty}
                                </button>
                            ))}
                        </div>
                    </div>
                </div>
            )}

            <Engagement
                userId={userId}
                liked={interactionState.liked}
                disliked={interactionState.disliked}
                favorite={interactionState.favorite}
                data={{
                    metrics: {
                        likes: metrics.likes,
                        dislikes: metrics.dislikes,
                        comments: metrics.comments,
                        shares: metrics.shares,
                        saved: metrics.saved,
                        messages: 0
                    }
                }}
                onLike={handleLikeClick}
                onMessageClick={handleOpenMessage}
                onCommentClick={handleCommentClick}
                onDislike={handleDislikeClick}
                onFavorite={handleFavorite}
                variant="product"
            />

            {/* COMMENTS */}
            {interactionState.showComments && (
                <div className={styles.commentsWrapper}>
                    <div>Comments section - CommentsSetup component needs proper import</div>
                </div>
            )}
        </div>
    );
});

DealCard.displayName = 'DealCard';
