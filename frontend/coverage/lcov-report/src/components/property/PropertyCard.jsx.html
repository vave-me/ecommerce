"use client";

import React, { useState, useCallback, useMemo, useEffect, useRef, memo } from 'react';
import Link from 'next/link';
import dayjs from 'dayjs';
import relativeTime from 'dayjs/plugin/relativeTime';
import {
    Bath,
    BedDouble,
    Camera,
    Clock,
    DoorOpen,
    Eye,
    Flame,
    Key,
    MapPin,
    Ruler,
    Zap,
    Building,
    CarFront,
    Star,
    Snowflake,
    BadgeCheck,
    MessageCircle
} from '@/icons';

import styles from './PropertyCard.module.css';
import { useAuth } from '../../context/AuthContext';
import { useDispatch } from 'react-redux';
import { openMessageModal } from '../../redux/slices/modalsSlice';
import UserHeader from '../../app/[locale]/design/UserHeader';
import { Engagement } from '../shared/Engagement';
import CommentsSetup from '../../features/Comments/CommentsSetup';
import { toast } from 'react-toastify';
import { getMediaByItem } from '../../api/mediaApi';
import useActivityApi from '../../hooks/useActivityApi';
import BadgeRow from '../shared/BadgeRow';
import PriceLocationRow from '../shared/PriceLocationRow';
import ExpandableDescription from '../shared/ExpandableDescription';
import CardTitle from '../shared/CardTitle';
import CardImageContainer from '../shared/CardImageContainer';

// Extend dayjs with relative time plugin
dayjs.extend(relativeTime);

/**
 * PropertyCard - Unified property card component following DealCard pattern
 * 
 * @param {Object} props - Component props
 * @param {Object} props.property - Property data object
 * @param {Array} props.preloadedMedia - Pre-fetched media from server
 * @param {string} props.className - Additional CSS classes
 */
const PropertyCard = memo(({ 
    property: propertyProp, 
    preloadedMedia = null, 
    className = ""
}) => {


    const isMountedRef = useRef(true);
    const { user } = useAuth();
    const userId = user?.userId;
    const dispatch = useDispatch();
    const { handleLike, handleDislike } = useActivityApi();

    // Process property data with comprehensive defaults - Following AutomotiveCard pattern
    const propertyWithDefaults = useMemo(() => {
        const actualProperty = propertyProp?.property || propertyProp || {};
        const propertyMetrics = actualProperty?.metrics || {};
        
        // Convert string metrics to numbers
        const convertMetricToNumber = (value) => parseInt(value || '0', 10);

        return {
            // Core property data
            id: actualProperty.id || '',
            name: actualProperty.name || actualProperty.title,
            description: actualProperty.description ,
            basePrice: actualProperty.listingPrice,
            priceType: actualProperty.priceType || actualProperty.listingType || 'rent',
            added: propertyProp?.createdAt || propertyProp?.updatedAt || new Date().toISOString(),
            
            // Property-specific data
            propertyType: actualProperty.propertyType || actualProperty.categorySlug || 'Property',
            bedrooms: parseInt(actualProperty.bedrooms ?? "0", 10),
            bathrooms: parseInt(actualProperty.bathrooms ?? "0", 10),
            yearBuilt: actualProperty.yearBuilt || null,
            size: actualProperty.squareFeet || actualProperty.area || actualProperty.size || null,
            energyRating: actualProperty.energyRating || null,
            parking: parseInt(actualProperty.parking || '0', 10),
            
            // Location
            address: actualProperty.address || "",
            city: actualProperty.city || "",
            postalCode: actualProperty.postalCode || "",
            location: actualProperty.address || actualProperty.city || '',
            lat: actualProperty.lat || null,
            lng: actualProperty.lng || null,
            
            // Media handling
            thumbnail: actualProperty.thumbnail || '/images/default-real-estate.webp',
            media: actualProperty.media || [],
            images: actualProperty.images || [],
            
            // Features
            furnished: actualProperty.furnished || false,
            petsAllowed: actualProperty.petsAllowed || false,
            
            // Engagement metrics
            likeCount: convertMetricToNumber(propertyMetrics.likesCount),
            dislikeCount: convertMetricToNumber(propertyMetrics.dislikesCount),
            wishlistCount: convertMetricToNumber(propertyMetrics.addedToWishlistCount),
            comments: [],
            
            // Status and features
            tags: actualProperty.tags || [],
            popularity: convertMetricToNumber(propertyMetrics.visitedCount) > 50 ? 2 :
                       convertMetricToNumber(propertyMetrics.visitedCount) > 20 ? 1 : 0,
            verified: actualProperty.verified !== false,
            
            // User data
            userId: actualProperty.userId || '',
            authorName: actualProperty.authorName || actualProperty.ownerName || 'Property Owner',
            authorUsername: actualProperty.authorUsername || actualProperty.username || '',
            avatar: actualProperty.avatar || actualProperty.userAvatar || '/images/default-avatar.jpg',

            categorySlug: actualProperty.categorySlug,
            categoryId: actualProperty.categoryId || '',
            
            // Calculated fields
            isNew: dayjs().diff(dayjs(actualProperty.added || actualProperty.createdAt), 'hour') < 24,
            
            // Status flags
            isPromoted: actualProperty.middlemanService || false,
            isPremium: actualProperty.featured || false,
            status: actualProperty.status || 'active',
            
            // Real metrics from API
            metrics: {
                likes: convertMetricToNumber(propertyMetrics.likesCount),
                dislikes: convertMetricToNumber(propertyMetrics.dislikesCount),
                comments: convertMetricToNumber(propertyMetrics.commentsCount),
                shares: convertMetricToNumber(propertyMetrics.sharedCount),
                views: convertMetricToNumber(propertyMetrics.visitedCount),
            }
        };
    }, [propertyProp]);

    // Destructure for easier access - Following AutomotiveCard pattern
    const {
        id, name, description, basePrice, priceType, added,
        propertyType, bedrooms, bathrooms, yearBuilt, size, energyRating, parking,
        address, city, postalCode, location, tags, popularity,
        verified, isNew, metrics, categorySlug, thumbnail, furnished, petsAllowed
    } = propertyWithDefaults;

    // State management - Following AutomotiveCard pattern
    const [interactionState, setInteractionState] = useState({
        favorite: false,
        liked: false,
        disliked: false,
        showComments: false,
        currentMediaIndex: 0,
    });

    const [mediaItems, setMediaItems] = useState([]);
    const [isLoadingMedia, setIsLoadingMedia] = useState(false);

    // Media resolution with fallback handling - Following AutomotiveCard pattern
    useEffect(() => {
        const resolveMedia = async () => {

            // Helper function to validate and process media URLs - Now returns null for invalid URLs
            const processMediaUrl = (url) => {
                if (!url || url === '' || url === 'undefined' || url === 'null') {
                    return null; // Return null instead of fallback - CardImageContainer will handle
                }
                return url;
            };

            if (!!preloadedMedia && preloadedMedia?.length > 0) {
                const processedMedia = preloadedMedia
                    .map(item => ({
                        ...item,
                        url: processMediaUrl(item.url)
                    }))
                    .filter(item => item.url !== null); // Filter out null URLs
                setMediaItems(processedMedia);
                return;
            }

            if (propertyWithDefaults.media?.length > 0) {
                const processedMedia = propertyWithDefaults.media
                    .map(item => ({
                        ...item,
                        url: processMediaUrl(item.url)
                    }))
                    .filter(item => item.url !== null); // Filter out null URLs
                setMediaItems(processedMedia);
                return;
            }

            if (id) {
                setIsLoadingMedia(true);
                try {
                    const mediaResponse = await getMediaByItem(id);
                    if (mediaResponse?.media?.mediaOrder?.length > 0) {
                        const processedMedia = mediaResponse.media.mediaOrder
                            .map(item => ({
                                url: processMediaUrl(item.url),
                                type: item.type || 'image',
                                alt: item.altText || name || 'Property image'
                            }))
                            .filter(item => item.url !== null); // Filter out null URLs
                        setMediaItems(processedMedia);
                    } else {
                        // Try thumbnail, but only if it's valid
                        const thumbnailUrl = processMediaUrl(thumbnail);
                        if (thumbnailUrl) {
                            setMediaItems([{ 
                                url: thumbnailUrl, 
                                type: 'image',
                                alt: name || 'Property image'
                            }]);
                        } else {
                            setMediaItems([]); // No valid media
                        }
                    }
                } catch (error) {
                    console.error('[PropertyCard] Error fetching media:', error);
                    // Try thumbnail as fallback, but only if it's valid
                    const thumbnailUrl = processMediaUrl(thumbnail);
                    if (thumbnailUrl) {
                        setMediaItems([{ 
                            url: thumbnailUrl, 
                            type: 'image',
                            alt: name || 'Property image'
                        }]);
                    } else {
                        setMediaItems([]); // No valid media
                    }
                } finally {
                    setIsLoadingMedia(false);
                }
            } else {
                // Try thumbnail, but only if it's valid
                const thumbnailUrl = processMediaUrl(thumbnail);
                if (thumbnailUrl) {
                    setMediaItems([{ 
                        url: thumbnailUrl, 
                        type: 'image',
                        alt: name || 'Property image'
                    }]);
                } else {
                    setMediaItems([]); // No valid media
                }
            }
        };

        resolveMedia();
    }, [id, !!preloadedMedia, preloadedMedia, propertyWithDefaults.media, thumbnail]);

    // Price calculations - Following AutomotiveCard pattern
    const formattedPrice = useMemo(() => {
        const price = parseInt(basePrice, 10);
        return price.toLocaleString('de-DE', { style: 'currency', currency: 'EUR' });
    }, [basePrice]);

    const timeAgo = useMemo(() => (
        added ? dayjs(added).fromNow() : ''
    ), [added]);

    // Get popularity status - Following AutomotiveCard pattern
    const getPopularityStatus = useCallback(() => {
        if (popularity >= 2) return { icon: <Flame size={14}/>, label: "Hot Property", className: styles.hotBadge };
        if (popularity >= 1) return { icon: <Star size={14}/>, label: "Popular", className: styles.popularBadge };
        if (popularity <= -2) return { icon: <Snowflake size={14}/>, label: "Cold", className: styles.coldBadge };
        return null;
    }, [popularity]);

    const popularityStatus = getPopularityStatus();

    // Navigation handler - Following DealCard pattern
    const handleGoToProperty = useCallback(() => {
        window.location.href = `/properties/${id}`;
    }, [id]);

    // Event handlers - Following AutomotiveCard pattern
    const handleCommentClick = useCallback(() => {
        setInteractionState(prev => ({
            ...prev,
            showComments: !prev.showComments
        }));
    }, []);

    const handleFavorite = useCallback(() => {
        if (!userId) {
            toast.warn("Please log in to save properties.");
            return;
        }
        setInteractionState(prev => ({
            ...prev,
            favorite: !prev.favorite
        }));
    }, [userId]);

    const handleLikeClick = useCallback(() => {
        if (!userId) {
            toast.warn("Please log in to like properties.");
            return;
        }
        if (!id) return;
        
        handleLike(id, userId).catch(() => {});
        setInteractionState(prev => ({
            ...prev,
            liked: true,
            disliked: false
        }));
    }, [id, userId, handleLike]);

    const handleDislikeClick = useCallback(() => {
        if (!userId) {
            toast.warn("Please log in to dislike properties.");
            return;
        }
        if (!id) return;
        
        handleDislike(id, userId).catch(() => {});
        setInteractionState(prev => ({
            ...prev,
            liked: false,
            disliked: true
        }));
    }, [id, userId, handleDislike]);

    const handleOpenMessage = useCallback(() => {
        if (!id || !propertyWithDefaults.userId) return;
        dispatch(
            openMessageModal({
                itemId: id,
                recipientId: propertyWithDefaults.userId,
            })
        );
    }, [id, propertyWithDefaults.userId, dispatch]);

    const handleMediaNavigation = useCallback((direction, e) => {
        e?.stopPropagation();
        
        setInteractionState(prev => ({
            ...prev,
            currentMediaIndex: direction === 'next'
                ? (prev.currentMediaIndex + 1) % mediaItems.length
                : (prev.currentMediaIndex - 1 + mediaItems.length) % mediaItems.length
        }));
    }, [mediaItems.length]);

    // Cleanup on unmount - Following AutomotiveCard pattern
    useEffect(() => {
        return () => {
            isMountedRef.current = false;
        };
    }, []);

    // currentMedia is no longer needed - CardImageContainer handles empty arrays by returning null

    // Don't render if no property data
    if (!propertyWithDefaults || !id) {
        return null;
    }

    return (
        <div className={`${styles.card} ${className}`}>
            {/* Status indicators - Following DealCard pattern */}
            <div className={styles.statusIndicators}>
                {isNew && (
                    <div className={styles.newBadge}>
                        NEW
                    </div>
                )}
                
                {propertyWithDefaults.isPremium && (
                    <div className={styles.premiumBadge}>
                        <Star size={10} />
                        PREMIUM
                    </div>
                )}
                
                {popularityStatus && popularity >= 2 && (
                    <div className={styles.hotBadge}>
                        <Flame size={10} />
                        HOT
                    </div>
                )}
            </div>

            {/* User header if we have owner info */}
            {propertyWithDefaults.userId && (
                <UserHeader 
                    userData={{
                        id: propertyWithDefaults.userId,
                        authorName: propertyWithDefaults.authorName,
                        authorUsername: propertyWithDefaults.authorUsername,
                        avatar: propertyWithDefaults.avatar,
                        verified: propertyWithDefaults.verified || false
                    }}
                    userId={propertyWithDefaults.userId}
                    entityId={id}
                    entityType="property"
                    timeAgo={timeAgo}
                    location={location}
                                    onEdit={(propertyId) => {/* TODO: Implement edit handler */}}
                onArchive={(propertyId) => {/* TODO: Implement archive handler */}}
                onDelete={(propertyId) => {/* TODO: Implement delete handler */}}
                onToggleVisibility={(propertyId) => {/* TODO: Implement visibility toggle */}}
                />
            )}

            {/* Image Container - Following DealCard pattern exactly */}
            <CardImageContainer
                media={mediaItems}
                currentIndex={interactionState.currentMediaIndex}
                isLoading={isLoadingMedia}

                alt={name || 'Property image'}
                onClick={handleGoToProperty}
                onNavigate={handleMediaNavigation}
                ariaLabel="Property images"
                badges={
                    formattedPrice && (
                        <div aria-label={`Price: ${formattedPrice}`}>
                            <Key size={12} aria-hidden="true" />
                            <span>{formattedPrice}</span>
                        </div>
                    )
                }
            />

            {/* Property info badges - Using shared BadgeRow component */}
            <BadgeRow
                type="properties"
                category={categorySlug}
                merchantName={propertyWithDefaults.authorName}
                views={metrics.views}
                createdAt={added}
                isHot={popularityStatus && popularity >= 2}
            />

            {/* PROPERTY TITLE - Following DealCard pattern */}
            <CardTitle 
                title={name}
                href={`/properties/${id}`}
            />

            {/* Rent/price and location - Using shared PriceLocationRow component */}
            <PriceLocationRow
                price={`${formattedPrice}/${priceType}`}
                priceLabel={`Price: ${formattedPrice} per ${priceType}`}
                location={location}
                lat={propertyWithDefaults.lat}
                lng={propertyWithDefaults.lng}
                merchantName={propertyWithDefaults.authorName}
            />

            {/* DESCRIPTION - Following DealCard pattern */}
            <ExpandableDescription text={description} />


            {/* ENGAGEMENT */}
            <Engagement
                userId={userId}
                liked={interactionState.liked}
                disliked={interactionState.disliked}
                favorite={interactionState.favorite}
                data={{
                    metrics: {
                        likes: metrics.likes,
                        dislikes: metrics.dislikes,
                        comments: metrics.comments,
                        shares: metrics.shares,
                        saved: metrics.saved,
                        messages: 0
                    }
                }}
                onLike={handleLikeClick}
                onMessageClick={handleOpenMessage}
                onCommentClick={handleCommentClick}
                onDislike={handleDislikeClick}
                onFavorite={handleFavorite}
                variant="product"
            />

            {/* COMMENTS - Following DealCard pattern */}
            {interactionState.showComments && (
                <div className={styles.commentsWrapper}>
                    <div>Comments section - CommentsSetup component needs proper import</div>
                </div>
            )}
        </div>
    );
});

PropertyCard.displayName = 'PropertyCard';

export default PropertyCard;
