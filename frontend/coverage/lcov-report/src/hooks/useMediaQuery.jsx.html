"use client";
import { useState, useEffect, useMemo, useRef } from 'react';
// Unified breakpoints - consistent with CSS
export const BREAKPOINTS = {
  mobile: 768,
  tablet: 1024,
  desktop: 1280,
  wide: 1536,
};
// Centralized state management for performance
let globalWindowSize = { width: undefined, height: undefined };
let listeners = new Set();
let isInitialized = false;
let resizeTimeoutId = null;
// Debounced resize handler to prevent excessive updates
const handleGlobalResize = () => {
  if (resizeTimeoutId) {
    clearTimeout(resizeTimeoutId);
  }
  resizeTimeoutId = setTimeout(() => {
    const newSize = {
      width: window.innerWidth,
      height: window.innerHeight
    };
    // Only update if size actually changed
    if (newSize.width !== globalWindowSize.width || newSize.height !== globalWindowSize.height) {
      globalWindowSize = newSize;
      listeners.forEach(callback => callback(newSize));
    }
  }, 16); // ~60fps throttling
};
// Initialize global resize listener only once
const initializeGlobalListener = () => {
  if (typeof window === 'undefined' || isInitialized) return;
  // Set initial size
  globalWindowSize = {
    width: window.innerWidth,
    height: window.innerHeight
  };
  window.addEventListener('resize', handleGlobalResize, { passive: true });
  isInitialized = true;
};
/**
 * Performance-optimized hook for handling media queries in a SSR-friendly way
 * Uses a single global resize listener shared across all instances
 * 
 * @param {Object} options - Configuration options
 * @returns {Object} Media query matching results
 */
export function useMediaQuery(options = {}) {
  const {
    defaultValue = false,
    breakpoints = BREAKPOINTS,
  } = options;
  const [windowSize, setWindowSize] = useState(globalWindowSize);
  const listenerRef = useRef(null);
  // Setup global listener and component-specific callback
  useEffect(() => {
    if (typeof window === 'undefined') return;
    initializeGlobalListener();
    // Create stable callback reference
    const updateSize = (newSize) => {
      setWindowSize(newSize);
    };
    listenerRef.current = updateSize;
    listeners.add(updateSize);
    // Set initial size if available
    if (globalWindowSize.width !== undefined) {
      setWindowSize(globalWindowSize);
    }
    return () => {
      if (listenerRef.current) {
        listeners.delete(listenerRef.current);
      }
    };
  }, []);
  // Memoized responsive flags with stable references
  const responsiveFlags = useMemo(() => {
    const width = windowSize.width;
    const isValidWidth = typeof width === 'number';
    return {
      isMobile: isValidWidth ? width <= breakpoints.mobile : defaultValue,
      isTablet: isValidWidth 
        ? width > breakpoints.mobile && width <= breakpoints.tablet
        : defaultValue,
      isDesktop: isValidWidth ? width > breakpoints.tablet : defaultValue,
      isWideScreen: isValidWidth ? width >= breakpoints.wide : defaultValue,
    };
  }, [windowSize.width, breakpoints.mobile, breakpoints.tablet, breakpoints.wide, defaultValue]);
  // Stable utility functions
  const utilityFunctions = useMemo(() => ({
    matchesMediaQuery: (query) => {
      if (typeof window === 'undefined' || !window.matchMedia) {
        return defaultValue;
      }
      return window.matchMedia(query).matches;
    },
    isMinWidth: (minWidth) => {
      return typeof windowSize.width === 'number'
        ? windowSize.width >= minWidth
        : defaultValue;
    },
    isMaxWidth: (maxWidth) => {
      return typeof windowSize.width === 'number'
        ? windowSize.width <= maxWidth
        : defaultValue;
    },
  }), [windowSize.width, defaultValue]);
  // Return stable object reference
  return useMemo(() => ({
    // Size flags
    ...responsiveFlags,
    // Raw dimensions
    width: windowSize.width,
    height: windowSize.height,
    // Breakpoint values for calculations
    breakpoints,
    // Utility functions
    ...utilityFunctions,
  }), [responsiveFlags, windowSize, breakpoints, utilityFunctions]);
}
// Optimized mobile-only detection with singleton pattern
let mobileDetectionCache = null;
let mobileDetectionListeners = new Set();
const updateMobileCache = () => {
  const newValue = typeof window !== 'undefined' && window.innerWidth <= BREAKPOINTS.mobile;
  if (newValue !== mobileDetectionCache) {
    mobileDetectionCache = newValue;
    mobileDetectionListeners.forEach(callback => callback(newValue));
  }
};
/**
 * Performance-optimized hook for components that only need mobile detection
 * Uses singleton pattern for maximum efficiency across the app
 */
export function useIsMobileOnly() {
  const [isMobile, setIsMobile] = useState(() => {
    if (typeof window === 'undefined') return false;
    return window.innerWidth <= BREAKPOINTS.mobile;
  });
  useEffect(() => {
    if (typeof window === 'undefined') return;
    const callback = (mobile) => setIsMobile(mobile);
    mobileDetectionListeners.add(callback);
    // Initialize if needed
    if (mobileDetectionCache === null) {
      updateMobileCache();
      if (!isInitialized) {
        initializeGlobalListener();
        listeners.add(updateMobileCache);
      }
    } else {
      setIsMobile(mobileDetectionCache);
    }
    return () => {
      mobileDetectionListeners.delete(callback);
    };
  }, []);
  return isMobile;
}
/**
 * Legacy hook with custom breakpoint support
 * Uses the global resize system for consistency
 */
export function useIsMobile(breakpoint = BREAKPOINTS.mobile) {
  const [isMobile, setIsMobile] = useState(() => {
    if (typeof window === 'undefined') return false;
    return window.innerWidth <= breakpoint;
  });
  useEffect(() => {
    if (typeof window === 'undefined') return;
    const checkMobile = () => {
      const mobile = window.innerWidth <= breakpoint;
      setIsMobile(mobile);
    };
    // Use the same global resize handler for consistency
    listeners.add(checkMobile);
    initializeGlobalListener();
    // Set initial value
    checkMobile();
    return () => {
      listeners.delete(checkMobile);
    };
  }, [breakpoint]);
  return isMobile;
}

/**
 * Custom media query hook with optimized matching
 * Provides matchMedia-based queries with fallback to window.innerWidth
 */
export function useCustomMediaQuery(query) {
  const [matches, setMatches] = useState(false);
  useEffect(() => {
    if (typeof window === 'undefined') return;
    let mediaQuery;
    let isSupported = false;
    try {
      mediaQuery = window.matchMedia(query);
      isSupported = true;
    } catch (e) {
    }
    const handleChange = () => {
      if (isSupported && mediaQuery) {
        setMatches(mediaQuery.matches);
      } else {
        // Fallback logic for common mobile queries
        if (query.includes('max-width: 768px') || query.includes('max-width:768px')) {
          setMatches(window.innerWidth <= 768);
        } else if (query.includes('min-width: 769px') || query.includes('min-width:769px')) {
          setMatches(window.innerWidth >= 769);
        } else {
          setMatches(false);
        }
      }
    };
    // Set initial value
    handleChange();
    if (isSupported && mediaQuery) {
      // Add listener with fallback for older browsers
      try {
        mediaQuery.addEventListener('change', handleChange);
      } catch (e) {
        mediaQuery.addListener(handleChange);
      }
      return () => {
        try {
          mediaQuery.removeEventListener('change', handleChange);
        } catch (e) {
          mediaQuery.removeListener(handleChange);
        }
      };
    } else {
      // Fallback to resize listener for unsupported queries
      window.addEventListener('resize', handleChange, { passive: true });
      return () => window.removeEventListener('resize', handleChange);
    }
  }, [query]);
  return matches;
}

/**
 * Performance-optimized responsive breakpoint hook
 */
export function useResponsiveBreakpoint() {
  const isMobile = useIsMobileOnly();
  const isTablet = useIsTablet();
  const isDesktop = useIsDesktop();
  const isSmallMobile = useIsSmallMobile();
  // Memoize computed properties to prevent unnecessary re-renders
  return useMemo(() => ({
    isMobile,
    isTablet,
    isDesktop,
    isSmallMobile,
    // Computed properties for common combinations
    isMobileOrTablet: isMobile || isTablet,
    isDesktopOrTablet: isDesktop || isTablet,
  }), [isMobile, isTablet, isDesktop, isSmallMobile]);
}
