"use client";
import React, {useState, useEffect} from 'react';
import {useTranslations} from 'next-intl'; // Import hook
import {format} from 'date-fns'; // Use format for specific date display
import {enUS, pl, de} from 'date-fns/locale'; // Import locales
import {
    Bath, BedDouble, Bookmark, Building, Camera, CarFront, Check, ChevronDown, ChevronUp, DoorOpen,
    ExternalLink, Eye, Flame, Grid, Heart, Key, MapPin, MessageCircle, MessagesSquare, PercentCircle,
    Ruler, Share2, ShieldCheck, Sparkles, ThumbsDown, ThumbsUp, Zap // Ensure all used icons are imported
} from 'lucide-react';
import styles from './PropertyCard.module.css'; // Adjust path if needed
 
// Map locales for date-fns
const dateLocales = {en: enUS, pl, de};
 
// --- Default Data Structure (Example using translations) ---
const getDefaultProperty = (t, common_t) => ({
    id: 'preview-prop-123',
    images: ["/images/house.webp", "/api/placeholder/500/300"],
    title: t('defaultTitle'),
    name: t('defaultTitle'),
    basePrice: "1250.00",
    price: "1250", // Raw value for display formatting fallback
    priceSuffix: t('priceSuffixRent'),
    priceType: "rent",
    deposit: `2500 ${common_t('currencySymbol')}`, // Example formatted deposit
    furnished: true,
    yearBuilt: "2018",
    size: "85", // Numeric value
    address: t('defaultAddress'),
    fullAddress: t('defaultFullAddress'),
    propertyType: t('defaultPropertyType'),
    bedrooms: 2,
    bathrooms: 1,
    availableFrom: new Date(Date.now() + 10 * 24 * 60 * 60 * 1000).toISOString(),
    petsAllowed: true, // Assuming 'pets' in original was boolean
    parking: 1,
    amenities: t('defaultAmenities', {returnObjects: true}) || [],
    description: t('defaultDescription'),
    agent: {
        name: t('defaultAgentName'),
        company: t('defaultAgentCompany'),
        verified: true,
        rating: 4.9,
        listings: 32, // Example - this might be dynamic
        experience: t('defaultAgentExperience')
    },
    stats: { // Nested stats structure
        views: 387,
        savedCount: 42,

        likes: 53,
        dislikes: 3,
        comments: 9
    },
    isFavorite: false,
    isPromoted: true,
    isPremium: true,
    tags: t('defaultTags', {returnObjects: true}) || [],
    heatScore: 88,
    energyRating: "B",
    virtualTour: true,
    floorplanAvailable: true,
    mortgage: {
        available: true,
        monthlyPayment: `1450 ${common_t('currencySymbol')}`, // Example formatted
        interestRate: "3.25%",
        term: "30 years" // Keep as string
    }
});
 
// --- Component ---
const PropertyCard = ({property: propertyProp, locale = 'en'}) => {
    const t = useTranslations('PropertyCard');
    const common_t = useTranslations('common'); // For shared terms like currency
 
    // Use passed property or generate default using translations
    const property = propertyProp || getDefaultProperty(t, common_t);
 
    // --- State ---
    const [expanded, setExpanded] = useState(false);
    const [saved, setSaved] = useState(property.isSaved || false); // Use property data if available
    const [favorite, setFavorite] = useState(property.isFavorite || false);
    const [liked, setLiked] = useState(property.isLiked || false);
    const [disliked, setDisliked] = useState(property.isDisliked || false);
    const [currentImageIndex, setCurrentImageIndex] = useState(0);
    const [showContact, setShowContact] = useState(false);
 
    // State for counts (allow optimistic updates)
    const [likesCount, setLikesCount] = useState(property.stats?.likes || 0);
    const [dislikesCount, setDislikesCount] = useState(property.stats?.dislikes || 0);
    const [savedCount, setSavedCount] = useState(property.stats?.savedCount || 0);
    const [commentsCount, setCommentsCount] = useState(property.stats?.comments || 0);
    const [messagesCount, setMessagesCount] = useState(property.stats?.inquiries || 0);
 
    // --- Derived Data & Formatting ---
    const images = property.images?.length > 0 ? property.images : ['/api/placeholder/600/400']; // Fallback
 
    // Price Formatting
    const formattedPrice = property.basePrice ? new Intl.NumberFormat(locale, {
        style: 'currency',
        currency: common_t('currencyCode'),
        minimumFractionDigits: 0,
        maximumFractionDigits: 0,
    }).format(parseFloat(property.basePrice)) : property.price; // Fallback to original string if no basePrice
 
    const priceSuffix = property.priceType === 'rent' ? t('priceSuffixRent') : '';
 
    // Available Date Formatting
    const availableDateFormatted = property.availableFrom ? format(new Date(property.availableFrom), 'PP', {locale: dateLocales[locale] || enUS}) : '';
 
    // Size Formatting
    const formattedSize = property.size ? `${property.size}${t('sizeUnit')}` : '';
 
    // Heat Level & Label
    const heatScore = property.heatScore || 0;
    const getHeatLevelClass = (score) => {
        if (score >= 80) return styles.heatIndicatorHot;
        if (score >= 60) return styles.heatIndicatorPopular;
        if (score >= 40) return styles.heatIndicatorGood;
        if (score >= 20) return styles.heatIndicatorNew;
        return styles.heatIndicatorListed;
    };
    const heatLabel = t(`heatLabel_${(getHeatLevelClass(heatScore).split('-').pop() || 'listed').toLowerCase()}`);
 
    // Energy Rating Class
    const getEnergyRatingClass = (rating) => {
        const ratingUpper = rating?.toUpperCase();
        switch (ratingUpper) {
            case 'A':
                return styles.energyRatingA;
            case 'B':
                return styles.energyRatingB;
            case 'C':
                return styles.energyRatingC;
            case 'D':
                return styles.energyRatingD;
            case 'E':
                return styles.energyRatingE;
            case 'F':
                return styles.energyRatingF;
            case 'G':
                return styles.energyRatingG;
            default:
                return "";
        }
    };
 
    // --- Handlers ---
    const nextImage = () => setCurrentImageIndex((prev) => (prev === images.length - 1 ? 0 : prev + 1));
    const prevImage = () => setCurrentImageIndex((prev) => (prev === 0 ? images.length - 1 : prev - 1));
    // Add API calls within these handlers
    const handleFavoriteToggle = () => setFavorite(!favorite);
    const handleLikeToggle = () => setLiked(!liked); // Add logic for dislike toggle if needed
    const handleDislikeToggle = () => setDisliked(!disliked); // Add logic for like toggle if needed
    const handleSaveToggle = () => setSaved(!saved);
 
    // --- Render ---
    return (
        <div className={styles.propertyCard} aria-labelledby={`property-title-${property.id}`}>
            {/* Heat indicator */}
            <div className={`${styles.heatIndicator} ${getHeatLevelClass(heatScore)}`}></div>
 
            {/* Image Section */}
            <div className={styles.imageSection}>
                {(property.isPromoted || property.isPremium) && (
                    <div className={property.isPremium ? styles.premiumBadge : styles.promotedBadge}>
                        {t(property.isPremium ? 'premiumBadge' : 'promotedBadge')}
                    </div>
                )}
                <div className={styles.imageContainer}>
                    <img src={images[currentImageIndex]}
                         alt={t('imageAlt', {title: property.title || property.name, index: currentImageIndex + 1})}
                         className={styles.propertyImage}/>
                    {/* Badges */}
                    <Badge icon={Camera} text={`${currentImageIndex + 1}/${images.length}`}
                           className={styles.imageCounterBadge}/>
                    {availableDateFormatted &&
                        <Badge icon={Key} text={t('availableFromLabel', {date: availableDateFormatted})}
                               className={styles.availableBadge}/>}
                    {property.energyRating &&
                        <Badge icon={Zap} text={t('energyRatingLabel', {rating: property.energyRating})}
                               className={`${styles.energyBadge} ${getEnergyRatingClass(property.energyRating)}`}/>}
                    <Badge icon={Flame} text={heatLabel} className={styles.heatBadge}/>
                    {property.virtualTour && <Badge text={t('virtualTourBadge')} className={styles.virtualTourBadge}/>}
                    {/* Image Navigation */}
                    {images.length > 1 && (
                        <>
                            <button onClick={prevImage}
                                    className={`${styles.imageNavButton} ${styles.imageNavButtonPrev}`}
                                    aria-label={t('prevImageAriaLabel')}></button>
                            <button onClick={nextImage}
                                    className={`${styles.imageNavButton} ${styles.imageNavButtonNext}`}
                                    aria-label={t('nextImageAriaLabel')}></button>
                        </>
                    )}
                    {/* Interaction Column */}
                    <div className={styles.interactionColumn}>
                        <InteractionButton icon={Heart} active={favorite} onClick={handleFavoriteToggle}
                                           ariaLabel={favorite ? t('unfavoriteAriaLabel') : t('favoriteAriaLabel')}
                                           count={savedCount} t={t}/>
                        <InteractionButton icon={ThumbsUp} active={liked} onClick={handleLikeToggle}
                                           ariaLabel={liked ? t('unlikeAriaLabel') : t('likeAriaLabel')}
                                           count={likesCount} t={t}/>
                        <InteractionButton icon={ThumbsDown} active={disliked} onClick={handleDislikeToggle}
                                           ariaLabel={disliked ? t('removeDislikeAriaLabel') : t('dislikeAriaLabel')}
                                           count={dislikesCount} t={t}/>
                        <InteractionButton icon={MessagesSquare}
                                           ariaLabel={t('viewCommentsAriaLabel', {count: commentsCount})}
                                           count={commentsCount} t={t}/>
                        <InteractionButton icon={MessageCircle} onClick={() => setShowContact(true)}
                                           ariaLabel={t('sendMessageAriaLabel', {count: messagesCount})}
                                           count={messagesCount} t={t}/>
                    </div>
                </div>
            </div>
 
            {/* Content Section */}
            <div className={styles.contentSection}>
                <div className={styles.titleSection}>
                    <div className={styles.titleRow}>
                        <h3 id={`property-title-${property.id}`}
                            className={styles.propertyTitle}>{property.title || property.name}</h3>
                        <div className={styles.viewsBadge} title={t('viewsTitle', {count: property.stats?.views || 0})}>
                            <Eye size={12} className={styles.iconEye} aria-hidden="true"/> {property.stats?.views || 0}
                        </div>
                    </div>
                    <div className={styles.priceRow}>
                        <div className={styles.price}>
                            <div>{formattedPrice}{priceSuffix &&
                                <span className={styles.priceSuffix}> {priceSuffix}</span>}</div>
                            {property.deposit && property.priceType === "rent" &&
                                <div className={styles.depositBadge}>{t('depositLabel')}: {property.deposit}</div>}
                            {property.priceType === "sale" && property.mortgage?.available &&
                                <div className={styles.mortgageBadge}><PercentCircle size={12}
                                                                                     className={styles.iconPercentCircle}/>
                                    <span>{t('mortgageAvailableBadge')}</span></div>}
                        </div>
                        <button className={styles.viewDetailsButton}>
                            <ExternalLink size={12} className={styles.iconExternalLink} strokeWidth={2}
                                          aria-hidden="true"/> {t('viewDetailsButtonText')}
                        </button>
                    </div>
                </div>
 
                {/* Property Specs */}
                <div className={styles.propertySpecs}>
                    <SpecItem icon={BedDouble} value={property.bedrooms}
                              label={t('specBedrooms', {count: property.bedrooms})}/>
                    <SpecItem icon={Bath} value={property.bathrooms}
                              label={t('specBathrooms', {count: property.bathrooms})}/>
                    {formattedSize && <SpecItem icon={Ruler} value={formattedSize}/>}
                    {property.parking > 0 && <SpecItem icon={CarFront} value={property.parking}
                                                       label={t('specParking', {count: property.parking})}/>}
                    {property.furnished !== undefined &&
                        <SpecItem icon={DoorOpen} value={t(property.furnished ? 'specFurnished' : 'specUnfurnished')}/>}
                </div>
 
                {/* Meta Info */}
                <div className={styles.metaInfo}>
                    <MetaItem icon={MapPin} value={property.address}/>
                    <MetaItem icon={Building}
                              value={`${property.propertyType || ''}${property.yearBuilt ? ` • ${t('builtYearLabel')} ${property.yearBuilt}` : ''}`}/>
                </div>
 
                {/* Tags */}
                {property.tags && property.tags.length > 0 && (
                    <div className={styles.tagsSection}>
                        {property.tags.map((tag, index) => <span key={index} className={styles.tag}>{tag}</span>)}
                    </div>
                )}
 
                {/* Description */}
                <div className={styles.descriptionSection}>
                    <div className={styles.descriptionHeader}>
                        <div className={styles.descriptionTitle}><span>{t('descriptionLabel')}</span></div>
                        <button onClick={() => setExpanded(!expanded)} className={styles.expandButton}
                                aria-expanded={expanded}
                                aria-label={expanded ? t('showLessAriaLabel') : t('showMoreAriaLabel')}>
                            {expanded ? <ChevronUp size={14} strokeWidth={2.5} aria-hidden="true"/> :
                                <ChevronDown size={14} strokeWidth={2.5} aria-hidden="true"/>}
                        </button>
                    </div>
                    <div
                        className={`${styles.descriptionText} ${expanded ? '' : styles.collapsed}`}>{property.description}</div>
 
                    {expanded && (
                        <>
                            {property.amenities && property.amenities.length > 0 && (
                                <Section title={t('amenitiesTitle')}>
                                    <div className={styles.amenitiesGrid}>
                                        {property.amenities.map((amenity, index) => <AmenityItem key={index}
                                                                                                 amenity={amenity}/>)}
                                    </div>
                                </Section>
                            )}
                            {property.priceType === "sale" && property.mortgage?.available && (
                                <Section title={t('mortgageTitle')}>
                                    <div className={styles.mortgageGrid}>
                                        <MortgageRow label={t('mortgageLabelMonthly')}
                                                     value={property.mortgage.monthlyPayment}/>
                                        <MortgageRow label={t('mortgageLabelInterest')}
                                                     value={property.mortgage.interestRate}/>
                                        <MortgageRow label={t('mortgageLabelTerm')} value={property.mortgage.term}/>
                                        <div className={styles.mortgageDetails}><span>{t('mortgageLinkText')}</span>
                                        </div>
                                    </div>
                                </Section>
                            )}
                        </>
                    )}
                </div>
 
                {/* Agent Info */}
                <div className={styles.agentSection}>
                    <div className={styles.agentInfo}>
                        <div className={styles.agentAvatar}>
                            <div
                                className={styles.avatarCircle}>{property.agent?.name?.charAt(0).toUpperCase() || '?'}</div>
                            {property.agent?.verified &&
                                <div className={styles.verifiedBadge} title={t('agentVerifiedTooltip')}><Check size={8}
                                                                                                               className={styles.iconCheck}/>
                                </div>}
                        </div>
                        <div className={styles.agentDetails}>
                            <div className={styles.agentName}>
                                {property.agent?.name || t('defaultAgentName')}
                                {property.agent?.verified && <span className={styles.agentBadge}><ShieldCheck size={10}
                                                                                                              className={styles.iconShield}/>{t('agentBadgeText')}</span>}
                            </div>
                            <div className={styles.agentCompanyInfo}>
                                <span
                                    className={styles.companyName}>{property.agent?.company || t('defaultAgentCompany')}</span>
                                <span className={styles.separator}>•</span>
                                <span>{property.agent?.experience ? t('experienceSuffix', {years: property.agent.experience}) : ''}</span>
                            </div>
                        </div>
                    </div>
                    <button className={styles.contactButton} onClick={() => setShowContact(!showContact)}>
                        <MessageCircle size={12} className={styles.iconMessageCircle} strokeWidth={2}
                                       aria-hidden="true"/> {t('contactButtonText')}
                    </button>
                </div>
 
                {/* Contact Form */}
                {showContact && (
                    <ContactFormProperty
                        agentName={property.agent?.name || t('defaultAgentName')}
                        propertyAddress={property.address}
                        propertyType={property.propertyType}
                        bedrooms={property.bedrooms}
                        t={t} // Pass translation function
                    />
                )}
 
                {/* Bottom Action Buttons */}
                <div className={styles.actionButtons}>
                    <ActionButton icon={Bookmark} active={saved} onClick={handleSaveToggle}
                                  text={saved ? t('savedButtonText') : t('saveButtonText')} t={t}/>
                    <ActionButton icon={Share2} text={t('shareButtonText')} t={t}/>
                    <ActionButton icon={Grid}
                                  text={t(property.floorplanAvailable ? 'floorplanButtonText' : 'scheduleVisitButtonText')}
                                  isSpecial={true} t={t}/>
                </div>
            </div>
        </div>
    );
};

// Add t prop or pass translated text
const Badge = ({icon: Icon, text, className = ''}) => (
    <div className={`${styles.badge} ${className}`}>
        {Icon && <Icon size={12} className={styles.iconMargin} strokeWidth={2.5} aria-hidden="true"/>}
        <span>{text}</span>
    </div>
);
 
const InteractionButton = ({icon: Icon, active = false, onClick, ariaLabel, count, t}) => (
    <button className={styles.interactionButton} onClick={onClick} aria-label={ariaLabel} aria-pressed={active}>
        <Icon size={16} className={active ? `${styles.iconBase} ${styles.active}` : styles.iconBase} strokeWidth={2}
              aria-hidden="true"/>
        {/* Format count if needed */}
        {count !== undefined && <span>{count}</span>}
    </button>
);
 
const SpecItem = ({icon: Icon, value, label = ''}) => (
    <div className={styles.specItem}>
        {Icon && <Icon size={14} className={styles.iconMargin} strokeWidth={2} aria-hidden="true"/>}
        <span>{value}{label && ` ${label}`}</span> {/* Conditionally add label */}
    </div>
);
 
const MetaItem = ({icon: Icon, value}) => (
    <div className={styles.metaBadge}>
        {Icon && <Icon size={12} className={styles.iconMargin} strokeWidth={2} aria-hidden="true"/>}
        <span>{value}</span>
    </div>
);
 
// Pass title prop instead of using t directly if preferred
const Section = ({title, children}) => (
    <div className={styles.amenitiesSection}> {/* Reused class, maybe rename */}
        <div className={styles.amenitiesTitle}>{title}</div>
        {/* Reused class */}
        {children}
    </div>
);
 
const AmenityItem = ({amenity}) => (
    <div className={styles.amenityItem}>
        <Sparkles size={12} className={styles.iconSparkles} strokeWidth={2} aria-hidden="true"/>
        <span>{amenity}</span>
    </div>
);
 
const MortgageRow = ({label, value}) => (
    <div className={styles.mortgageRow}>
        <span className={styles.mortgageLabel}>{label}</span>
        <span className={styles.mortgageValue}>{value}</span>
    </div>
);
 
// Contact Form using passed 't' function
const ContactFormProperty = ({agentName, propertyAddress, propertyType, bedrooms, t}) => {
    const defaultMessage = t('formDefaultMessage', {
        agentName,
        bedrooms,
        propertyType: propertyType?.toLowerCase() || t('fallbackPropertyType'),
        address: propertyAddress
    });
    return (
        <div className={styles.contactForm}>
            <div className={styles.contactFormTitle}>{t('contactFormTitle', {agentName})}</div>
            <div className={styles.contactFormFields}>
                <input type="text" placeholder={t('formPlaceholderName')} className={styles.formInput}
                       aria-label={t('formPlaceholderName')}/>
                <input type="email" placeholder={t('formPlaceholderEmail')} className={styles.formInput}
                       aria-label={t('formPlaceholderEmail')}/>
                <textarea placeholder={t('formPlaceholderMessage')} rows={2} className={styles.formInput}
                          defaultValue={defaultMessage} aria-label={t('formPlaceholderMessage')}></textarea>
                <div className={styles.formButtonContainer}>
                    <button className={styles.sendButton}>{t('formSubmitMessage')}</button>
                </div>
            </div>
        </div>
