"use client";

import React, { useState, useCallback, useMemo } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { useTranslations } from 'next-intl';
import { 
  Image, 
  Video, 
  FileText, 
  Download, 
  Trash2, 
  Eye, 
  Filter,
  Upload,
  Search,
  Grid,
  List
} from 'lucide-react';
import { toast } from 'react-toastify';
import styles from './MediaManagement.module.css';

const MediaManagement = () => {
  // Handle missing translations gracefully
  let t;
  try {
    t = useTranslations('AdminMedia');
  } catch (e) {
    // Fallback function for missing translations
    t = (key, options) => options?.defaultValue || key;
  }
  
  const queryClient = useQueryClient();
  const [viewMode, setViewMode] = useState('grid');
  const [selectedType, setSelectedType] = useState('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedItems, setSelectedItems] = useState([]);

  // Mock data for now - replace with API calls
  const { data: mediaData, isLoading } = useQuery({
    queryKey: ['admin-media', selectedType, searchQuery],
    queryFn: async () => {
      // TODO: Replace with actual API call
      return {
        items: [
          {
            id: '1',
            type: 'image',
            name: 'product-image-1.jpg',
            url: '/placeholder.jpg',
            size: '2.5 MB',
            uploadedAt: new Date().toISOString(),
            uploadedBy: 'user123',
            dimensions: '1920x1080'
          },
          {
            id: '2',
            type: 'video',
            name: 'demo-video.mp4',
            url: '/placeholder.mp4',
            size: '15.3 MB',
            uploadedAt: new Date().toISOString(),
            uploadedBy: 'user456',
            duration: '2:35'
          }
        ],
        total: 2
      };
    }
  });

  const deleteMutation = useMutation({
    mutationFn: async (mediaId) => {
      // TODO: Replace with actual API call
      await new Promise(resolve => setTimeout(resolve, 1000));
      return { success: true };
    },
    onSuccess: () => {
      queryClient.invalidateQueries(['admin-media']);
      toast.success(t('deleteSuccess'));
      setSelectedItems([]);
    },
    onError: () => {
      toast.error(t('deleteError'));
    }
  });

  const handleDelete = useCallback((mediaId) => {
    if (window.confirm(t('confirmDelete'))) {
      deleteMutation.mutate(mediaId);
    }
  }, [deleteMutation, t]);

  const handleBulkDelete = useCallback(() => {
    if (selectedItems.length === 0) return;
    if (window.confirm(t('confirmBulkDelete', { count: selectedItems.length }))) {
      selectedItems.forEach(id => deleteMutation.mutate(id));
    }
  }, [selectedItems, deleteMutation, t]);

  const getMediaIcon = (type) => {
    switch (type) {
      case 'image':
        return <Image className={styles.typeIcon} />;
      case 'video':
        return <Video className={styles.typeIcon} />;
      default:
        return <FileText className={styles.typeIcon} />;
    }
  };

  const filteredItems = useMemo(() => {
    if (!mediaData?.items) return [];
    return mediaData.items.filter(item => {
      const matchesType = selectedType === 'all' || item.type === selectedType;
      const matchesSearch = item.name.toLowerCase().includes(searchQuery.toLowerCase());
      return matchesType && matchesSearch;
    });
  }, [mediaData, selectedType, searchQuery]);

  return (
    <div className={styles.container}>
      <div className={styles.header}>
        <div>
          <h1 className={styles.title}>{t('title')}</h1>
          <p className={styles.subtitle}>{t('subtitle')}</p>
        </div>
        <button className={styles.uploadButton}>
          <Upload size={20} />
          {t('uploadNew')}
        </button>
      </div>

      <div className={styles.controls}>
        <div className={styles.searchBar}>
          <Search size={20} />
          <input
            type="text"
            placeholder={t('searchPlaceholder')}
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className={styles.searchInput}
          />
        </div>

        <div className={styles.filters}>
          <select 
            value={selectedType} 
            onChange={(e) => setSelectedType(e.target.value)}
            className={styles.filterSelect}
          >
            <option value="all">{t('filterAll')}</option>
            <option value="image">{t('filterImages')}</option>
            <option value="video">{t('filterVideos')}</option>
            <option value="document">{t('filterDocuments')}</option>
          </select>

          <div className={styles.viewToggle}>
            <button
              className={viewMode === 'grid' ? styles.active : ''}
              onClick={() => setViewMode('grid')}
            >
              <Grid size={20} />
            </button>
            <button
              className={viewMode === 'list' ? styles.active : ''}
              onClick={() => setViewMode('list')}
            >
              <List size={20} />
            </button>
          </div>
        </div>
      </div>

      {selectedItems.length > 0 && (
        <div className={styles.bulkActions}>
          <span>{t('selected', { count: selectedItems.length })}</span>
          <button 
            className={styles.bulkDeleteButton}
            onClick={handleBulkDelete}
          >
            <Trash2 size={16} />
            {t('deleteSelected')}
          </button>
        </div>
      )}

      {isLoading ? (
        <div className={styles.loading}>{t('loading')}</div>
      ) : (
        <div className={viewMode === 'grid' ? styles.gridView : styles.listView}>
          {filteredItems.map(item => (
            <div key={item.id} className={styles.mediaItem}>
              <input
                type="checkbox"
                checked={selectedItems.includes(item.id)}
                onChange={(e) => {
                  if (e.target.checked) {
                    setSelectedItems([...selectedItems, item.id]);
                  } else {
                    setSelectedItems(selectedItems.filter(id => id !== item.id));
                  }
                }}
                className={styles.checkbox}
              />
              
              <div className={styles.preview}>
                {item.type === 'image' ? (
                  <img src={item.url} alt={item.name} />
                ) : (
                  getMediaIcon(item.type)
                )}
              </div>

              <div className={styles.info}>
                <h3>{item.name}</h3>
                <p>{item.size} • {new Date(item.uploadedAt).toLocaleDateString()}</p>
                {item.dimensions && <p>{item.dimensions}</p>}
                {item.duration && <p>{item.duration}</p>}
              </div>

              <div className={styles.actions}>
                <button className={styles.actionButton} title={t('view')}>
                  <Eye size={16} />
                </button>
                <button className={styles.actionButton} title={t('download')}>
                  <Download size={16} />
                </button>
                <button 
                  className={styles.actionButton} 
                  onClick={() => handleDelete(item.id)}
                  title={t('delete')}
                >
                  <Trash2 size={16} />
                </button>
              </div>
            </div>
          ))}
        </div>
      )}

      {filteredItems.length === 0 && !isLoading && (
        <div className={styles.empty}>
          <p>{t('noMedia')}</p>
        </div>
      )}
    </div>
  );
};

export default MediaManagement;
