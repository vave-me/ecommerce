// src/features/CreatePropertyModal/components/steps/BasicInfoStep/BasicInfoStep.jsx
import React, { useState, useMemo, useCallback, memo } from 'react';
import PropTypes from 'prop-types';
import styles from '../../../PropertyModal.module.css';
import {FormActions} from "../../../../../common/components/FormActions";
import { NameDescriptionInputs } from "../../../../shared/components/NameDescriptionInputs";
import { CategorySelector } from "../../../../shared/components/CategorySelector/CategorySelector";
import { LocationInput } from "../../../../shared/components/LocationInput";
import { UrlReferenceInput } from "../../../../shared/components/UrlReferenceInput/UrlReferenceInput";
import { OfferTypeSelect } from "../../../../shared/components/OfferTypeSelect/OfferTypeSelect";
import { MerchantNameInput } from "../../../../shared/components/MerchantNameInput/MerchantNameInput";

export const BasicInfoStep = memo(function BasicInfoStep({
    initialData,
    onComplete,
    onBack,
    isLoading,
    errors,
    isUserLoggedIn,
    categories
}) {
    // Unified form data state
    const [formData, setFormData] = useState({
        name: initialData?.name || '',
        description: initialData?.description || '',
        price: initialData?.price || '',
        categoryId: initialData?.categoryId || '',
        categorySlug: initialData?.categorySlug || '',
        categoryName: initialData?.categoryName || '',
        propertyType: initialData?.propertyType || 'house',
        transactionType: initialData?.transactionType || 'sale',
        negotiable: initialData?.negotiable || false,
        userType: initialData?.userType || 'private',
        tags: Array.isArray(initialData?.tags) ? initialData.tags : [],
        bedrooms: initialData?.bedrooms || '',
        bathrooms: initialData?.bathrooms || '',
        squareMeters: initialData?.squareMeters || '',
        landArea: initialData?.landArea || '',
        yearBuilt: initialData?.yearBuilt || '',
        parkingSpots: initialData?.parkingSpots || '',
        // Location data
        address: initialData?.address || '',
        lat: initialData?.lat || null,
        lng: initialData?.lng || null,
        // New fields
        urlReference: initialData?.urlReference || '',
        offerType: initialData?.offerType || '',
        merchantName: initialData?.merchantName || ''
    });

    // Tag input state
    const [currentTag, setCurrentTag] = useState('');

    // Form validation
    const canProceed = useMemo(() => {
        return formData.name.trim() !== '' &&
            formData.description.trim() !== '' &&
            formData.price.trim() !== '' &&
            formData.categoryId !== '' &&
            formData.propertyType &&
            formData.transactionType;
    }, [formData]);

    // Single handler for form updates
    const updateFormData = useCallback((field, value) => {
        setFormData(prev => ({
            ...prev,
            [field]: value
        }));
    }, []);

    // Category handlers
    const handleCategorySelect = useCallback((category) => {
        if (category) {
            setFormData(prev => ({
                ...prev,
                categoryId: category.id,
                categorySlug: category.slug || '',
                categoryName: category.name
            }));
        }
    }, []);

    const handleCategoryClear = useCallback(() => {
        setFormData(prev => ({
            ...prev,
            categoryId: '',
            categorySlug: '',
            categoryName: ''
        }));
    }, []);

    // Tag management
    const addTag = useCallback(() => {
        const tag = currentTag.trim();
        if (tag && !formData.tags.includes(tag)) {
            updateFormData('tags', [...formData.tags, tag]);
            setCurrentTag('');
        }
    }, [currentTag, formData.tags, updateFormData]);

    const removeTag = useCallback((tagToRemove) => {
        updateFormData('tags', formData.tags.filter(tag => tag !== tagToRemove));
    }, [formData.tags, updateFormData]);

    const handleTagKeyPress = useCallback((e) => {
        if (e.key === 'Enter' || e.key === ',') {
            e.preventDefault();
            addTag();
        }
    }, [addTag]);

    // Toggle handlers
    const handleToggle = useCallback((field) => {
        updateFormData(field, !formData[field]);
    }, [formData, updateFormData]);

    // Form submission
    const handleSubmit = useCallback((e) => {
        e.preventDefault();
        if (!canProceed) return;
        onComplete(formData);
    }, [canProceed, formData, onComplete]);

    return (
        <div className={styles.formContainer}>
            <form className={styles.form} onSubmit={handleSubmit}>
                {/* Property Name & Description */}
                <NameDescriptionInputs
                    nameId="property-name"
                    descriptionId="property-description"
                    nameLabel="Property Name"
                    descriptionLabel="Description"
                    nameValue={formData.name}
                    descriptionValue={formData.description}
                    onNameChange={(value) => updateFormData('name', value)}
                    onDescriptionChange={(value) => updateFormData('description', value)}
                    namePlaceholder="e.g. Modern 3-Bedroom House in Downtown"
                    descriptionPlaceholder="Describe the property features, condition, location highlights..."
                    nameError={errors?.name}
                    descriptionError={errors?.description}
                    styles={styles}
                    nameMaxLength={150}
                    descriptionRows={4}
                />

                {/* New Shared Components Row */}
                <div className={styles.formRow}>
                    <UrlReferenceInput
                        id="property-url-reference"
                        value={formData.urlReference}
                        onChange={(value) => updateFormData('urlReference', value)}
                        label="URL Reference"
                        placeholder="Enter reference URL"
                        error={errors?.urlReference}
                        styles={styles}
                    />
                    
                    <OfferTypeSelect
                        id="property-offer-type"
                        value={formData.offerType}
                        onChange={(value) => updateFormData('offerType', value)}
                        label="Type of Offer"
                        error={errors?.offerType}
                        styles={styles}
                    />
                </div>

                <MerchantNameInput
                    id="property-merchant-name"
                    value={formData.merchantName}
                    onChange={(value) => updateFormData('merchantName', value)}
                    label="Merchant Name"
                    placeholder="Enter merchant or business name"
                    error={errors?.merchantName}
                    styles={styles}
                />

                {/* Price Fields Row */}
                <div className={styles.formRow}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="property-price">
                            Property Price <span className={styles.requiredMark}>*</span>
                        </label>
                        <div className={styles.inputWithUnit}>
                            <input
                                id="property-price"
                                type="number"
                                className={`${styles.formInput} ${errors?.price ? styles.inputError : ''}`}
                                value={formData.price}
                                onChange={(e) => updateFormData('price', e.target.value)}
                                placeholder="Enter price"
                                min="0"
                                step="0.01"
                            />
                            <span className={styles.unitLabel}>€</span>
                        </div>
                        {errors?.price && (
                            <div className={styles.fieldError}>{errors.price}</div>
                        )}
                    </div>

                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="transaction-type">
                            Transaction Type
                        </label>
                        <select
                            id="transaction-type"
                            className={styles.formSelect}
                            value={formData.transactionType}
                            onChange={(e) => updateFormData('transactionType', e.target.value)}
                        >
                            <option value="sale">For Sale</option>
                            <option value="rent">For Rent</option>
                            <option value="short_term_rental">Short-term Rental</option>
                        </select>
                    </div>
                </div>

                {/* Property Details Row */}
                <div className={styles.formRow}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="property-type">
                            Property Type
                        </label>
                        <select
                            id="property-type"
                            className={styles.formSelect}
                            value={formData.propertyType}
                            onChange={(e) => updateFormData('propertyType', e.target.value)}
                        >
                            <option value="house">House</option>
                            <option value="apartment">Apartment</option>
                            <option value="land">Land</option>
                            <option value="commercial">Commercial</option>
                            <option value="other">Other</option>
                        </select>
                    </div>

                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="user-type">
                            Seller Type
                        </label>
                        <select
                            id="user-type"
                            className={styles.formSelect}
                            value={formData.userType}
                            onChange={(e) => updateFormData('userType', e.target.value)}
                        >
                            <option value="private">Private Owner</option>
                            <option value="business">Business</option>
                            <option value="agent">Real Estate Agent</option>
                            <option value="developer">Developer</option>
                        </select>
                    </div>
                </div>

                {/* Property Specifications Row */}
                <div className={styles.formRow}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="bedrooms">
                            Bedrooms
                        </label>
                        <input
                            id="bedrooms"
                            type="number"
                            className={styles.formInput}
                            value={formData.bedrooms}
                            onChange={(e) => updateFormData('bedrooms', e.target.value)}
                            placeholder="Number of bedrooms"
                            min="0"
                        />
                    </div>

                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="bathrooms">
                            Bathrooms
                        </label>
                        <input
                            id="bathrooms"
                            type="number"
                            className={styles.formInput}
                            value={formData.bathrooms}
                            onChange={(e) => updateFormData('bathrooms', e.target.value)}
                            placeholder="Number of bathrooms"
                            min="0"
                            step="0.5"
                        />
                    </div>
                </div>

                {/* Area and Size Row */}
                <div className={styles.formRow}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="square-meters">
                            Square Meters
                        </label>
                        <div className={styles.inputWithUnit}>
                            <input
                                id="square-meters"
                                type="number"
                                className={styles.formInput}
                                value={formData.squareMeters}
                                onChange={(e) => updateFormData('squareMeters', e.target.value)}
                                placeholder="Living area"
                                min="0"
                                step="0.01"
                            />
                            <span className={styles.unitLabel}>m²</span>
                        </div>
                    </div>

                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="land-area">
                            Land Area
                        </label>
                        <div className={styles.inputWithUnit}>
                            <input
                                id="land-area"
                                type="number"
                                className={styles.formInput}
                                value={formData.landArea}
                                onChange={(e) => updateFormData('landArea', e.target.value)}
                                placeholder="Total land area"
                                min="0"
                                step="0.01"
                            />
                            <span className={styles.unitLabel}>m²</span>
                        </div>
                    </div>
                </div>

                {/* Year Built and Parking Row */}
                <div className={styles.formRow}>
                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="year-built">
                            Year Built
                        </label>
                        <input
                            id="year-built"
                            type="number"
                            className={styles.formInput}
                            value={formData.yearBuilt}
                            onChange={(e) => updateFormData('yearBuilt', e.target.value)}
                            placeholder="Year of construction"
                            min="1800"
                            max={new Date().getFullYear()}
                        />
                    </div>

                    <div className={styles.formGroup}>
                        <label className={styles.formLabel} htmlFor="parking-spots">
                            Parking Spots
                        </label>
                        <input
                            id="parking-spots"
                            type="number"
                            className={styles.formInput}
                            value={formData.parkingSpots}
                            onChange={(e) => updateFormData('parkingSpots', e.target.value)}
                            placeholder="Number of parking spots"
                            min="0"
                        />
                    </div>
                </div>

                {/* Category Selection - Full Width */}
                <CategorySelector
                    categories={categories}
                    selectedCategoryId={formData.categoryId}
                    selectedCategoryName={formData.categoryName}
                    onCategorySelect={handleCategorySelect}
                    onCategoryClear={handleCategoryClear}
                    categoryType="property"
                    error={errors?.categoryId}
                    styles={styles}
                    placeholder="Select property category..."
                />

                {/* Tags Input - Full Width */}
                <div className={styles.formGroup}>
                    <label className={styles.formLabel} htmlFor="tags">
                        Tags
                    </label>
                    <div className={styles.tagsContainer}>
                        {formData.tags.map((tag, index) => (
                            <div key={index} className={styles.tag}>
                                {tag}
                                <button
                                    type="button"
                                    className={styles.tagRemove}
                                    onClick={() => removeTag(tag)}
                                    aria-label={`Remove ${tag} tag`}
                                >
                                    ×
                                </button>
                            </div>
                        ))}
                        <input
                            id="tags"
                            type="text"
                            className={styles.formInput}
                            value={currentTag}
                            onChange={(e) => setCurrentTag(e.target.value)}
                            onKeyDown={handleTagKeyPress}
                            onBlur={addTag}
                            placeholder="Add tags (e.g., garden, garage, swimming pool)..."
                        />
                    </div>
                </div>

                {/* Toggles Section */}
                <div className={styles.formGroup}>
                    <div className={styles.toggleGroup}>
                        <span className={styles.formLabel}>Price is negotiable</span>
                        <div 
                            className={`${styles.toggleSwitch} ${formData.negotiable ? styles.checked : ''}`}
                            onClick={() => handleToggle('negotiable')}
                        >
                            <div className={styles.toggleSlider}></div>
                        </div>
                    </div>
                </div>

                {/* Location Input */}
                <LocationInput
                    value={formData.address}
                    latitude={formData.lat}
                    longitude={formData.lng}
                    onLocationChange={(address) => updateFormData('address', address)}
                    onCoordinatesChange={(lat, lng) => {
                        updateFormData('lat', lat);
                        updateFormData('lng', lng);
                    }}
                    placeholder="Enter zipcode or city name"
                    label="Property Location (Optional)"
                    error={errors?.location}
                    showCurrentLocationButton={true}
                />

                {/* Form Actions */}
                <FormActions
                    onCancel={onBack}
                    isSubmitting={isLoading}
                    isDisabled={!canProceed || isLoading}
                    submitLabel={isLoading ? 'Saving...' : 'Continue to Media'}
                />
            </form>
        </div>
    );
});

BasicInfoStep.propTypes = {
    initialData: PropTypes.object,
    onComplete: PropTypes.func.isRequired,
    onBack: PropTypes.func.isRequired,
    isLoading: PropTypes.bool,
    errors: PropTypes.object,
    isUserLoggedIn: PropTypes.bool,
    categories: PropTypes.array
};
