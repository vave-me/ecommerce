"use client";

import React, {useState, useRef, useEffect} from 'react';
import { useTranslations } from 'next-intl'; //  Import hook
import {FaInfoCircle} from 'react-icons/fa';
import styles from './SearchHint.module.css'; // Assuming CSS Module name is SearchHint.module.css

// Use a map for stable keys associated with symbols
const QUICK_FILTER_MAP = {
    '%': 'deals',
    '#': 'posts',
    '@': 'users',
    '$': 'products',
    '*': 'videos',
    '&': 'cars',
    '!': 'jobs',
    '^': 'estate'
};

const QuickFilterTooltip = () => {
    const t = useTranslations('QuickFilterTooltip'); //  Instantiate hook
    const [isVisible, setIsVisible] = useState(false);
    const [position, setPosition] = useState({top: 0, left: 0});
    const triggerRef = useRef(null);
    const tooltipRef = useRef(null);

    // --- Tooltip positioning and visibility logic (remains the same) ---
    const handleMouseEnter = () => { // Removed event arg as it wasn't used
        if (triggerRef.current) {
            const rect = triggerRef.current.getBoundingClientRect();
            const scrollY = window.scrollY || window.pageYOffset;
            const scrollX = window.scrollX || window.pageXOffset;
            setPosition({
                // Position below the icon, accounting for scroll
                top: rect.bottom + scrollY + 5, // Added small gap
                left: rect.left + scrollX
            });
            setIsVisible(true);
        }
    };

    const handleMouseLeave = (e) => {
        // Delay allows moving cursor into the tooltip
        const leaveTimeout = setTimeout(() => {
            if (
                tooltipRef.current &&
                triggerRef.current &&
                !tooltipRef.current.contains(document.elementFromPoint(e.clientX, e.clientY)) &&
                !triggerRef.current.contains(document.elementFromPoint(e.clientX, e.clientY))
            ) {
                setIsVisible(false);
            }
        }, 150); // Adjusted delay

        // Store timeout to clear if needed (e.g., on unmount)
        // tooltipRef.current.__leaveTimeout = leaveTimeout; // Example using ref property
    };

    // Clear timeout on unmount if stored
    // useEffect(() => {
    //    const ref = tooltipRef.current;
    //    return () => {
    //        if (ref && ref.__leaveTimeout) {
    //            clearTimeout(ref.__leaveTimeout);
    //        }
    //    };
    // }, []);


    const handleTooltipMouseEnter = () => {
        // Optional: Clear any pending leave timeout if mouse enters tooltip
        // if (tooltipRef.current && tooltipRef.current.__leaveTimeout) {
        //     clearTimeout(tooltipRef.current.__leaveTimeout);
        // }
    };


    const handleTooltipMouseLeave = (e) => {
        // Check if the new target isn't the trigger icon
        if (!triggerRef.current?.contains(e.relatedTarget)) {
            setIsVisible(false);
        }
    };
    // --- End Tooltip logic ---


    return (
        <div className={styles.quickFilterInfoContainer}>
            <FaInfoCircle
                ref={triggerRef}
                className={styles.quickFilterIcon}
                onMouseEnter={handleMouseEnter}
                onMouseLeave={handleMouseLeave}
                onFocus={handleMouseEnter} // Show on focus for accessibility
                onBlur={handleMouseLeave}  // Hide on blur
                tabIndex={0} // Make it focusable
                //   Use translation
                aria-label={t('infoIconAriaLabel')}
            />

            {isVisible && (
                <div
                    ref={tooltipRef}
                    className={styles.quickFilterTooltip}
                    style={{
                        // Use CSS variables or ensure correct positioning library if fixed doesn't work
                        position: 'fixed', // Use fixed positioning if tooltip should ignore parent scroll
                        top: `${position.top}px`,
                        left: `${position.left}px`,
                        // Add transform for potential centering adjustment if needed
                        // transform: 'translateX(-50%)', // Example centering adjustment
                    }}
                    onMouseEnter={handleTooltipMouseEnter} // Keep tooltip open when mouse enters it
                    onMouseLeave={handleTooltipMouseLeave}
                    role="tooltip" // Added role
                >
                    <div className={styles.quickFilterTooltipContent}>
                        {/*   Use translation */}
                        <h4>{t('tooltipTitle')}</h4>
                        <ul aria-label={t('tooltipTitle')}> {/* Added aria-label to list */}
                            {/* Iterate over the map using stable keys */}
                            {Object.entries(QUICK_FILTER_MAP).map(([symbol, key]) => (
                                <li key={symbol}>
                                    <span
                                        className={styles.symbolBadge}
                                        style={{ backgroundColor: getSymbolColor(symbol) }}
                                        aria-hidden="true" // Symbol is decorative, text explains it
                                    >
                                        {symbol}
                                    </span>
                                    {/*   Use translation with derived key */}
                                    {t(`quickFilter_${key}`)}
                                </li>
                            ))}
                        </ul>
                        {/*   Use translation */}
                        <p className={styles.tooltipTip}>
                            {t('tooltipTip')}
                        </p>
                    </div>
                </div>
            )}
        </div>
    );
};

// Helper function to get consistent colors (remains the same)
function getSymbolColor(symbol) {
    const colorMap = {
        '%': '#FFA500',  // Orange
        '#': '#4CAF50',  // Green
        '@': '#2196F3',  // Blue
        '$': '#9C27B0',  // Purple
        '*': '#FF5722',  // Deep Orange
        '&': '#795548',  // Brown
        '!': '#607D8B',  // Blue Grey
        '^': '#3F51B5'   // Indigo
    };
    return colorMap[symbol] || '#2ec4b6'; // Default color
}

export default QuickFilterTooltip;
