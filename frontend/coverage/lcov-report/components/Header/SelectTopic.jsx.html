"use client";
 
import React, { useState, useRef, useEffect, useContext } from "react";
import { ChevronDown } from "lucide-react";
import { useRouter } from "../../i18n/navigation";
import { NavBarContext } from "../../context/NavBarContext";
import { useTranslations } from "next-intl";
import styles from "./SelectTopic.module.css";
 
import PropTypes from 'prop-types';
import { useLocale } from 'next-intl';
import { useMediaQuery } from "@/components/Header/hooks/useMediaQuery.js";
 
// Import extracted components
 
import TopicButtonsRow from './TopicButtonsRow';
import TopicDropdown from './TopicDropdown';
 
// Import API functions
import { fetchMainCategories, fetchSubCategories } from '../../api/categories';
 
// TOPIC_CONFIG remains unchanged
const TOPIC_CONFIG = [
    {value: "dashboard", labelKey: "topic_dashboard_label", routeBase: "/dashboard", badgeKey: "dashboard_badge"},
    {
        value: "marketplace",
        labelKey: "topic_marketplace_label",
        routeBase: "/marketplace",
        badgeKey: "marketplace_badge"
    },
    {value: "deals", labelKey: "topic_deals_label", routeBase: "/deals"},
    {value: "jobs", labelKey: "topic_jobs_label", routeBase: "/jobs"},
    {value: "news_info", labelKey: "topic_news_info_label", routeBase: "/news"},
    {value: "social_content", labelKey: "topic_social_content_label", routeBase: "/social"},
    {value: "property", labelKey: "topic_property_label", routeBase: "/properties"},
    {value: "services", labelKey: "topic_services_label", routeBase: "/services"},
    {value: "videos", labelKey: "topic_videos_label", routeBase: "/videos"},
    {value: "automotive", labelKey: "topic_automotive_label", routeBase: "/cars", badgeKey: "cars_badge"},
];
 
const MOBILE_BREAKPOINT = 768;
const API_PAGE_SIZE = 50;
 
export default function SelectTopic({
    defaultTopic = "dashboard",
    onChange,
    // Auto-scroll configuration options
    enableAutoScroll = true,
    autoScrollSpeed = 0.5,
    autoScrollPauseDuration = 2000,
    autoScrollDisableOnInteraction = true,
    enablePerformanceMonitoring = false,
    performanceThreshold = 50,
}) {
    const t = useTranslations('Topics');
    const locale = useLocale();
    const router = useRouter();
 
    // Main state
    const [activeTopicValue, setActiveTopicValue] = useState(defaultTopic);
    const [selectedCategoryValue, setSelectedCategoryValue] = useState("all");
    const [openTopicValue, setOpenTopicValue] = useState(null);
    const [activeAnchorEl, setActiveAnchorEl] = useState(null);
    const [fetchedCategories, setFetchedCategories] = useState({});
 
    // New state for subcategories
    const [selectedParentId, setSelectedParentId] = useState(null);
    const [selectedParentSlug, setSelectedParentSlug] = useState(null);
    const [subcategories, setSubcategories] = useState([]);
    const [isLoadingSubcategories, setIsLoadingSubcategories] = useState(false);
 
    const isMobileView = useMediaQuery(`(max-width: ${MOBILE_BREAKPOINT}px)`);
    const scrollContainerRef = useRef(null);
    const topicButtonRefs = useRef({});
 
    const translatedTopics = React.useMemo(() => {
        return TOPIC_CONFIG.map(topic => ({
            ...topic,
            label: t(topic.labelKey),
            badge: topic.badgeKey ? t(topic.badgeKey) : undefined,
        }));
    }, [t]);
 
    // Effect to fetch main categories
    useEffect(() => {
        if (!openTopicValue) return;

        const cacheEntry = fetchedCategories[openTopicValue];
        const shouldFetch = !cacheEntry || (!cacheEntry.isLoading && cacheEntry.error);
 
        if (shouldFetch) {
            const fetchAndSetCategories = async () => {
                setFetchedCategories(prev => ({...prev, [openTopicValue]: {data: null, isLoading: true, error: null}}));

                try {
                    const apiFilters = {categoryType: openTopicValue, lang: locale, page: 0, pageSize: API_PAGE_SIZE};
                    const result = await fetchMainCategories(apiFilters);

                    const mappedData = (result?.categories || []).map(cat => {
                        return {
                            value: cat.slug || '',
                            label: cat.name || cat.description || 'Unknown Category',
                            id: cat.id,
                            featured: cat.featured,
                            count: cat.count,
                            // Force true for main categories to try loading subcategories
                            hasSubcategories: true,
                            subcategories: cat.subcategories || []
                        };
                    });

                    setFetchedCategories(prev => ({
                        ...prev,
                        [openTopicValue]: {data: mappedData, isLoading: false, error: null}
                    }));
                } catch (err) {
                    console.error(`Error fetching/setting categories for ${openTopicValue}:`, err);
                    setFetchedCategories(prev => ({
                        ...prev,
                        [openTopicValue]: {data: null, isLoading: false, error: err}
                    }));
                }
            };

            fetchAndSetCategories();
        }
    }, [openTopicValue, locale, fetchedCategories]);
 
    // Main handlers
    const handleCloseDropdown = React.useCallback(() => {
        setOpenTopicValue(null);
        setActiveAnchorEl(null);
        setSelectedParentId(null);
        setSelectedParentSlug(null);
        setSubcategories([]);
    }, []);
 
    const handleTopicToggle = React.useCallback((topicValue, targetElement) => {
        if (openTopicValue === topicValue) {
            handleCloseDropdown();
        } else {
            setActiveTopicValue(topicValue);
            setOpenTopicValue(topicValue);
            setActiveAnchorEl(targetElement);
            setSelectedCategoryValue("all");
            setSelectedParentId(null);
            setSelectedParentSlug(null);
            setSubcategories([]);
        }
    }, [openTopicValue, handleCloseDropdown]);
 
    // Simplified navigation - always uses the direct category path
    const handleCategorySelect = React.useCallback((categoryValueSlug) => {
        const currentTopicConfig = TOPIC_CONFIG.find(topic => topic.value === activeTopicValue);
        if (!currentTopicConfig) {
            console.error("Nav Error: Config not found", activeTopicValue);
            handleCloseDropdown();
            return;
        }

        // Set selected category
        setSelectedCategoryValue(categoryValueSlug);

        // Close dropdown
        handleCloseDropdown();

        // Notify parent component
        onChange?.(activeTopicValue, categoryValueSlug);

        // Build path - always simply the base path + category
        const basePath = currentTopicConfig.routeBase; // e.g., "/deals"

        // Don't include the category in the path for "all" - just navigate to base path
        if (categoryValueSlug === "all") {
            try {
                router.push(basePath);
            } catch (error) {
                console.error("Nav Failed:", error);
            }
            return;
        }

        // Simple path: just topic + category, regardless of parent/child relationship
        const newPath = `${basePath}/${categoryValueSlug}`;

        try {
            router.push(newPath);
        } catch (error) {
            console.error("Nav Failed:", error);
        }
    }, [activeTopicValue, onChange, handleCloseDropdown, router]);
 
    // New effect to fetch subcategories
    useEffect(() => {
        if (!selectedParentId || !openTopicValue) return;

        const fetchSubcategoriesData = async () => {
            setIsLoadingSubcategories(true);

            try {
                const filters = {lang: locale};
                const subcatsData = await fetchSubCategories(selectedParentId, filters);

                // If no subcategories were found, navigate to the category page directly
                if (!subcatsData || subcatsData.length === 0) {
                    if (selectedParentSlug) {
                        handleCategorySelect(selectedParentSlug);
                    }
                    return;
                }

                const mappedSubcats = subcatsData.map(cat => ({
                    value: cat.slug || '',
                    label: cat.name || cat.description || 'Unknown Category',
                    id: cat.id,
                    featured: cat.featured,
                    count: cat.count,
                    hasSubcategories: false, // Subcategories don't have further sub-subcategories
                    parentId: selectedParentId,
                    subcategories: cat.subcategories || []
                }));

                setSubcategories(mappedSubcats);
            } catch (err) {
                console.error(`Error fetching subcategories for parent ${selectedParentId}:`, err);
                // If there was an error, navigate to the category page as fallback
                if (selectedParentSlug) {
                    handleCategorySelect(selectedParentSlug);
                }
                setSubcategories([]);
            } finally {
                setIsLoadingSubcategories(false);
            }
        };

        fetchSubcategoriesData();
    }, [selectedParentId, openTopicValue, locale, selectedParentSlug, handleCategorySelect]);
 
    // New handler for showing subcategories
    const handleShowSubcategories = React.useCallback((parentId, parentSlug) => {
        if (!parentId) {
            // Back to main categories
            setSelectedParentId(null);
            setSelectedParentSlug(null);
            setSubcategories([]);
            return;
        }

        // Set parent ID to trigger subcategory fetch
        setSelectedParentId(parentId);
        setSelectedParentSlug(parentSlug);
    }, []);
 
    // Auto-scroll effect for mobile
    useEffect(() => {
        if (!isMobileView || !scrollContainerRef.current) return;
        const activeButton = topicButtonRefs.current[activeTopicValue];
        const scrollContainer = scrollContainerRef.current;
        if (activeButton && scrollContainer) {
            const containerRect = scrollContainer.getBoundingClientRect();
            const buttonRect = activeButton.getBoundingClientRect();
            if (buttonRect.left < containerRect.left || buttonRect.right > containerRect.right) {
                const scrollOffset = buttonRect.left - containerRect.left - (containerRect.width / 2) + (buttonRect.width / 2);
                scrollContainer.scrollBy({left: scrollOffset, behavior: 'smooth'});
            }
        }
    }, [activeTopicValue, isMobileView]);
 
    // Prepare data for the open dropdown
    const openTopicData = React.useMemo(() => {
        if (!openTopicValue) return null;
        const baseTopic = translatedTopics.find(topic => topic.value === openTopicValue);
        if (!baseTopic) return null;

        return {
            ...baseTopic,
            categories: fetchedCategories[openTopicValue] || {data: null, isLoading: false, error: null}
        };
    }, [openTopicValue, translatedTopics, fetchedCategories]);
 
    // Render
    return (
        <div className={styles.container}>
            {/* Topic Buttons Row */}
            <TopicButtonsRow
                ref={scrollContainerRef}
                translatedTopics={translatedTopics}
                activeTopicValue={activeTopicValue}
                openTopicValue={openTopicValue}
                topicButtonRefs={topicButtonRefs}
                onTopicToggle={handleTopicToggle}
                enableAutoScroll={enableAutoScroll}
                autoScrollSpeed={autoScrollSpeed}
                autoScrollPauseDuration={autoScrollPauseDuration}
                autoScrollDisableOnInteraction={autoScrollDisableOnInteraction}
                enablePerformanceMonitoring={enablePerformanceMonitoring}
                performanceThreshold={performanceThreshold}
            />
 
            {/* Render Dropdown with Subcategories Support */}
            {openTopicData && (
                <TopicDropdown
                    key={openTopicData.value}
                    topicData={openTopicData}
                    isOpen={!!openTopicValue}
                    onClose={handleCloseDropdown}
                    onSelectCategory={handleCategorySelect}
                    selectedCategoryValue={selectedCategoryValue}
                    anchorEl={activeAnchorEl}
                    isMobileView={isMobileView}
                    id={`topic-dropdown-${openTopicData.value}`}
                    activeSubcategories={subcategories}
                    isLoadingSubcategories={isLoadingSubcategories}
                    selectedParentId={selectedParentId}
                    onShowSubcategories={handleShowSubcategories}
                />
            )}
        </div>
    );
}
 
// PropTypes
SelectTopic.propTypes = {
    defaultTopic: PropTypes.string,
    onChange: PropTypes.func,
    // Auto-scroll configuration PropTypes
    enableAutoScroll: PropTypes.bool,
    autoScrollSpeed: PropTypes.number,
    autoScrollPauseDuration: PropTypes.number,
    autoScrollDisableOnInteraction: PropTypes.bool,
    enablePerformanceMonitoring: PropTypes.bool,
    performanceThreshold: PropTypes.number,
};
