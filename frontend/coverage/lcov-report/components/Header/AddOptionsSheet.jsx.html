"use client";
 
import React, { useEffect, useRef, useState } from 'react';
import PropTypes from 'prop-types';
import { useTranslations } from 'next-intl'; //  Import hook
 
import styles from './AddOptionsSheet.module.css';
 
/**
 * Enhanced Bottom sheet component for displaying Add content options.
 * Assumes 'items' prop contains pre-translated labels and descriptions.
 * Uses next-intl for its own UI text (title, cancel button, ARIA labels).
 */
const AddOptionsSheet = ({
                             isOpen,
                             onClose,
                             onSelectItem,
                             items // Expects items like: [{ id, label, description, icon }, ...] where label/desc are pre-translated
                         }) => {
    const t = useTranslations('AddOptionsSheet'); //  Instantiate hook
    const [isClosing, setIsClosing] = useState(false);
    const sheetRef = useRef(null);
    const [activeItemId, setActiveItemId] = useState(null); // Used for hover/focus indication

    // Handle escape key press & body scroll lock
    useEffect(() => {
        const handleEscKey = (e) => {
            if (e.key === 'Escape' && isOpen) {
                handleClose();
            }
        };

        document.addEventListener('keydown', handleEscKey);

        if (isOpen) {
            document.body.style.overflow = 'hidden';
        } else {
            // Ensure scroll is restored if component might unmount before closing animation finishes
            document.body.style.overflow = '';
        }

        return () => {
            document.removeEventListener('keydown', handleEscKey);
            // Restore scroll on cleanup
            document.body.style.overflow = '';
        };
    }, [isOpen]); // Dependency on isOpen

    // Reset active item when sheet closes or items change
    useEffect(() => {
        if (!isOpen) {
            setActiveItemId(null);
        }
    }, [isOpen]);

    // Animation handling for closing
    const handleClose = () => {
        // Prevent double close triggers
        if (isClosing) return;

        setIsClosing(true);
        setTimeout(() => {
            setIsClosing(false);
            onClose(); // Call parent onClose after animation
        }, 200); // Match animation duration (adjust if needed)
    };

    // Select item and close
    const handleSelectItem = (itemId) => {
        onSelectItem(itemId);
        handleClose(); // Close sheet after selection
    };

    // If not open and not in the process of closing, render nothing
    if (!isOpen && !isClosing) return null;

    return (
        <div
            className={`${styles.backdrop} ${isOpen && !isClosing ? styles.backdropOpen : ''} ${isClosing ? styles.backdropClosing : ''}`}
            onClick={handleClose} // Close when clicking backdrop
            aria-modal="true"
            role="dialog"
            //   Use translation
            aria-label={t('ariaLabel')}
        >
            <div
                ref={sheetRef}
                className={`${styles.sheetContainer} ${isOpen && !isClosing ? styles.sheetOpen : ''} ${isClosing ? styles.sheetClosing : ''}`}
                onClick={(e) => e.stopPropagation()} // Prevent backdrop click when clicking sheet content
                role="document" // Role for the sheet content itself
            >
                {/* Optional visual indicator for pull-to-close */}
                <div className={styles.pullIndicator} />

                <div className={styles.header}>
                    {/*   Use translation */}
                    <h3 className={styles.title}>{t('title')}</h3>
                </div>

                <ul className={styles.optionsList}>
                    {items.map((item) => (
                        <li key={item.id} className={styles.optionItem}>
                            <button
                                className={`${styles.optionButton} ${activeItemId === item.id ? styles.optionButtonActive : ''}`}
                                onClick={() => handleSelectItem(item.id)}
                                onMouseEnter={() => setActiveItemId(item.id)}
                                onMouseLeave={() => setActiveItemId(null)} // Clear active state on mouse leave
                                onFocus={() => setActiveItemId(item.id)}
                                onBlur={() => setActiveItemId(null)} // Clear active state on blur
                                //   Use translation with interpolation (label comes from pre-translated prop)
                                aria-label={t('createAriaLabel', { label: item.label })}
                            >
                                <div className={styles.iconWrapper} aria-hidden="true">
                                    <item.icon size={22} />
                                </div>
                                <div className={styles.textContent}>
                                    {/* Render pre-translated label */}
                                    <span className={styles.optionLabel}>{item.label}</span>
                                    {/* Render pre-translated description if available */}
                                    {item.description && (
                                        <span className={styles.optionDescription}>{item.description}</span>
                                    )}
                                </div>
                            </button>
                        </li>
                    ))}
                </ul>

                <button
                    className={styles.cancelButton}
                    onClick={handleClose}
                    //   Use translation
                    aria-label={t('cancelButtonAria')}
                >
                    {/*   Use translation */}
                    {t('cancelButtonText')}
                </button>
            </div>
        </div>
    );
};
 
AddOptionsSheet.propTypes = {
    isOpen: PropTypes.bool.isRequired,
    onClose: PropTypes.func.isRequired,
    onSelectItem: PropTypes.func.isRequired,
    items: PropTypes.arrayOf(
        PropTypes.shape({
            id: PropTypes.string.isRequired,
            label: PropTypes.string.isRequired, // Should be pre-translated
            description: PropTypes.string,     // Should be pre-translated
            icon: PropTypes.elementType.isRequired,
        })
    ).isRequired,
};
 
export default AddOptionsSheet;
