import React, { useState, useMemo, useCallback, useEffect } from 'react';
import PropTypes from 'prop-types';
import { Check, Camera, MapPin } from "lucide-react";
import { FormActions } from "../FormActions.jsx";
import { SuccessStep } from "./SuccessStep.jsx";

/**
 * Shared FinalizeStep Component
 * Used across all creation modals for consistent finalization functionality
 */
export function FinalizeStep({
    itemData, // Contains basicInfo and media for preview
    initialLocation,
    onFinalize, // Callback to submit final data
    onBack,
    isLoading = false,
    errors = {},
    isUserLoggedIn = true,
    isSuccess = false, // Prop to indicate if publishing was successful
    onClose, // Prop to close the modal (used by SuccessStep)
    styles, // CSS modules passed from parent modal
    
    // Configurable content
    title = "Finalize Your Listing",
    description = "Review your listing details and add optional location information.",
    publishLabel = "Publish Listing",
    itemType = "listing", // deal, job, service, product, etc.
    
    // Feature flags
    showLocation = true,
    showTerms = true,
    showPreview = true,
    requireTerms = true,
    
    // Success step configuration
    successConfig = {}
}) {
    // --- Local State for Step Fields ---
    const [latitude, setLatitude] = useState(initialLocation?.lat || initialLocation?.latitude || null);
    const [longitude, setLongitude] = useState(initialLocation?.lng || initialLocation?.longitude || null);
    const [address, setAddress] = useState(initialLocation?.address || '');
    const [termsAccepted, setTermsAccepted] = useState(false);
    const [locationLoading, setLocationLoading] = useState(false);
    const [locationError, setLocationError] = useState('');

    // --- Derived State & Validation ---
    const canProceed = useMemo(() => {
        if (requireTerms && showTerms && !termsAccepted) return false;
        return true; // Location is typically optional
    }, [requireTerms, showTerms, termsAccepted]);

    const isPrimaryDisabled = useMemo(() => 
        isLoading || !isUserLoggedIn || !canProceed, 
        [isLoading, isUserLoggedIn, canProceed]
    );

    const primaryLabel = useMemo(() => {
        if (isLoading) return "Publishing...";
        return publishLabel;
    }, [isLoading, publishLabel]);

    // --- Handlers ---
    const handleUseCurrentLocation = useCallback(() => {
        if (navigator.geolocation) {
            setLocationLoading(true);
            setLocationError('');
            navigator.geolocation.getCurrentPosition(
                (position) => {
                    setLatitude(position.coords.latitude);
                    setLongitude(position.coords.longitude);
                    setLocationLoading(false);
                },
                (error) => {
                    console.error("Error getting location:", error);
                    setLocationError(`Error: ${error.message}. Please ensure location permissions are granted.`);
                    setLocationLoading(false);
                },
                { timeout: 10000 }
            );
        } else {
            setLocationError("Geolocation is not supported by your browser.");
        }
    }, []);

    const handleClearLocation = useCallback(() => {
        setLatitude(null);
        setLongitude(null);
        setAddress('');
        setLocationError('');
    }, []);

    const handleSubmit = useCallback((e) => {
        e.preventDefault();
        if (!canProceed || isLoading) return;

        if (requireTerms && showTerms && !termsAccepted) {
            setLocationError('You must accept the terms to publish.');
            return;
        }

        // Pass the location data and other finalization data up to the parent handler
        onFinalize({
            latitude,
            longitude,
            address,
            termsAccepted,
            status: 'active'
        });
    }, [canProceed, isLoading, onFinalize, latitude, longitude, address, termsAccepted, requireTerms, showTerms]);

    // --- Render Logic ---

    // If publishing was successful, render the SuccessStep
    if (isSuccess) {
        return (
            <SuccessStep
                itemData={itemData}
                itemType={itemType}
                onClose={onClose}
                styles={styles}
                {...successConfig}
            />
        );
    }

    // Format price based on item type
    const formatPrice = (price) => {
        if (!price) return 'Price not set';
        const numPrice = typeof price === 'string' ? parseFloat(price) : price;
        if (isNaN(numPrice)) return 'Price not set';
        
        switch (itemType) {
            case 'job':
                return `$${numPrice.toLocaleString()}`;
            default:
                return `€${numPrice.toFixed(2)}`;
        }
    };

    // Otherwise, render the Finalize form
    return (
        <div className={styles.formContainer}>
            <h2 className={styles.formTitle}>{title}</h2>
            <p className={styles.formDescription}>
                {description}
            </p>

            {/* Display Finalize Error Message Passed From Parent */}
            {errors?.finalize && (
                <div className={`${styles.fieldError} ${styles.submitError}`} role="alert">
                    {errors.finalize}
                </div>
            )}

            <form className={styles.form} onSubmit={handleSubmit}>

                {/* --- Item Preview Section --- */}
                {showPreview && (
                    <div className={styles.itemPreview || styles.dealPreview}>
                        <div className={styles.previewHeader}>
                            <h3>Listing Preview</h3>
                        </div>
                        <div className={styles.previewContent}>
                            {/* Display thumbnail or placeholder */}
                            {itemData?.thumbnail || (itemData?.images && itemData.images[0]) ? (
                                <img
                                    src={itemData.thumbnail || itemData.images[0]}
                                    alt={itemData?.name || `${itemType} image`}
                                    className={styles.previewImage}
                                />
                            ) : (
                                <div className={styles.placeholderImage}>
                                    <Camera size={48} aria-hidden="true" />
                                    <span>No image</span>
                                </div>
                            )}
                            {/* Display basic item details */}
                            <div className={styles.previewDetails}>
                                <h4 className={styles.previewTitle}>{itemData?.name || 'N/A'}</h4>
                                <p className={styles.previewPrice}>
                                    {formatPrice(itemData?.basePrice || itemData?.price || itemData?.salary)}
                                </p>
                                <div className={styles.previewMeta}>
                                    <span className={styles.previewCategory}>
                                        {itemData?.categoryName || 'No Category'}
                                    </span>
                                    {itemData?.condition && (
                                        <span className={styles.previewCondition}>
                                            {itemData.condition.charAt(0).toUpperCase() + itemData.condition.slice(1)}
                                        </span>
                                    )}
                                </div>
                            </div>
                        </div>
                    </div>
                )}

                {/* --- Location Section --- */}
                {showLocation && (
                    <>
                        <div className={styles.sectionTitle}>Location Information (Optional)</div>
                        <div className={styles.locationSection}>
                            <p className={styles.locationHint}>
                                Adding your location helps local buyers find your listing.
                            </p>
                            
                            {/* Address Input */}
                            <div className={styles.formGroup}>
                                <label className={styles.formLabel} htmlFor="address-input">
                                    Address
                                </label>
                                <input
                                    id="address-input"
                                    type="text"
                                    className={styles.formInput}
                                    placeholder="Enter address or location"
                                    value={address}
                                    onChange={(e) => setAddress(e.target.value)}
                                />
                            </div>
                            
                            {/* Display location error if geolocation fails */}
                            {locationError && (
                                <div className={`${styles.fieldError} ${styles.locationError}`} role="alert">
                                    {locationError}
                                </div>
                            )}
                            
                            <div className={styles.locationButtons}>
                                <button
                                    type="button"
                                    className={`${styles.locationButton} ${latitude && longitude ? styles.activeLocationButton : ''}`}
                                    onClick={handleUseCurrentLocation}
                                    disabled={locationLoading || (!!latitude && !!longitude)}
                                >
                                    <MapPin size={16} />
                                    {locationLoading ? 'Getting Location...' : 'Use My Current Location'}
                                </button>

                                {/* Show clear button only if location is set */}
                                {(latitude && longitude) && (
                                    <button
                                        type="button"
                                        className={styles.clearLocationButton}
                                        onClick={handleClearLocation}
                                        disabled={locationLoading}
                                    >
                                        Clear Location
                                    </button>
                                )}
                            </div>
                            
                            {/* Confirmation message */}
                            {(latitude && longitude) && !locationLoading && (
                                <div className={styles.locationConfirmation}>
                                    <Check size={16} className={styles.locationCheckIcon} aria-hidden="true"/>
                                    <span>Location Set ({latitude.toFixed(4)}, {longitude.toFixed(4)})</span>
                                </div>
                            )}
                        </div>
                    </>
                )}

                {/* --- Terms & Publish Section --- */}
                {showTerms && (
                    <div className={styles.publishSection}>
                        <div className={styles.termsSection}>
                            <label className={styles.checkboxLabel}>
                                <input
                                    type="checkbox"
                                    className={styles.checkbox}
                                    checked={termsAccepted}
                                    onChange={(e) => setTermsAccepted(e.target.checked)}
                                    required={requireTerms}
                                />
                                <span className={styles.checkboxText}>
                                    I agree to the{' '}
                                    <a href="/terms" target="_blank" rel="noopener noreferrer" className={styles.termsLink}>
                                        Terms of Service
                                    </a>{' '}
                                    and{' '}
                                    <a href="/privacy" target="_blank" rel="noopener noreferrer" className={styles.termsLink}>
                                        Privacy Policy
                                    </a>
                                    {requireTerms && <span className={styles.requiredMark}>*</span>}
                                </span>
                            </label>
                        </div>
                    </div>
                )}

                <FormActions
                    primaryLabel={primaryLabel}
                    primaryIcon="check"
                    secondaryLabel="Back"
                    onCancel={onBack}
                    onPrimaryAction={handleSubmit}
                    isSubmitting={isLoading}
                    isDisabled={isPrimaryDisabled}
                />
            </form>
        </div>
    );
}

FinalizeStep.propTypes = {
    itemData: PropTypes.object,
    initialLocation: PropTypes.shape({
        lat: PropTypes.number,
        lng: PropTypes.number,
        latitude: PropTypes.number,
        longitude: PropTypes.number,
        address: PropTypes.string
    }),
    onFinalize: PropTypes.func.isRequired,
    onBack: PropTypes.func.isRequired,
    isLoading: PropTypes.bool,
    errors: PropTypes.object,
    isUserLoggedIn: PropTypes.bool,
    isSuccess: PropTypes.bool,
    onClose: PropTypes.func.isRequired,
    styles: PropTypes.object.isRequired,
    title: PropTypes.string,
    description: PropTypes.string,
    publishLabel: PropTypes.string,
    itemType: PropTypes.string,
    showLocation: PropTypes.bool,
    showTerms: PropTypes.bool,
    showPreview: PropTypes.bool,
    requireTerms: PropTypes.bool,
    successConfig: PropTypes.object
};
