"use client";
import React, {useCallback, useEffect, useMemo, useState} from 'react';
import {useDispatch} from 'react-redux';
import {useTranslations} from 'next-intl';

// Import extracted components
import FloatingFilterButton from './FloatingFilterButton';
import SidebarHeader from './SidebarHeader';
import ErrorDisplay from './ErrorDisplay';
import GenericFilters from './GenericFilters';
import ActionButtons from './ActionButtons';

// Import hooks and utilities
import { useDebounce } from './hooks/useDebounce';
import {resetFilters, setFilters} from "../../redux/slices/listingFiltersSlice";
import {useFilteredSearch} from '../../hooks/useFilteredSearch';
import {useRenderCount} from '../../utils/performance';
import {getEntityTypeFromContentType, logEntityConversion} from '../../utils/entityMapper';

// Import type-specific filter components
import VideoFilters from './TypeFilters/VideoFilters';
import TweetFilters from './TypeFilters/TweetFilters';
import JobFilters from './TypeFilters/JobFilters';
import ServiceFilters from './TypeFilters/ServiceFilters';
import DealFilters from './TypeFilters/DealFilters';
import ShortFilters from './TypeFilters/ShortFilters';
import VehicleFilters from './TypeFilters/VehicleFilters';
import PropertyFilters from './TypeFilters/PropertyFilters';
import NewsFilters from './TypeFilters/NewsFilters';

// Import FeedContext
import { useFeed } from '../Feed/FeedProvider.client';

import styles from './Leftside.module.css';

/**
 * Leftside component for filtering search results
 * - Uses Redux for filter state management
 * - Optimized with React Query for performance
 * - Implements efficient UI update patterns
 * - Refactored into smaller, reusable components
 */
function Leftside() {
    const t = useTranslations('Filters');
    const dispatch = useDispatch();
    const [open, setOpen] = useState(false);
    const [isMobile, setIsMobile] = useState(false);
    const [errorMessageKey, setErrorMessageKey] = useState('');
    
    // Check if FeedContext is available
    let feedContext;
    try {
        feedContext = useFeed();
    } catch (error) {
        // FeedContext not available, will use fallback behavior
        feedContext = null;
    }

    // Track render count using our performance monitor - DISABLED to prevent excessive re-renders
    // useRenderCount('Leftside');

    // Get filter state and utilities from the custom hook
    const {
        filters: globalFilters,
        updateFilters,
        prefetchResults,
        cancelSearch,
        isLoading,
    } = useFilteredSearch({
        // Enable hook but don't trigger search until Apply is clicked
        enabled: false,
    });

    // Local filters state to prevent excessive API calls
    const [localFilters, setLocalFilters] = useState({...globalFilters});

    // Debounce location and tags inputs to avoid excessive state updates
    const debouncedLocation = useDebounce(localFilters.location, 300);
    const debouncedTags = useDebounce(localFilters.tags, 300);

    // Prefetch results when debounced values change
    useEffect(() => {
        if (debouncedLocation !== undefined && debouncedLocation !== globalFilters.location) {
            prefetchResults({location: debouncedLocation});
        }
    }, [debouncedLocation, globalFilters.location, prefetchResults]);

    useEffect(() => {
        if (debouncedTags !== undefined &&
            JSON.stringify(debouncedTags) !== JSON.stringify(globalFilters.tags)) {
            prefetchResults({tags: debouncedTags});
        }
    }, [debouncedTags, globalFilters.tags, prefetchResults]);

    // Sync local filters when global filters change
    useEffect(() => {
        setLocalFilters(globalFilters);
    }, [globalFilters]);

    // Mobile detection for responsive layout
    useEffect(() => {
        const checkMobile = () => {
            setIsMobile(window.innerWidth <= 768);
        };

        // Use ResizeObserver for more efficient resize detection
        if (typeof ResizeObserver !== 'undefined') {
            const resizeObserver = new ResizeObserver(checkMobile);
            resizeObserver.observe(document.body);
            return () => resizeObserver.disconnect();
        } else {
            // Fallback to window event listeners
            checkMobile();
            window.addEventListener('resize', checkMobile);
            return () => window.removeEventListener('resize', checkMobile);
        }
    }, []);

    // Auto-open on mobile when component mounts
    useEffect(() => {
        if (isMobile) {
            setOpen(true);
        }
    }, [isMobile]);

    // Close sidebar on Escape key
    useEffect(() => {
        if (!open) return;
        const handleKeyDown = (event) => {
            if (event.key === 'Escape') {
                setOpen(false);
            }
        };
        document.addEventListener('keydown', handleKeyDown);
        return () => document.removeEventListener('keydown', handleKeyDown);
    }, [open]);

    // Optimized change handlers with useCallback to prevent rerenders
    const handleChange = useCallback((key, value) => {
        // Special handling for contentType changes
        if (key === 'contentType') {
            // Use centralized entity type mapper
            const entityType = logEntityConversion(value);
            
            // Update local filters - we'll use a unified approach for all entity types
            setLocalFilters(prev => ({
                ...prev, 
                [key]: value,
                // Add the entityType reference
                entityType: entityType
            }));
            
            // Prefetch results for this content type change
            prefetchResults({
                [key]: value,
                entityType: entityType
            });
            return;
        }
        
        // Normal case for other filters
        setLocalFilters(prev => ({...prev, [key]: value}));

        // If it's a simple filter value like sortBy or userType, prefetch results
        if (key === 'sortBy' || key === 'userType') {
            prefetchResults({[key]: value});
        }
    }, [prefetchResults, localFilters]);

    const handleToggle = useCallback((key) => {
        setLocalFilters(prev => ({...prev, [key]: !prev[key]}));
    }, []);

    const handleTagChange = useCallback((e) => {
        const splitted = e.target.value
            .split(',')
            .map((t) => t.trim())
            .filter(Boolean);
        setLocalFilters(prev => ({...prev, tags: splitted}));
    }, []);

    // Simplified validation for basic price range only
    const validateRanges = useCallback(() => {
        const fp = parseFloat;
        const { minPrice, maxPrice } = localFilters;

        if (minPrice && maxPrice && fp(minPrice) > fp(maxPrice)) {
            return 'error_minMaxPrice';
        }
        return '';
    }, [localFilters]);

    // Apply filters and execute search
    const handleApply = useCallback(() => {
        setErrorMessageKey('');
        const errKey = validateRanges();
        if (errKey) {
            setErrorMessageKey(errKey);
            return;
        }
        
        // Ensure content type is properly set in filters
        const updatedFilters = {
            ...localFilters,
            // Initialize content specific filters if needed
            ...(localFilters.contentType && !localFilters[`${localFilters.contentType}Initialized`] ? {
                [`${localFilters.contentType}Initialized`]: true
            } : {})
        };
        
        // Update Redux state
        dispatch(setFilters(updatedFilters));
        
        // If FeedContext is available, update its filters too
        if (feedContext && feedContext.updateFilters) {
            // Use the FeedProvider's unified update approach
            // It will handle entity-specific filtering internally
            feedContext.updateFilters(updatedFilters);
        } else {
            // Use the optimized dispatcher from our custom hook as fallback
            updateFilters(updatedFilters);
        }
        
        setOpen(false);
    }, [localFilters, updateFilters, validateRanges, dispatch, feedContext]);

    // Reset filters to defaults
    const handleClear = useCallback(() => {
        setErrorMessageKey('');

        // Cancel any ongoing searches first
        cancelSearch();

        // Reset Redux filters
        dispatch(resetFilters());
        
        // Reset local filters
        setLocalFilters({});
        
        // If FeedContext is available, update with empty filters or default filters
        if (feedContext && feedContext.updateFilters) {
            // Pass an empty object to reset to default filters
            feedContext.updateFilters({});
        }
        
        setOpen(false);
    }, [dispatch, cancelSearch, feedContext]);

    // Toggle sidebar open/closed
    const toggleSidebar = useCallback(() => {
        setOpen(prev => !prev);
    }, []);

    // Get the appropriate filter component based on content type
    const typeSpecificFilters = useMemo(() => {
        switch (localFilters.contentType) {
            case 'videos':
                return <VideoFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'tweets':
                return <TweetFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'jobs':
                return <JobFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'services':
                return <ServiceFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'deals':
                return <DealFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'shorts':
                return <ShortFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'vehicles':
                return <VehicleFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'properties':
                return <PropertyFilters localFilters={localFilters} handleChange={handleChange}/>;
            case 'news':
                return <NewsFilters localFilters={localFilters} handleChange={handleChange}/>;
            default:
                return null;
        }
    }, [localFilters.contentType, localFilters, handleChange]);

    return (
        <>
            {/* Floating Filter Button for Mobile */}
            <FloatingFilterButton 
                isOpen={open}
                onToggle={toggleSidebar}
                isMobile={isMobile}
            />

            {/* Backdrop - only shown when sidebar is open on mobile */}
            {isMobile && open && (
                <div
                    className={styles.backdrop}
                    onClick={() => setOpen(false)}
                    aria-hidden="true"
                />
            )}

            {/* Sidebar Container */}
            <div
                className={`${styles.sidebarContainer} ${open ? styles.open : ''} ${isMobile ? styles.mobile : ''}`}
                role="dialog"
                aria-modal={isMobile}
                aria-labelledby="filter-sidebar-title"
            >
                <SidebarHeader onClose={() => setOpen(false)} />

                <ErrorDisplay errorMessageKey={errorMessageKey} />

                <div className={styles.scrollContainer}>
                    <GenericFilters 
                        localFilters={localFilters}
                        handleChange={handleChange}
                        handleTagChange={handleTagChange}
                    />

                    {/* Type-specific Filters */}
                    {typeSpecificFilters}
                </div>

                {/* Actions */}
                <ActionButtons 
                    onApply={handleApply}
                    onClear={handleClear}
                    isLoading={isLoading}
                />
            </div>
        </>
    );
}

// Use memo to prevent unnecessary re-renders
export default React.memo(Leftside);
