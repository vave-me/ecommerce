 * TANSTACK QUERY V5.77.0 OPTIMIZED CONFIGURATION
 * Following official documentation: https://tanstack.com/query/v5/docs/framework/react/installation

let browserQueryClient = undefined;
 
/**
 * Creates a new QueryClient instance with optimized defaults
 * @returns {QueryClient} Configured QueryClient instance
 */
function makeQueryClient() {
  return new QueryClient({
    defaultOptions: {
      queries: {
        // Optimized stale time for better UX
        staleTime: 60 * 1000, // 1 minute
        // Reduced garbage collection time for better memory management
        gcTime: 5 * 60 * 1000, // 5 minutes (was cacheTime in v4)
        // Retry configuration
        retry: (failureCount, error) => {
          // Don't retry on 4xx errors except 408, 429
          if (error?.status >= 400 && error?.status < 500 && ![408, 429].includes(error.status)) {
            return false;
          }
          // Retry up to 3 times for other errors
          return failureCount < 3;
        },
        retryDelay: (attemptIndex) => Math.min(1000 * 2 ** attemptIndex, 30000),
        // Network mode for better offline handling
        networkMode: 'online',
        // Refetch configuration
        refetchOnWindowFocus: false, // Disabled for better UX
        refetchOnReconnect: 'always',
        refetchOnMount: true,
      },
      mutations: {
        // Retry failed mutations once
        retry: 1,
        // Network mode for mutations
        networkMode: 'online',
      },
    },
  });
}
 
/**
 * Server-side QueryClient factory
 * Creates a new instance for each request to prevent data contamination
 * @returns {QueryClient} Fresh QueryClient instance for server-side use
 */
export function getQueryClient() {
  // Server: always make a new query client
  if (typeof window === 'undefined') {
    return makeQueryClient();
  }
 
  // Browser: make a new query client if we don't already have one
  // This is very important, so we don't re-make a new client if React
  // suspends during the initial render. This may not be needed if we
  // have a suspense boundary BELOW the creation of the query client
  if (!browserQueryClient) browserQueryClient = makeQueryClient();
  return browserQueryClient;
}

/**
 * Invalidates queries by entity type with improved error handling
 * @param {QueryClient} queryClient - The query client instance
 * @param {string} entityType - Type of entity (posts, products, etc.)
 * @param {string|number} [entityId] - Optional specific entity ID
 * @returns {Promise<void>}
 */
export async function invalidateEntityQueries(queryClient, entityType, entityId = null) {
  try {
    const queryKey = entityId ? [entityType, entityId] : [entityType];
    
    await queryClient.invalidateQueries({
      queryKey,
      exact: false, // Invalidate all queries that start with this key
      refetchType: 'active', // Only refetch active queries
    });
    
    console.log(`✅ Invalidated ${entityType} queries${entityId ? ` for ID: ${entityId}` : ''}`);
  } catch (error) {
    console.error(`❌ Failed to invalidate ${entityType} queries:`, error);
    // Don't throw - invalidation failures shouldn't break the app
  }
}

/**
 * Optimistic update helper with rollback capability
 * @param {QueryClient} queryClient - The query client instance
 * @param {Array} queryKey - Query key to update
 * @param {Function} updateFn - Function to update the data
 * @returns {Function} Rollback function
 */
export function optimisticUpdate(queryClient, queryKey, updateFn) {
  // Cancel any outgoing refetches
  queryClient.cancelQueries({ queryKey });

  // Snapshot the previous value
  const previousData = queryClient.getQueryData(queryKey);

  // Optimistically update to the new value
  queryClient.setQueryData(queryKey, updateFn);

  // Return a rollback function
  return () => {
    queryClient.setQueryData(queryKey, previousData);
  };
}
 
/**
 * Batch invalidation for multiple entity types
 * @param {QueryClient} queryClient - The query client instance
 * @param {Array<string>} entityTypes - Array of entity types to invalidate
 * @returns {Promise<void>}
 */
export async function batchInvalidateQueries(queryClient, entityTypes) {
  try {
    const invalidationPromises = entityTypes.map(entityType =>
      queryClient.invalidateQueries({
        queryKey: [entityType],
        exact: false,
        refetchType: 'active',
      })
    );

    await Promise.allSettled(invalidationPromises);
    console.log(`✅ Batch invalidated queries for: ${entityTypes.join(', ')}`);
  } catch (error) {
    console.error('❌ Failed to batch invalidate queries:', error);
  }
}
